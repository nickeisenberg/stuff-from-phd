\begin{frame}[t]
\frametitle{Gaussian Processes}

A Gaussian Processes, $G := \{G_t : t \in T\}$ is a stochastic process such that for and finite set of $t_1,...,t_n \in T$, the random vector $(G(t_1),...,G(t_n))$ follows a $n$-dimensional Gaussian distribution. The processes $G$ is uniquely determined by
\begin{enumerate}
		\item $\mu(t) = \E(G(t))$
		\item $C(s,t) = \E(G(t)G(s))$
\end{enumerate}

\begin{theorem}[Important]
	The space of all symmetric and non-negative definite functions, $f(s,t)$, on $T\times T$ matches the space of all covariance functions of all Gaussian processes on $T$. Thus, given a non-negative definite function on $T \times T$ then there exists a Gaussian Processes that possesses this covariance function.
\end{theorem}
\end{frame}

\begin{frame}[t]
\frametitle{Examples of Gaussian Processes}
\begin{enumerate}
	\item \textbf{Brownian Motion:} Consider the following mean and covariance functions on $T=[0, \infty)$.
			\begin{itemize}
				\item $\mu(t) = 0$
				\item $C(s,t) = \min(s,t)$
			\end{itemize}
		It can be shown that $C$ is non-negative definite and the resulting Gaussian Processes is referred to as Brownian Motion, $B= \{ B(t) : t \ge 0 \}$.

	\item \textbf{White Noise:} Consider the following mean and covariance functions on $T=\mathscr{B}(\R^N)$, the Borel sets of $\R^N$.
		\begin{itemize}
			\item $\mu(A) = 0$
			\item $C(A,B) = \lambda^N(A \cap B)$, $\lambda^N$ = $N$-dimensional Lebesgue measure.
		\end{itemize}
	Again it can be shown that $C$ is non-negative definite and thus the resulting Gaussian processes on $T$ is referred to as White Noise, $\dot{W} = \{ \dot{W}(A) : A \in \mathscr{B}(\R^N)  \}$.
\end{enumerate}
\end{frame}

\begin{frame}[t]
	\frametitle{Brownian Motion in terms of white noise}
	\begin{itemize}
		\item	Consider a white noise, $\dot{W} = \{\dot{W}(A):A \in \mathscr{B}(\R) \}$
		\item Define $B(t) = \dot{W}([0,t])$
		\item Since $\dot{W}$ is a white noise we have $\mathbb{E}(B_sB_s) = \mathbb{E}(\dot{W}[0,t]\dot{W}[0,s]) = \lambda([0,t]\cap[0,s]) = \min\{s,t\}$
		\item This shows that $B$ is a Brownian Motion 
	\end{itemize}
\end{frame}

\begin{frame}[t]
	\frametitle{White Noise as an Ito Integral}
	We define a set function on $\mathscr{B}(\R)$
	\[
	\dot{W}(A) = \int_0^\infty 1_A(t) \ud B(t,\omega)
	\]
	For any borel set, $A$, the deterministic indicator function $1_A \in L^2(\mathscr{P})$ and so the Ito Integral exists.
	
	From the theorem in the previous slide we see that
	\begin{itemize}
		\item $\E\dot{W}(A) = 0 $
		\item $ \E(\dot{W}(A) \dot{W}(B) = \int_0^\infty 1_A(t)1_B(t) \ud t = \lambda (A \cap B) $
	\end{itemize}
	Thus, $\dot{W}$ is a white noise on $\mathscr{B}(\R)$
	
\end{frame}

\begin{frame}[t]
	\frametitle{Ito Integral}

Wfe will briefly explain the meaning of $\int_0^\infty u(t,\omega) \ud B(t,\omega)$, the Ito Integral of $u$, where $B$ is a Brownian Motion.

\begin{definition}[Progressively Measurable]
	A processes, $u(t,\omega)$, is progressively measurable if $u$ restricted to $\Omega \times [0,t]$ is $\mathcal{F}_t \times \mathscr{B}([0,t])$ measurable where $\mathcal{F}_t = \sigma\{B(s) : 0 \le s \le t \}  $.
\end{definition}

\begin{definition}[Progressively Measurable Hilbert space, $L^2(\mathscr{P})$]
We define $L^2(\mathscr{P}) = L^2(\Omega \times [0, \infty), \mathscr{P}, \mathbb{P}\times \lambda ])$ where $\mathscr{P}$ is the sigma-alegebra generated by all the sets $A \subset \Omega \times [0, \infty)$ such that $1_A(t,\omega)$ is progressively measurable. The norm of this Hilbert Space is $\Norm{u}_{\mathscr{P}}^2 = \E \left( \int_0^\infty u(t,\omega)^2 \; \ud t \right)$
\end{definition}

\begin{definition}[Simple Processes and their Ito Integral]
	A process of the form $u(t,\omega) = \sum_{k=1}^n \phi_k(\omega) 1_{(t_k,t_{k+1}]}(t)$ where $\phi_k$ is $\mathcal{F}_k$ measureable is called a simple process. Its Ito integral is defined to be $\int_0^\infty u(t,\omega) \ud B(t,\omega) = \sum_{k=1}^n \phi_k(\omega)(B(t_{k+1},\omega)-B(t_k, \omega))$
\end{definition}
\end{frame}

\begin{frame}[t]
	\frametitle{Ito Integral}
\begin{theorem}
	The simple processes are dense in $L^2(\mathscr{P})$. Thus for any processes, $u(t,\omega)$, there exists a sequence of simple processes, $u_n(t,\omega)$ such that $\E \left( \int_0^\infty (u(t,\omega) - u_n(t,\omega))^2 \ud t \right) \to 0$.
\end{theorem}

\begin{definition}[Ito Integral of a $L^2(\mathscr{P})$ processes]
	For any processes $u(t,\omega) \in L^2(\mathscr{P})$ we define its Ito Integral to be
	\[\int_0^\infty u(t,\omega) \ud B(t,\omega) := \lim_{n \to \infty} \int_0^\infty u_n(t,\omega) \ud B(t, \omega)
	\]
	where the limit is with respect to the $L^2(\mathscr{P})$ norm and the $u_n's$ are from the Theorem.
\end{definition}

\begin{theorem}
	\begin{enumerate}
		\item $ \E \int_0^\infty u(t,\omega) \ud B(t,\omega) = 0$
		\item $ \E\left( \int_0^\infty u(t,\omega) \ud B(t,\omega)\cdot \int_0^\infty v(t,\omega) \ud B(t,\omega) \right) = \E \left( \int_0^\infty u(t,\omega)v(t,\omega) \ud t \right) $
	\end{enumerate}
\end{theorem}
\end{frame}

\begin{frame}[t]
	\frametitle{Walsh Integral}
	\[
		\int_0^t\int_{\R^d}f(s,y)W(\ud s, \ud y) 
	\]
\begin{itemize}
	\item Generalization of the Ito Integral
	\item Defined for two parameter worthy martingale measure integrators.
		\begin{itemize}
			\item The white noise processes $	W_t(A) = \dot{W}([0,t] \times A)$ is "worthy" and is used in the $1$-$d$ SHE and SWE. 
			\item A "colored" Gaussian noise will be used in higher dimensions
		\end{itemize}  
	\item Used to define the solution to the SHE and SWE for two parameter noise forcing terms. 
\end{itemize}

\end{frame}

\begin{frame}[t]
	\frametitle{Walsh Integral Construction}
	
\begin{itemize}
	\item Simple functions: Let $\mathscr{L}$ be class of linear combinations of processes of the form $f(t,x,\omega) = X(\omega)1_{(a,b]}(t)1_A(x)$ where $X \in \mathcal{F}_a$ and let $\mathscr{P} = \sigma(\mathscr{L})$
	
	\item The integral for a simple processes is
	\[
		(f\cdot M)_t(B) = X(\omega)\left[M_{b \wedge t}(A \cap B) - M_{a \wedge t}(A \cap B)\right]
	\]
\end{itemize}

	\begin{theorem}
		For $\mathscr{P}$ measurable processes, $f(t,x,\omega)$ that satisfy 
	\[
		\int_0^T\iint_{\R^{2n}}|f(x,t)f(y,t)|K(\ud x, \ud y, \ud t) < \infty
	\]
	there exists a sequence of simple processes whose integrals are Cauchy in $L^2(\mathscr{P})$. The limit is the walsh integral of $f$. 
	\end{theorem}



\end{frame}

\begin{frame}[t]
	\frametitle{Colored Noise}
	
\begin{itemize}
	\item For non-negative and non-negative definite measure $\Gamma$ define on $\mathscr{S}(\R^d)$
	\begin{align*}
	C(\varphi, \psi) &= \int_0^\infty \ud s \int_{\R^d} \Gamma(\ud x) (\varphi(s,\cdot)*\tilde{\psi}(s,\cdot))(x)
	\\& = \int_0^\infty \ud s \int_{\R^d} \ud x \int_{\R^d} \ud y \varphi(s,x)f(x-y)\psi(s,y)
	\end{align*}
	
	\item $C$ is non-negative definite and the resulting $0$-mean Gaussian processes is referred to as a colored Gaussian noise, $F = \{F(\varphi): \varphi \in \mathscr{S}(\R^d) \}$. 
	
	\item Bochner's Theorem implies the existence of non-negative measure $\mu$ (spectral density) such that 
	\[
		\int \Gamma(\ud x) \varphi(x) = \int \mu(\ud x) \mathcal{F}\varphi(x)
	\]
\end{itemize}

\end{frame}

\begin{frame}[t]
	\frametitle{Walsh integral with Colored Noise Integrator}
	
	\begin{itemize}
		\item Take $\varphi_n \in \mathscr{S}(\R^d)$ such that $\varphi_n \to 1_A$
		\item Can be shown $\E[(F(\varphi_n) - F(\varphi_m))^2] \to 0$. Call the limit $F(A)$.
		\item $M_t(A) := F([0,t] \times A)$ is worthy and can be used as an integrator. 
		\item Through the construction we can show that 
		\[
		F(\varphi) = \int_0^\infty \int_{\R^d} \varphi(s,x) M(\ud x, \ud s)
		\]
		\item The Walsh integral for integrator $M$ can be defined for processes satisfying
			\[
		\int_0^\infty \ud s \int_{\R^d} \ud x \int_{\R^d} \ud y \varphi(s,x) f(x-y) \varphi(s,y) < \infty
		\] 
	\end{itemize}
\end{frame}

\begin{frame}[t]
	\frametitle{Stochastic Heat Equation}
	\[
	 \begin{cases}
	\dfrac{\partial u}{\partial t}(t,x)- \frac{1}{2}\Delta_xu(t,x) = \sigma(u(t,x)\dot{M}(t,x) &
	\\ u(0,\cdot) = \mu(\cdot)  &
	\end{cases}
	\]
	
\begin{itemize}
	\item The rigorous meaning of the SHE is 
	\[
	u(t,x) = \int_{\R^d} G(t,x-y) \mu(\ud y) + \int_0^t \int_{\R^d}G(t-s,x-y)\sigma(u(s,y))M(\ud s, \ud y)
	\]
	\item Informally the $\dot{M}$ is acting like the density (does not exist) of $M$ in the sense that 
	\[
	\int_0^t \int_{\R^d}\varphi(s,y) M(\ud s, \ud y) = \int_0^t \int_{\R^d}\varphi(s,y)\dot{M}(s, y)\ud s \ud y
	\]
\end{itemize}
\end{frame}

\begin{frame}[t]
	\frametitle{Definition of Invariant Measures}
	Consider an arbitrary measure space $(E,\mathscr E)$ and denote $\mathcal M_1(E)$ to be the space of all probability measure on $(E,\mathscr E)$.
	\begin{definition}[Markovian Transition Function]  We say that $P_t(x,\Gamma)$, $t>0$, $x\in E$, $\Gamma \in \mathcal E$, is a Markovian Transition Function is:

		\begin{enumerate}[(i)]
			\item $P_t(x,\cdot)$ is a probability measure on $(E,\mathscr E)$ for $t \ge 0$, $x \in  E$
			\item $P_t(\cdot,\Gamma)$ is a $\mathscr E$-measurable function for $t \ge 0$ and $\Gamma \in \mathscr E$
			\item $P_{t+s}(x,\Gamma) = \int_EP_t(x,\ud y)P_s(y,\Gamma)$
			\item $P_0(x,\Gamma)=1_\Gamma(x)$ where $1_\Gamma(x)$ is the indicator function.
		\end{enumerate}
	\end{definition}

\end{frame}

\begin{frame}[t]
	\frametitle{Invariant Measures viewed as an average}
	\begin{definition} [Invariant Measure]
		A measure $\eta(\cdot) \in \mathcal M_1(E)$ is invariant for transition functions $\{P_t(x,\Gamma)\} _{t\ge 0}$ if for any $\Gamma \in \mathscr E$ we have that $$ \eta(\Gamma) = \int_E P_t(x,\Gamma)\eta(\ud x) \quad \text{for any $t\ge 0$} $$
	\end{definition}

	Recall from Calculus that the average of a function $f:[a,b] \to \R$ is defined as
	$$ \text{AVG}(f) = \dfrac{1}{a-b}\int_a^b f(x) \ud x$$
	 Thus,  $\eta(\Gamma)$ is the average of the function $ x \mapsto P_t(x,\Gamma)$ over  $(E,\mathscr E, \eta)$.

\end{frame}

\begin{frame}[t]
	\frametitle{Main Result}
	The laws of the solution the the SHE defines Markovian transition functions
	\[
	P_t(\varphi , \Gamma) = \mathscr L (u^\varphi(t,\cdot))(\Gamma) = P[\omega \in \Omega : u^\varphi(t,\cdot)\in \Gamma] \quad \Gamma \in \mathscr B (L^2_\rho)
	\]
	where $u^\varphi(t,x)$ is the solution to the SHE starting at $\varphi \in L^2_\rho$. We have the following main results
	\begin{theorem}
		Under certain conditions of the spectral density $\hat{f}$ and on the admissible weight $\rho(x)$, there exists a measure $\eta$ that is invariant for the solution to the stochastic heat equation for any $L^2_\rho$ initial condition. In other words, for $P_t(\varphi,\Gamma)$ we have that, $ \eta(\Gamma) = \int_{L^2_\rho(\R^d)}P_t(x,\Gamma)\eta( \ud x) $ for $ t\ge T >0$ for any $T>0$.
	\end{theorem}

	\begin{theorem}
		Under the same assumptions as the theorem above, there exists a measure $\eta$ that is invariant for the solution to the stochastic heat equation with the $\delta_0(x)$ initial condition.
	\end{theorem}

\end{frame}

		\item Existence of Invariant Measures for a the SHE with a broader range of initial conditions, in particular measure valued initial conditions such as $\delta_0$.

\begin{frame}[t]
	\frametitle{Current Project: Fractional SPDE}
	\begin{equation}
	\begin{cases}
		\left(\partial^b + \frac{\nu}{2}(-\Delta)^{a/2}\right)u(t,x) = I^r_t \left[\sqrt{\theta}u(t,x)\dot W(x) \right] & \text{$x\in \R^d$ , $t>0$} \\
		u(0,\cdot) = 1                                                                                           & \beta \in (0,1]            \\
		u(0,\cdot) = 1, \quad u_t(0,\cdot) = 0                                                                   & \beta \in (1,2)
	\end{cases}  \label{fspde}
	\end{equation}
	
	\begin{itemize}
		\item $a \in (0,2]$, $b \in (0,2)$, $\nu>0$, $r>0$ and $\theta >0$
		\item $(-\Delta)^{a/2}$ refers to the fractional Laplacian \cite{FPDE} 
		\item $\partial^b$ refers to the Caputo derivative \cite{PodFDE}
		\item $I^r_t$ is the Riemann-Liouville fractional integral operator.
		\item The time independent noise $W:= \{ W(\varphi) : \varphi \in \mathcal{D}(\R^d) \}$ arises from the covariance function 
		\[
	  \mathbb{E}(W(\varphi)W(\psi)) = 	\langle \varphi, \psi \rangle_{\mathcal{H}} =  \int_{\R^d}\mathcal{F}\varphi(\xi)\overline{\mathcal{F}\psi}(\xi) \mu(\ud \xi)
		\]
	\end{itemize}

\end{frame}

\begin{frame}[t]
	\frametitle{Fractional Laplacian:}
	\begin{itemize}
		\item \textbf{Fractional Laplacian:} One way to define the fractional Laplacian is through the Fourier Transform. It can be shown that 
		\[
			\mathcal{F}(-\Delta f)(\xi) = K |\xi|^2\mathcal{F}f(\xi)
		\]
		where $K$ depends on the definition of the Fourier transform. So for $\psi \in \mathscr{S}(\R^d)$ we define 
		\[
			\mathcal{F}(\psi(-\Delta)f)(\xi) = \psi(K|\xi|^2)\mathcal{F}f(\xi)
		\]
		and in particular we can show that if we want to solve 
		\[
			(-\Delta^{\alpha/2})u = f \implies u = 	(-\Delta^{-\alpha/2})f
		\]
		then $u$ is given by $f*R_\alpha$ where $R_\alpha$ is the Reisz Kernal.
	\[
		R_\alpha(x) = \frac{\Gamma((1/2)(d-\alpha)}{\Gamma((1/2)\alpha)2^{\alpha}\pi^{d/2}|x|^{d-\alpha}}
	\]
	\end{itemize}
\end{frame}

\begin{frame}[t]
	\frametitle{Fundamental solution to the deterministic FPDE}
		\begin{equation}
	\begin{cases}
	\left(\partial^b + \frac{\nu}{2}(-\Delta)^{a/2}\right)u(t,x) = I^r_t \left[f(t,x) \right] & \text{$x\in \R^d$ , $t>0$} \\
	u(0,\cdot) = \mu                                                                                           & \beta \in (0,1]            \\
	u(0,\cdot) = \mu_0, \quad u_t(0,\cdot) = \mu_1                                                                  & \beta \in (1,2)
	\end{cases}  \label{fpde}
	\end{equation}
	
	\[
	u(t,x) = \begin{cases} 
	(Z(t,\cdot)*\mu)(x)  + (Y \star f)(t,x)   &   \beta \in (0,1]\\
	(\frac{\partial Z}{\partial t}(t,\cdot)*\mu_0)(x) + 	(Z(t,\cdot)*\mu_1)(x) + (Y \star f)(t,x) & \beta \in (1,2)
	\end{cases}	
	\]
\begin{itemize}
	\item $ Z(t,x) = \pi^{-d/2}t^{\Ceil{\beta} -1}|x|^{-d} H^{2,1}_{2,3}\left(\frac{|x|^\alpha}{2^{\alpha -1} \nu t^\beta} \bigg\vert \begin{matrix} (1,1) & (\Ceil{\beta}, \beta)
	\\ (d/2,\alpha/2) & (1,1) & (1,\alpha /2 )\end{matrix} \right) $
	
	\item $Y(t,x) = {}_{t}D_{+}^{\Ceil{\beta}-\beta-\gamma}Z(t,x)$
\end{itemize}	
	
\end{frame}

\begin{frame}[t]
	\frametitle{Reason for constant 1 initial conditions}
	
	It can be shown that for the above initial conditions that 
	
	\begin{itemize}
		\item 	$(Z(t,\cdot)*1)(x) = 1$ when $\beta \in (0,1]$
		\item  $ (\frac{\partial Z}{\partial t}(t,\cdot)*1)(x) =1$ when $\beta \in (1,2) $
	\end{itemize}

In addition, it can be shown that 
\[
	\mathcal{F}Y(ct,\cdot)(\xi) = c^{b+r -1}\mathcal{F}Y(t,\cdot)(c^{b/a} \xi). 
\]

where as for the Wave equation we have 

\[
	\mathcal{F}G(ct, \cdot)(\xi) = c \mathcal{F}G(t,\cdot)(c\xi) 
\]

and because of these similar scaling properties, we can almost treat $Y$ and $G$ as the same. 
\end{frame}

\begin{frame}[t]
\frametitle{FSPDE Integral Equation}
\begin{itemize}
	\item The time independence leads us to rigoursly defing the FSPDE through the following Skorohod integral equation 
	\[
	u(t,x) = 1 + \sqrt{\theta}\int_0^t \left( \int_{\R^d} Y(t-s,x-y)u(s,y) W(\ud y)  \right) \ud s
	\]
	\item Through Piccard iteration we show that 
	\[
	u(t,x) = 1 + \sum_{n \ge 1} \theta^{n/2}I_n(f_n(\cdot,x,t))
	\] 
	where
	\[
	f_n(x_1,...,x_n,x,t) = \int_{[0,t]^n<} G(t-t_n,x-x_n)\cdots G(t_2-t_1,x_2-x_1) \ud \textbf{t}
	\]
\end{itemize}

\end{frame}
