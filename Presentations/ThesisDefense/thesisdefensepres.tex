	\begin{frame}{The stochastic heat equation}
		\[
		\begin{cases}
		\dfrac{\partial u}{\partial t}(t,x)- \frac{1}{2}\Delta_xu(t,x) = b(x, u(t,x))\dot{W}(t,x), & t \ge 0, \: x \in \R^d
		\\ u(0,\cdot) = \mu(\cdot)  &
		\end{cases} \vspace{.1in}
		\]

\rdot The Stochastic Heat Equation is interpreted as follows: \vspace{0in}
\begin{align*}
u(t,x) &= \int_{\R^d} G(t,x-y) \mu(\ud y) + \int_0^t \int_{\R^d}G(t-s,x-y)b(u(s,y))W(\ud s, \ud y) 
\end{align*}
where $G(t,x)$ is the heat kernel \vspace{0in}
\[
G(t,x) = (2 \pi t)^{-d/2} \exp\left(- \frac{|x|^2}{2t} \right) \vspace{.1in}
\]
\rdot A processes, $\{u(t,x) : t \ge 0, \; x\in \R^d \}$, that satisfies this relation is referred to as the {\it mild solution}.


\end{frame}

\begin{frame}[t]{The noise term $\dot{W}$}
\rdot $W = \{W(\phi): \: \phi \in \mathscr{S}(\R^d) \}$ is a centered Gaussian process uniquely defined through the following covariance functional: 
\[
	J(\phi, \psi) := \mathbb{E}\left[W(\phi)W(\psi)\right] = \int_0^\infty \ud s \int_{\R^d} \Gamma(\ud x) (\phi(s,\cdot)*\tilde{\psi}(s,\cdot))(x),
\]
where $\tilde{\psi}(s,x) = \psi(s,-x)$. We have the following nomenclature. \vspace{.25in} 
	
 \rdot {\em Space-time white noise}: $\Gamma(\ud x) = \delta_0(\ud x)$. \vspace{.15 in}
 
 \rdot {\em Time white and spatially colored:} $\Gamma(\ud x) = f(x)\ud x$ where $f$ is a non-negative and non-negative definite function referred to as the {\em Correlation function} and its Fourier transform,
 	\[
 		\widehat{f}(\xi) := \mathcal{F}f(\xi) = \int_{\R^d} \exp\left(-i x \cdot \xi\right) f(x) \ud x,
 	\]
 is referred to as the {\em spectral density}.
\end{frame}

		A space-time white noise, $dB$, is the distributional derivative of a Brownian sheet, $B(t,x)$.
	\end{proposition}

\begin{frame}{Existence and uniqueness with space-time white noise}
	\rdot A necessary condition for global existence, {\em Dalang's condition} \cite{Dal99}, is the following:
	\[
		\int_0^t \ud s \int_{\R^d} \ud y \: G^2(s,y) < \infty, \quad \forall t > 0.
	\] 
	\rdot Direct calculations yields that,
	\[
		\int_0^t \ud s \int_{\R^d} \ud y \: G^2(s,y) = 
		\frac{\Gamma(d/2)}{2(2\pi)^d} \int_0^t \frac{1}{s^{d/2}} \ud s,
	\]
	which forces,
	\[
		d/2 < 1 \quad \iff \quad d < 2.
	\]
	\rdot Thus a global space-time white noise solution may only exist when the spatial dimension is $1$. 
\end{frame}

\begin{frame}{Will the second moments ever be bounded in time?}
	\rdot Several slides below will require the second moments to be bounded in time:
		\[
			\sup_{t\ge0} \Norm{u(t,x)}_2 < \infty, \qquad \text{where } \Norm{X}_p = \left[\mathbb{E}\left(|X|^p\right)\right]^{1/p}.
		\]
	
	\rdot This is possible, however this will require us to only consider $d \ge 3$, and therefore we must only consider a spatially colored noise. \vspace{.2in}
	
	\rdot Recall from above, that the covariance of a spatially colored noise is given by the following:
	\begin{align*}
 \mathbb{E}\left[W(\phi)W(\psi)\right] &= \int_0^\infty \ud s \int_{\R^d} \ud x \: f(x) (\phi(s,\cdot)*\tilde{\psi}(s,\cdot))(x) \\
 &= \int_0^\infty \ud s \int_{\R^{2d}} \ud x \ud y \: \phi(s,x)f(x-y)\psi(s,y) \\
 &= (2\pi)^{-d} \int_0^{\infty} \ud s \int_{\R^d} \ud \xi \: \widehat{f}(\xi) \mathcal{F}\phi(s,\xi) \overline{\mathcal{F}\psi(s,\xi)}
 \end{align*}
\end{frame}
	
\begin{frame}{$u_t - \frac{1}{2}u_{xx} =  b(u) \dot{W}$ with $u(0,\cdot) = \mu(\cdot)$ \hfill (colored noise)}
\begin{theorem}[Chen and Huang {\cite[Theorem 1.2]{CH19}}]
There exists a unique global mild solution under the following:
\end{theorem} \vspace{.1in}

\rdot $b: \R^d \times \R \to \R$ such that there exists a $L_b,L_0 >0$ so for all $u,v \in \R$ and $x \in \R^d$,
\[
|b(x,u)-b(x,v)| \le L_b|u-v| \quad \text{and} \quad  |b(x,0)| \le L_0.
\] \vspace{0in}

\rdot The initial measure, $\mu(\ud x)$ (resp. $\varphi(x) \ud x$), satisfies for some $a > 0$,
\[
\int_{\R^d} e^{-a|x|^2} |\mu|(\ud x) < \infty \quad \left(\text{resp. } \int_{\R^d} e^{-a|x|^2} |\varphi(x)|\ud x < \infty \right),
\] 
where $|\mu| = \mu^+ + \mu^-$ is the {\em Hahn decomposition}. \vspace{.2in}

\rdot The spectral density, $\widehat{f}$, satisfies {\it Dalang's condition for some $\beta >0$:}
\[
\Upsilon(\beta) : = \int_{\R^d} \frac{\hat{f}(\ud \xi)}{\beta + |\xi|^2} < \infty.
\]

\end{frame}

\begin{frame}{Transition functions and their semigroup}
\begin{definition}[Markovian Transition Functions]
		\rdot Let $H$ be a Hilbert space and $\mathscr{B}(H)$ be the borel sets of $H$. \\
		\rdot $\{ P_t(x,\Gamma): t \ge 0, x \in H, \Gamma \in \mathscr{B}(H) \}$ forms a family of {\em Markovian transition functions} if,
		\begin{enumerate}
			\item $x \mapsto P_t(x, \Gamma)$ if measurable,
			\item $\Gamma \mapsto P_t(x, \Gamma)$ is a probability measure on $H$,
			\item $P_{t+s}(x,\Gamma) = \int_{H} P_t(x,\ud y) P_s(y, \Gamma)$,
			\item $P_0(x,\Gamma) = \textbf{1}_{\Gamma}(x)$.
		\end{enumerate}
\end{definition}

\begin{definition}[Transition Semigroup]
	The above transition functions form the following semi group on $\mathcal{B}(H)$, which denotes the bounded Borel measurable functions on $H$:
	\[
		P_t\varphi(x) = \int_H \varphi(y) P_t(x, \ud y), \qquad \varphi \in \mathcal{B}(H).
	\]
\end{definition}

\end{frame}

\begin{frame}{Invariant measures}
	\begin{definition}[Invariant measures]
		\rdot A probability measure, $\eta$, on $H$ is said to be {\em invariant} for the transition functions $P_t$ if for all $\Gamma \in \mathscr{B}(H)$ and $t > 0$,
		\[
		P_t^*(\eta)(\Gamma) := \int_H P_t(x,\Gamma) \eta(\ud x) = \int_H P_0(x,\Gamma) \eta(\ud x) = \int_H \textbf{1}_{\Gamma}(x) \eta(\ud x) = \eta(\Gamma),
		\]
		which is equivalent to,
		\[
			\int_H P_t\varphi(x) \eta(\ud x) = \int_H \varphi(x) \eta(\ud x), \qquad \text{for all } \varphi \in \mathcal{B}(H). 
		\]
	\end{definition}

\rdot This implies that the $\eta$-average of $x \mapsto P_t(x,\Gamma)$ is $\eta(\Gamma)$ and it is independent of time:
\[
	\text{AVG}_{\eta}\big[P_t(\cdot,\Gamma)\big] = \frac{1}{\eta(H)}\int_H P_t(x,\Gamma) \eta(\ud x) = \int_H P_t(x,\Gamma) \eta(\ud x) = \eta(\Gamma).
\]
\end{frame}

\begin{frame}{Invariant measures for the stochastic heat equation}
	\rdot First we define the law of the solution as follows:
	\[
		\mathscr{L}(u(t,\cdot;\zeta))(\Gamma) = \mathbb{P}\left[\omega \in \Omega : u(t,\cdot;\zeta) \in \Gamma\right], \quad \text{for all } \zeta \in H \text{ and } \Gamma \in \mathscr{B}(H),
	\]
	and for the mean time we are intentionally vague about $H$. \vspace{.2in} \\
	
	\rdot Our intention is for 
		\[
			P_t(\zeta,\Gamma) := \mathscr{L}(u(t,\cdot;\zeta))(\Gamma)
		\]
		to form a Markovian semigroup. \vspace{.2in} \\

	\rdot Established theory by DaPrato and Zabczyk \cite{DZ14} requires the solution, 
	\[
	\zeta \mapsto u(t,\cdot;\zeta),
	\]
	 to be almost surely $H$-valued for $t > 0$ and $\zeta \in H$. \vspace{.2in} \\

	\rdot So what is $H$? 
	
\end{frame}

\begin{frame}{$H = L^2_{\rho}(\R^d)$: weighted square integrable functions}
	\begin{definition}[Admissible Weight]
		A positive, bounded and continuous $L^1(\R^d)$ function, $\rho$, is an {\em admissible weight} if 
			\[
				\bigg( G(t,\cdot)*\rho(\cdot) \bigg)(x) \le C_{\rho} \rho(x), \qquad \text{for all } t \in [0,T].
			\] 
	\end{definition}

	\begin{definition}[$L^2_{\rho}(\R^d)$]
		The space $H := L^2_{\rho}(\R^d)$ is the space of $\rho$-weighted square integrable functions:
		\[
			\Norm{\zeta}^2_{\rho} := \int_{\R^d} |\zeta(x)|^2 \rho(x) \ud x < \infty.
		\]
		Moreover, $H$ is a Hilbert space with inner product defined as 
		\[
			\langle \zeta, \psi \rangle_{\rho} := \int_{\R^d} \zeta(x) \psi(x) \rho(x) \ud x.
		\]
	\end{definition}
\end{frame}

	\[

	\]
	
	\rdot One can show that the following is not admissible for any $a>0$,
	\[
		\exp\left(-a|x|^2\right),

\begin{frame}{The solution is almost surely $H$-valued.}
	\begin{theorem}

Let $\rho$ be an admissible weight and suppose that
\[
	\Upsilon(0) = \int_{\R^d} \frac{\widehat{f}(\xi)}{|\xi|^2} \ud \xi < \infty \qquad \text{and} \qquad L_b^2 < \frac{1}{128 \Upsilon(0)}.
\]
Now suppose our initial measure, $\mu$, satisfies the following:
\[
	\mathcal{G}_{\rho}(t;\mu) := \int_{\R^d} \left( \int_{\R^d} G(t,x-y) (1+|\mu|)(\ud y) \right)^2 \rho(x) \ud x < \infty.
\]
Then for all $t > 0$,
\[
	\mathbb{E}\left( \Norm{u(t,\cdot;\mu)}^2_{\rho} \right) \le C\mathcal{G}_{\rho}(t;\mu) < \infty.
\]
In particular, for any $\zeta \in H$ and for all $T>0$, there exists a $C_T$ such that
	\[
			\mathcal{G}_{\rho}(t;\zeta) \le 2\Norm{\rho}_{L^1(\R^d)} + 2C_T\Norm{\zeta}_{\rho} < \infty, \quad \text{for all } t \in [0,T]. 
	\]
	\end{theorem}
\end{frame} 

\begin{frame}{Invariant measures for the stochastic heat equation}
	\rdot So the solution acts as an operator, mapping $H$ to $H$. \vspace{.2in} \\
	
	\rdot Moreover, the Markov property of the solution \cite[Th 9.21]{DZ14} implies that 
		\[
			P_t(\zeta,\Gamma) := \mathscr{L}(u(t,\cdot;\zeta))(\Gamma), \quad \text{for all } \zeta \in H \text{ and } \Gamma \in \mathscr{B}(H)
		\]
		for a family of Markovian transition functions. \vspace{.3in}
	
	\begin{definition}
		A measure $\eta$ is invariant for the stochastic heat equation if $\eta$ is invariant for the transition functions $P_t$. In other words, 
			\[
				P_t^*(\eta)(\Gamma) = \int_{H} P_t(\zeta, \Gamma) \eta(\ud \zeta) = \eta(\Gamma), \quad \text{for all } \Gamma \in \mathscr{B}(H).
			\]
	\end{definition}
\end{frame}

\begin{frame}{Convergence of measures}
	\rdot Let $\mathscr{M}(H)$ and $C_b(H)$ denote the probability measures and bounded continuous functions on $H$. \vspace{.3in} \\
	
	\rdot For $\varphi \in C_b(H)$ and $\nu \in \mathscr{M}(H)$, 
	\[
		\langle \varphi, \nu \rangle = \int_{H} \varphi(\zeta) \nu(\ud \zeta).
	\] \vspace{.1in}

	\begin{definition}
		We say $\nu_n$ converges {\em weakly} to $\eta$ in $\mathscr{M}(H)$ if 
			\[
				\langle \varphi, \nu_n \rangle \to 	\langle \varphi, \nu \rangle, \quad \text{as } n \to \infty.
			\] 
		We will use the notation, $\nu_n \to \nu$, to denote the weak convergence of measures. 
	\end{definition}
\end{frame}

\begin{frame}{Some interesting results}
	\begin{theorem}[DaPrato {\cite[Propositions 11.2 and 11.3]{DZ14}}]
		If $\zeta_1, \zeta_2 \in H$ and $\eta_1, \eta_2 \in \mathscr{M}(H)$ are such that 
		{\small
			\[
				\mathscr{L}\bigg( u(t,\cdot;\zeta_1) \bigg) \to \eta_1 \text{ as } t \to \infty
				\quad \text{or} \quad 
				\frac{1}{t_n} \int_0^{t_n} \mathscr{L}\bigg( u(s,\cdot;\zeta_2) \bigg) \ud s \to \eta_2 \text{ as } n \to \infty,
			\] 
		}
		for some $t_n \uparrow \infty$, then both $\eta_1$ and $\eta_2$ are invariant. 
	\end{theorem}

	\begin{theorem}[DaPrato { \cite[Proposition 11.5]{DZ14} }]
		Suppose $\eta$ is invariant for the stochastic heat equation and $\zeta \in H$ is random and $\mathscr{L}(\zeta) = \eta$. Then $\mathscr{L}\bigg(u(t,\cdot;\zeta)\bigg)$ is stationary:
			\[
				\mathbb{E}\bigg[\varphi_1\big(u(t_1+h,\cdot;\zeta)\big) \cdots \varphi_n\big(u(t_n+h,\cdot;\zeta)\big)\bigg]
			\]
		is independent of $h$ for any $n \in \mathbb{N}$ and $\varphi_i \in C_b(H)$.
	\end{theorem}

\end{frame}

\begin{frame}{Permissible initial conditions}
	\begin{enumerate}
	\item For any $\zeta \in L^{\infty}(\R^d)$ then for all $t \ge 0$
		\[
			\mathcal{G}_{\rho}(t;\mu) \le \left(1+\Norm{\zeta}_{L^{\infty}(\R^d)}\right)^2 \Norm{\rho}_{L^1{\R^d}} < \infty.
		\]
		
	\item For $\delta_0(\cdot)$ and for all $ t > 0$,
		\[
			\mathcal{G}_{\rho}(t;\delta_0) \le \left(2 + G(t,0)^2\right) \Norm{\rho}_{L^1(\R^d)} < \infty.
		\]
	
	\item Consider $|\cdot|^{-\lambda}$ for any $\lambda \in (0,d)$. Then for all $ t> 0$,
		\[
			\mathcal{G}_{\rho}(t;|\cdot|^{-\lambda}) \le 2(1 + Ct^{-\lambda})\Norm{\rho}_{L^1(\R^d)} < \infty, \quad C = 2^{-\lambda} \left(\frac{\Gamma((d-\lambda)/2)}{\Gamma(d/2)}\right)^2,
		\] 
	\end{enumerate}
\rdot $\delta_0(\cdot) \not \in L^2_{\rho}(\R^d)$ but still permissible. \vspace{.2in} \\
\rdot $|\cdot|^{-\lambda} \not \in L^2_{\rho}(\R^d)$ for $\lambda \in (d/2, d)$, however, it is still applicable.
\end{frame}
