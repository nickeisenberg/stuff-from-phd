	\begin{frame}{The stochastic heat equation}
		\[
		\begin{cases}
		\dfrac{\partial u}{\partial t}(t,x)- \frac{1}{2}\Delta_xu(t,x) = b(x, u(t,x))\dot{W}(t,x), & t \ge 0, \: x \in \R^d
		\\ u(0,\cdot) = \mu(\cdot)  &
		\end{cases} \vspace{.1in}
		\]

\rdot The Stochastic Heat Equation is interpreted as follows: \vspace{0in}
\begin{align*}
u(t,x) &= \int_{\R^d} G(t,x-y) \mu(\ud y) + \int_0^t \int_{\R^d}G(t-s,x-y)b(u(s,y))W(\ud s, \ud y) 
\end{align*}
where $G(t,x)$ is the heat kernel \vspace{0in}
\[
G(t,x) = (2 \pi t)^{-d/2} \exp\left(- \frac{|x|^2}{2t} \right) \vspace{.1in}
\]
\rdot A processes, $\{u(t,x) : t \ge 0, \; x\in \R^d \}$, that satisfies this relation is referred to as the {\it mild solution}.


\end{frame}

\begin{frame}[t]{The noise term $\dot{W}$}
\rdot $W = \{W(\phi): \: \phi \in \mathscr{S}(\R^d) \}$ is a centered Gaussian process uniquely defined through the following covariance functional: 
\[
	J(\phi, \psi) := \mathbb{E}\left[W(\phi)W(\psi)\right] = \int_0^\infty \ud s \int_{\R^d} \Gamma(\ud x) (\phi(s,\cdot)*\tilde{\psi}(s,\cdot))(x),
\]
where $\tilde{\psi}(s,x) = \psi(s,-x)$. We have the following nomenclature. \vspace{.25in} 
	
 \rdot \textcolor{blue}{{\em Space-time white noise}}: $\Gamma(\ud x) = \delta_0(\ud x)$. \vspace{.15 in}
 
 \rdot \textcolor{blue}{{\em Time white and spatially colored}}: $\Gamma(\ud x) = f(x)\ud x$ where $f$ is a non-negative and non-negative definite function referred to as the {\em Correlation function} and its Fourier transform,
 	\[
 		\widehat{f}(\xi) := \mathcal{F}f(\xi) = \int_{\R^d} \exp\left(-i x \cdot \xi\right) f(x) \ud x,
 	\]
 is referred to as the {\em spectral density}.
\end{frame}

\begin{frame}{Existence and uniqueness with space-time white noise}
	\rdot A necessary condition for global existence, {\em Dalang's condition} \cite{Dal99}, is the following:
	\[
		\int_0^t \ud s \int_{\R^d} \ud y \: G^2(s,y) < \infty, \quad \forall t > 0.
	\] 
	\rdot Direct calculations yields that,
	\[
		\int_0^t \ud s \int_{\R^d} \ud y \: G^2(s,y) = 
		\frac{\Gamma(d/2)}{2(2\pi)^d} \int_0^t \frac{1}{s^{d/2}} \ud s,
	\]
	which forces,
	\[
		d/2 < 1 \quad \iff \quad d < 2.
	\]
	\rdot Thus a global space-time white noise solution may only exist when the spatial dimension is $1$. 
\end{frame}

\begin{frame}{Will the second moments ever be bounded in time?}
	\rdot Several slides below will require the second moments to be bounded in time:
		\[
			\sup_{t\ge0} \Norm{u(t,x)}_2 < \infty, \qquad \text{where } \Norm{X}_p = \left[\mathbb{E}\left(|X|^p\right)\right]^{1/p}.
		\]
	
	\rdot This is possible, however this will require us to only consider $d \ge 3$, and therefore we must only consider a spatially colored noise. \vspace{.2in}
	
	\rdot Recall from above, that the covariance of a spatially colored noise is given by the following:
	\begin{align*}
 \mathbb{E}\left[W(\phi)W(\psi)\right] &= \int_0^\infty \ud s \int_{\R^d} \ud x \: f(x) (\phi(s,\cdot)*\tilde{\psi}(s,\cdot))(x) \\
 &= \int_0^\infty \ud s \int_{\R^{2d}} \ud x \ud y \: \phi(s,x)f(x-y)\psi(s,y) \\
 &= (2\pi)^{-d} \int_0^{\infty} \ud s \int_{\R^d} \ud \xi \: \widehat{f}(\xi) \mathcal{F}\phi(s,\xi) \overline{\mathcal{F}\psi(s,\xi)}
 \end{align*}
\end{frame}
	
\begin{frame}{$u_t - \frac{1}{2}u_{xx} =  b(u) \dot{W}$ with $u(0,\cdot) = \mu(\cdot)$ \hfill (colored noise)}
\begin{theorem}[Chen and Huang {\cite[Theorem 1.2]{CH19}}]
There exists a unique global mild solution under the following:
\end{theorem} \vspace{.1in}

\rdot $b: \R^d \times \R \to \R$ such that there exists a $L_b,L_0 >0$ so for all $u,v \in \R$ and $x \in \R^d$,
\[
|b(x,u)-b(x,v)| \le L_b|u-v| \quad \text{and} \quad  |b(x,0)| \le L_0.
\] \vspace{0in}

\rdot The initial measure, $\mu(\ud x)$ (resp. $\varphi(x) \ud x$), satisfies for some $a > 0$,
\[
\int_{\R^d} e^{-a|x|^2} |\mu|(\ud x) < \infty \quad \left(\text{resp. } \int_{\R^d} e^{-a|x|^2} |\varphi(x)|\ud x < \infty \right),
\] 
where $|\mu| = \mu^+ + \mu^-$ is the {\em Hahn decomposition}. \vspace{.2in}

\rdot The spectral density, $\widehat{f}$, satisfies {\it Dalang's condition for some $\beta >0$:}
\[
\Upsilon(\beta) : = \int_{\R^d} \frac{\hat{f}(\ud \xi)}{\beta + |\xi|^2} < \infty.
\]

\end{frame}

\begin{frame}{Bounded moments in time}
	\begin{theorem}[Chen and Kim {\cite[Theorem 1.3 and Lemma 2.5]{CK19}}]
		Suppose that we strengthen Dalang's condition and the Lipschitz assumption to the following:
		\[
		 \Upsilon(0) := \left(2\pi\right)^{-d} \int_{\R^d} \frac{\hat{f}(\ud \xi)}{|\xi|^2} <\infty \quad \text{and} \quad L^2_{b}   <  \frac{1}{128 \Upsilon(0)}.
		\] 
		Then the following holds:
		\begin{align*}
			\Norm{u(t,x)}_2 &\le C_2\bigg[1 + \Big(G(t,\cdot) * |\mu|\Big)(x)\bigg]\\
			 &=C_2\bigg[1 + \int_{\R^d} G(t,x-y) \: |\mu|(\ud y)\bigg], \quad \text{for all } t \ge 0.
		\end{align*}
	\end{theorem} 
\rdot If $\mu = \delta_0$ then $\Norm{u(t,x)}_2 \le C_2\left[1 + (2 \pi t)^{-d/2}\right]$. \vspace{.1in} \\

\rdot Later, it will suffice consider $\mu$ such that $\sup_{t\ge t_0} \Norm{u(t,x)}_2 < \infty$.
\end{frame}

\begin{frame}{Transition functions and their semigroup}
\begin{definition}[Markovian Transition Functions]
		\rdot Let $H$ be a Hilbert space and $\mathscr{B}(H)$ be the borel sets of $H$. \\
		\rdot $\{ P_t(x,\Gamma): t \ge 0, x \in H, \Gamma \in \mathscr{B}(H) \}$ forms a family of {\em Markovian transition functions} if, \vspace{.1in}
	
	\rnum{1} $x \mapsto P_t(x, \Gamma)$ if measurable, \vspace{.1in} 
	
	\rnum{2} $\Gamma \mapsto P_t(x, \Gamma)$ is a probability measure on $H$, \vspace{.1in} 
	
	\rnum{3} $P_{t+s}(x,\Gamma) = \int_{H} P_t(x,\ud y) P_s(y, \Gamma)$, \vspace{.1in} 
	
	\rnum{4} $P_0(x,\Gamma) = \textbf{1}_{\Gamma}(x)$.

\end{definition}

\begin{definition}[Transition Semigroup]
	The above transition functions form the following semi group on $\mathcal{B}(H)$, which denotes the bounded Borel measurable functions on $H$:
	\[
		P_t\varphi(x) = \int_H \varphi(y) P_t(x, \ud y), \qquad \varphi \in \mathcal{B}(H).
	\]
\end{definition}

\end{frame}

\begin{frame}{Invariant measures}
	\begin{definition}[Invariant measures]
		\rdot A probability measure, $\eta$, on $H$ is said to be {\em invariant} for the transition functions $P_t$ if for all $\Gamma \in \mathscr{B}(H)$ and $t > 0$,
		\[
		\textcolor{blue}{\int_H P_t(x,\Gamma) \eta(\ud x)} = \int_H P_0(x,\Gamma) \eta(\ud x) = \int_H \textbf{1}_{\Gamma}(x) \eta(\ud x) = \textcolor{blue}{\eta(\Gamma)},
		\]
		which is equivalent to,
		\[
			\int_H P_t\varphi(x) \eta(\ud x) = \int_H \varphi(x) \eta(\ud x), \qquad \text{for all } \varphi \in \mathcal{B}(H). 
		\]
	\end{definition}

\rdot This implies that the $\eta$-average of $x \mapsto P_t(x,\Gamma)$ is $\eta(\Gamma)$ and it is independent of time:
\[
	\text{AVG}_{\eta}\big[P_t(\cdot,\Gamma)\big] = \frac{1}{\eta(H)}\int_H P_t(x,\Gamma) \eta(\ud x) = \int_H P_t(x,\Gamma) \eta(\ud x) = \eta(\Gamma).
\]
\end{frame}

\begin{frame}{Invariant measures for the stochastic heat equation}
	\rdot First we define the law of the solution as follows:
	\[
		\mathscr{L}(u(t,\cdot;\zeta))(\Gamma) = \mathbb{P}\left[\omega \in \Omega : u(t,\cdot;\zeta) \in \Gamma\right], \quad \text{for all } \zeta \in H \text{ and } \Gamma \in \mathscr{B}(H),
	\]
	and for the mean time we are intentionally vague about $H$. \vspace{.2in} \\
	
	\rdot Our intention is for 
		\[
			P_t(\zeta,\Gamma) := \mathscr{L}(u(t,\cdot;\zeta))(\Gamma)
		\]
		to form a Markovian semigroup. \vspace{.2in} \\

	\rdot Established theory by DaPrato and Zabczyk \cite{DZ14} requires the solution, 
	\[
	\zeta \mapsto u(t,\cdot;\zeta),
	\]
	 to be almost surely $H$-valued for $t > 0$ and $\zeta \in H$. \vspace{.2in} \\

	\rdot So what is $H$? 
	
\end{frame}

\begin{frame}{$H = L^2_{\rho}(\R^d)$: weighted square integrable functions}
	\begin{definition}[Admissible Weight]
		A positive, bounded and continuous $L^1(\R^d)$ function, $\rho$, is an {\em admissible weight} if 
			\[
				\frac{\bigg( G(t,\cdot)*\rho(\cdot) \bigg)(x)}{\rho(x)} \le C_{\rho}(T), \qquad \text{for all } t \in [0,T] \text{ and } x \in \R^d.
			\] 
	\end{definition}

	\begin{definition}[$L^2_{\rho}(\R^d)$]
		The space $H := L^2_{\rho}(\R^d)$ is the space of $\rho$-weighted square integrable functions:
		\[
			\Norm{\zeta}^2_{\rho} := \int_{\R^d} |\zeta(x)|^2 \rho(x) \ud x < \infty.
		\]
		Moreover, $H$ is a Hilbert space with inner product defined as 
		\[
			\langle \zeta, \psi \rangle_{\rho} := \int_{\R^d} \zeta(x) \psi(x) \rho(x) \ud x.
		\]
	\end{definition}
\end{frame}

\begin{frame}{The solution is almost surely $H$-valued.}
	\begin{theorem}

Let $\rho$ be an admissible weight and suppose that
\[
	\Upsilon(0) = \int_{\R^d} \frac{\widehat{f}(\xi)}{|\xi|^2} \ud \xi < \infty \qquad \text{and} \qquad L_b^2 < \frac{1}{128 \Upsilon(0)}.
\]
Now suppose our initial measure, $\mu$, satisfies the following:
\[
	\mathcal{G}_{\rho}(t;\textcolor{blue}{|\mu|}) := \int_{\R^d} \left( \int_{\R^d} G(t,x-y) \textcolor{blue}{|\mu|(\ud y)} \right)^2 \rho(x) \ud x < \infty.
\]
Then for all $t > 0$,
\[
	\mathbb{E}\left( \Norm{u(t,\cdot;\mu)}^2_{\rho} \right) \le C\mathcal{G}_{\rho}(t; 1 + |\mu|) < \infty.
\]
In particular, for any $\zeta \in H$ and for all $T>0$, there exists a $C_T$ such that
	\[
			\mathcal{G}_{\rho}(t;\zeta) \le 2\Norm{\rho}_{L^1(\R^d)} + 2C_T\Norm{\zeta}_{\rho} < \infty, \quad \text{for all } t \in [0,T]. 
	\]
	\end{theorem}
\end{frame} 

\begin{frame}{Invariant measures for the stochastic heat equation}
	\rdot So the solution acts as an operator, mapping $H$ to $H$. \vspace{.2in} \\
	
	\rdot Moreover, the Markov property of the solution \cite[Th 9.21]{DZ14} implies that 
		\[
			P_t(\zeta,\Gamma) := \mathscr{L}(u(t,\cdot;\zeta))(\Gamma), \quad \text{for all } \zeta \in H \text{ and } \Gamma \in \mathscr{B}(H)
		\]
		for a family of Markovian transition functions. \vspace{.3in}
	
	\begin{definition}
		A measure $\eta$ is {\em invariant} for the stochastic heat equation if $\eta$ is invariant for the transition functions $P_t$. In other words, 
			\[
				\int_{H} P_t(\zeta, \Gamma) \eta(\ud \zeta) = 	\int_{H} \mathscr{L}(u(t,\cdot;\zeta))(\Gamma) \eta(\ud \zeta) = \eta(\Gamma), \quad \text{for all } \Gamma \in \mathscr{B}(H).
			\]
	\end{definition}
\end{frame}

\begin{frame}{Weak convergence of measures}
	\rdot Let $\mathscr{M}(H)$ and $C_b(H)$ denote the probability measures and bounded continuous functions on $H$. \vspace{.3in} \\
	
	\rdot For $\varphi \in C_b(H)$ and $\nu \in \mathscr{M}(H)$, 
	\[
		\langle \varphi, \nu \rangle = \int_{H} \varphi(\zeta) \nu(\ud \zeta).
	\] \vspace{.1in}

	\begin{definition}
		We say $\nu_n$ converges {\em weakly} to $\eta$ in $\mathscr{M}(H)$ if for all $\varphi \in C_b(H)$,
			\[
				\langle \varphi, \nu_n \rangle \to 	\langle \varphi, \nu \rangle, \quad \text{as } n \to \infty.
			\] 
		We will use the notation, $\nu_n \to \nu$, to denote the weak convergence of measures. 
	\end{definition}
\end{frame}

\begin{frame}{Some existence results}
	\begin{theorem}[DaPrato {\cite[Propositions 11.2 and 11.3]{DZ14}}]
		If $\zeta_1, \zeta_2 \in H$ and $\eta_1, \eta_2 \in \mathscr{M}(H)$ are such that 
		{\small
			\[
				\textcolor{red}{\mathscr{L}\bigg( u(t,\cdot;\zeta_1) \bigg) \to \eta_1 \text{ as } t \to \infty}
				\quad \text{or} \quad 
				\textcolor{blue}{\frac{1}{t_n} \int_0^{t_n} \mathscr{L}\bigg( u(s,\cdot;\zeta_2) \bigg) \ud s \to \eta_2 \text{ as } n \to \infty},
			\] 
		}
		for some $t_n \uparrow \infty$, then both $\eta_1$ and $\eta_2$ are invariant. 
	\end{theorem} \vspace{.1in}

\rdot \textcolor{red}{Red} implies \textcolor{blue}{Blue} but not vice versa. \vspace{.1in} \\

\rdot  For example, suppose $f(t) \to L$ as $t \to \infty$. Then for $\epsilon > 0$, there exists $N$ so for all $n>N$ we have the following:
\[
	\lim_{n \to \infty} \frac{1}{n} \int_0^n f(t) \ud t \: \nobarfrac{\le}{\ge} \: \lim_{n \to \infty} \frac{1}{n}\left( \int_0^N f(t) \ud t + \int_N^n L \pm \epsilon \: \ud t \right),
\]
thus,
\[
	L- \epsilon \le 	\lim_{n \to \infty} \frac{1}{n} \int_0^n f(t) \ud t \le L+\epsilon.
\]

\begin{frame}{Under what setup has an invariant measure been proven to exist?}
	\rdot Research on invariant measures for $u_t - \frac{1}{2}\Delta u = \textcolor{red}{f(u)} + b(u)\dot{W}$ on the whole space with a globally Lipschitz $b$ is not vast. \vspace{.2in}
	
	\rdot Tessitore and Zabczyk \cite{TZ96} is the only other paper which studies $u_t - \frac{1}{2}\Delta u =  b(u)\dot{W}$ on the whole space with  with a globally Lipschitz $b$. \vspace{.2in}
	
	\rdot Tessitore and Zabczyk showed there exists an $\eta \in \mathscr{M}(H)$ such that,	
		\[
			\frac{1}{t_n} \int_1^{t_n} \mathscr{L}\bigg(u(s,\cdot;1)\bigg) \ud s \to \eta, \quad \text{as } n \to \infty,
		\]
	under the following assumption on the spectral density $\widehat{f}$:
	{ \small
		\[
		\int_{\R^d} \: |\xi|^{2-d} \left[\left| \mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right)\: \right| * \left|\mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right) \: \right| \right](\xi)  \: \ud \xi < \infty.
		\]
	}
		
\end{frame}

\begin{frame}
\rdot	If one could find a $f$ such that $\mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right) \ge 0$ and $\widehat{f} \ge 0$ then 
		\begin{align*}
				\int_{\R^d} & \: |\xi|^{2-d} \left[\left| \mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right)\: \right| * \left|\mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right) \: \right| \right](\xi)  \: \ud \xi \\
				& = 	\int_{\R^d} \: |\xi|^{2-d} \left[\mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right)  * \mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right) \right](\xi)  \: \ud \xi \\
				&= K \int_{\R^d} \: |\xi|^{2-d} \left[\mathcal{F}^{-1}\left(\widehat{f} \: \right) \right](\xi)  \: \ud \xi \\
				&= K \int_{\R^d} \: |\xi|^{2-d} f(\xi)  \: \ud \xi \\
				& = K \int_{\R^d} \: |\xi|^{-2} \widehat{f}(\xi)  \: \ud \xi = \Upsilon(0),
		\end{align*}
	which is precisely Dalang's condition. \vspace{.2in}
	
	\rdot $\mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right) \ge 0$ is restrictive.
\end{frame}

\begin{frame} $\;$ \\
	\rdot $\mathcal{F}^{-1}\left(\sqrt{\widehat{f}} \: \right) \ge 0$ requires $\sqrt{\widehat{f}}$ to be non-negative definite and this may not be true even is $\widehat{f}$ is non-negative and non-negative definite.
	
	\begin{example}
		Consider the pair 
		\[
		f(x) = (2\pi)^{-1}\sin(x^2)/x^2 
		\quad \text{and} \quad 
		\widehat{f}(\xi) = \frac{1}{4} \max \{ 2-|\xi|, 0 \}.
		\] 
		Both $f$ and $\widehat{f}$ is clearly non-negative and thus both are non-negative definite. However, direct calculation shows that 
		\[
		\mathcal{F}^{-1} \left( \sqrt{\widehat{f}} \: \right)(x) = 
		\frac{ \sqrt{\frac{\pi}{2}}\left( -\cos(2x)\mathcal{S}\left(\frac{2 \sqrt{x}}{\sqrt{\pi}}\right) + \sin(2x) \mathcal{C}\left(\frac{2\sqrt{x}}{\sqrt{\pi}}\right) \right) }{ x^{3/2} }
		\]
	\end{example}
Where $\mathcal{S}$ and $\mathcal{C}$ are the sine and cosine Fresnel integrals,
\[
	\mathcal{C}(z) = \int_0^z \cos\left(\frac{\pi t^2}{2}\right) \ud t \quad \text{and} \quad \mathcal{S}(z) = \int_0^z \sin\left(\frac{\pi t^2}{2}\right) \ud t.
\]
\end{frame}

\begin{frame}[t]
\small
	\rdot Recall that 
	\[
	\mathbb{E}\left(\Norm{u(t,\cdot;\mu)}_\rho\right) \le	C\mathcal{G}_{\rho}(t;\mu) := C\int_{\R^d} \left( \int_{\R^d} G(t,x-y) (1+|\mu|)(\ud y) \right)^2 \rho(x) \ud x 
	\]

	\begin{theorem}
Suppose that the initial measure, $\mu$, the diffusion term, $b$, and the spectral density, $\widehat{f}$ satisfy 
	\[
		\Upsilon_\alpha (0) := \int_{\R^d} \frac{\widehat{f}(\xi) \ud \xi}{|\xi|^{2(1-\alpha)}} < \infty , \quad \limsup_{t > 0} \mathcal{G}_{\rho}(t; \mu) < \infty, \quad \text{and} \quad L_b^2 < \frac{1}{2^7 \Upsilon(0)},
	\]
for some $\alpha \in \left(2^7\Upsilon(0)L_b^2,1\right)$ and also for $\alpha = 0$. Then there exists an invariant measure for the stochastic heat equation. More precisely, the following weak limit exists for each $t_0 > 0$:
	\[
		\frac{1}{t_0} \int_{t_0}^{t_n + t_0} \mathscr{L}\bigg(u(s,\cdot;\mu)\bigg) \ud s \to \eta_{t_0},
	\]
where $\eta_{t_0}$ is the invariant measure. 
	\end{theorem}
	
\end{frame}

\begin{frame}{$\widehat{f}(\xi)$ and $\Upsilon_{\alpha}(0)$}
	\rdot It is easy to verify that $\widehat{f}(\xi) = \max\{ 2-|\xi|, 0 \}$ is allowable using our condition above. Indeed, for any $\alpha \in [0,1)$,
	\begin{align*}
		\Upsilon_{\alpha}(0) &= \int_{\R^d} \frac{\max\{ 2-|\xi|, 0 \}}{|\xi|^{2(1-\alpha)}} \ud \xi \\
		&= \int_0^{\infty} \frac{\max\{ 2-r, 0 \}}{r^{2(1-\alpha)}} r^{d-1}\ud r \\
		&= \int_0^{2}  \frac{2-r}{r^{2(1-\alpha)}} r^{d-1}\ud r \\
		&= \int_0^2 2 r^{d-1-2(1-\alpha)} - r^{d-2(1-\alpha)} \ud r \\
		&< \infty \text{ when }
			\begin{cases}
				\frac{d}{2} > 1 - \alpha & 0 < \alpha < 1 \\
				\frac{d}{2} > 1 & \alpha = 0.
			\end{cases}
	\end{align*}
\rdot Thus, $\frac{d}{2} > 1$, being the more restrictive condition, is what we need to assume. In other words, the spatial dimension must be $d >2$ which means $d \ge 3$.  
	
\end{frame} 


\begin{frame}{A comment about $\Upsilon_{\alpha}(0) < \infty$}
	\small
	\rdot  For $\alpha \in (0,1/2)$, $\Upsilon_{2\alpha}(0) < \infty$ implies H\"older continuity of the solution \cite{CH19}. \vspace{.1in}
	
	\rdot This condition may be slightly "weakened" to the following: 
	\[
	\mathcal{H}_{\alpha}(t) := \int_0^t \ud r \: r^{-2\alpha} \int_{\R^d} \widehat{f}(\ud \xi) \exp(-r|\xi|^2)< \infty \quad \text{for all } t > 0 \text{ and some } \alpha \in \left(0,\frac{1}{2}\right).
	\]
	\begin{lemma} We have the following relations between $\Upsilon_{\alpha}(0), \Upsilon(0)$ and $\mathcal{H}_{\alpha}(t)$: \vspace{.1in}
		
		\rnum{1} However, $\Upsilon_{2\alpha}(0) < \infty \implies \mathcal{H}_{\alpha}(t) < \infty$ for all $t > 0$. \vspace{.1in}
		
		\rnum{2} $\lim \limits_{t\to \infty} \mathcal{H}_{\alpha}(t) = \Gamma(1- 2\alpha) \Upsilon_{2\alpha}(0)$. \vspace{.1in}
		
		\rnum{3} When $\Upsilon(0) < \infty$ then $\Upsilon_{2\alpha}(0) < \infty \iff \mathcal{H}_{\alpha}(t) < \infty$ for all $t > 0$. \vspace{.1in}
	\end{lemma}	
	
\end{frame}

\begin{frame}{Strategy}
	\rdot The strategy is to prove tightness of the laws: when given $\epsilon \in (0,1)$, to prove the existence of a compact set $\mathcal{K}$ such that 
	\begin{align*}
	\mathscr{L}u(t,\cdot;\mu)(\mathscr{K}) \ge 1- \epsilon \quad \text{for all } t \ge t_0 >0.
	\end{align*}
	
	\rdot This implies that 
	\begin{align*}
	A_{n}(\textcolor{blue}{\mathscr{K}}):=\frac{1}{n} \int_{t_0}^{n + t_0} \mathscr{L}\bigg(u(s,\cdot;\mu)\bigg)(\textcolor{blue}{\mathscr{K}}) \ud s \ge \frac{1}{n} \int_{t_0}^{n + t_0} 1 - \epsilon \: \ud s  =1 - \epsilon,
	\end{align*} 
	and so the sequence of measures, $A_{n}$ is tight. \vspace{.15in} \\
	
	\rdot Every tight sequence of measures has a weakly convergent subsequence that converges to an invariant measure, $\eta$. Hence, there exists $(t_n) \uparrow \infty$ such that,
		\[
			A_{t_n} \to \eta 
		\] \vspace{0in} 
		
	\rdot Unfortunately this does not imply that 
		\[
			\mathscr{L}u(t,\cdot ; \mu) \to \eta \quad \text{as } t \to \infty.
		\]
\end{frame}

\begin{frame}{Permissible spectral density}
\rdot Recall that $\widehat{f}$ must satisfy for both $\alpha = 0$ and $\alpha \in \left(2^7\Upsilon(0)L_b^2,1\right)$ the following:
	\[
			\Upsilon_\alpha (0) := \int_{\R^d} \frac{\widehat{f}(\xi) \: \ud \xi}{|\xi|^{2(1-\alpha)}} < \infty.
	\]

\rdot Recall that $d\ge3$ and so $\widehat{f}$ may have slow blow up at the origin but must have fast decay at $\pm \infty$ to counter act the thick tails of $|\xi|^{2(1-\alpha)}$. \vspace{.25in}

\rdot This excludes the Riesz kernel. \vspace{.25in}

\rdot However, the Bessel kernel has polynomial blowup at the origin and exponential decay at $\pm \infty$ and will be permissible.
\end{frame} 

\begin{frame}{Bessel kernel}
	\small
	\rdot Let $f_s$ denote the Bessel kernel. When $0 < s < d$ we have that
	\[
		f_s(x) \sim \exp(-|x|/2) \text{ for } |x| > 2 \quad \text \quad f_s(x) \sim \frac{1}{|x|^{d-s+2}} \text{ for } |x| < 2.
	\]
	
	\rdot The Fourier transform $\widehat{f}$ is given by $\widehat{f}(\xi) = (1+|\xi|^2)^{-s/2}$.
	
	\begin{example}
		When considering the Bessel kernel, direct calculation yields that 
			\[
				\Upsilon_{\alpha}(0) = \frac{\Gamma\left( \frac{d}{2} - 1 + \alpha \right) \Gamma\left( \frac{s-d}{2} + 1 - \alpha \right)}{2^d \pi^{d/2} \Gamma(s/2)},
			\]
		and when $\alpha = 0$, this simplifies to
			\[
				\Upsilon(0) = \frac{\Gamma\left( \frac{2 + s - d}{2} \right)}{2^{d-1} \pi^{d/2} (d-2)\Gamma(s/2)},
			\]
		and we must assume that 
			\[
				0 < \alpha < 1 - \frac{(d-s)_+}{2} \quad \text{and} \quad s > d - 2 > 0.
			\]
	\end{example}
\end{frame}

\begin{frame}{Quick verification for the Bessel kernel}
	\rdot Recall that \textcolor{blue}{$\Upsilon_{\alpha}(0) = C \int_{\R^d} \frac{\widehat{f}(\xi)}{|\xi|^{2(1-\alpha)}} \ud \xi$} and \textcolor{blue}{$\widehat{f}(\xi) = (1+ |\xi|^2)^{-s/2}$}.
	\begin{align*}
	\Upsilon_{\alpha}(0) &= C\int_{\R^d} \frac{1}{|\xi|^{2(1-\alpha)}(1+|\xi|^2)^{s/2}} \ud \xi \\ 
	&= C\int_{0}^{\infty} \frac{z^{d-1}}{z^{2(1-\alpha)}(1+z^2)^{s/2}} \ud z ,\\ 
	\end{align*}
	\rdot Now by the change of variables
	$z=r^2/(1+r^2)$, we can evaluate the above integral explicitly by transforming it to the Beta
	  \begin{align*}
	C\int_0^\infty \frac{r^{d-1}}{r^{2(1-\alpha)}\left(1+r^2\right)^{s/2}} \ud r
	& = C \int_0^1 z^{d/2+\alpha-2}(1-z)^{(s-d)/2-\alpha}\ud z \\
	& = C\frac{\Gamma\left(d/2-1+\alpha\right)\Gamma\left((s-d)/2+1-\alpha\right)}{2\Gamma(s/2)},
	\end{align*}
\end{frame}

\begin{frame}{Riesz type kernel}
	\rdot The $g_s$ denote the Riesz kernel for $0 < s < d$: $g_s(x) = |x|^{-s}$. \vspace{.2in}
	
	\rdot $g_s$ will not satisfy 
	\[
	\Upsilon(0) = C \int_{\R^d} \widehat{g}_s(\xi)|x|^{-2} \ud \xi = \int_{\R^d} |\xi|^{d-s} |\xi|^{-2} \ud \xi < \infty.
	\] \vspace{.2in}
	
	\rdot Instead, we can use the following 
		\[
			\widehat{r}_s(\xi) := \widehat{r}_{s_1, s_2}(\xi) = \widehat{f}_{s_1}(\xi) + f_{s_2}(\xi),
		\]
	where $s_1>0$, $0 < s_2 < d$ and $f_{s_1}$ and $f_{s_2}$ are Bessel kernels. \vspace{.2in}
	
	\rdot $\widehat{r}_s$ has polynomial growth at the origin and polynomial decay at infinity: 	
	\begin{align*} \widehat{r}_{s}(\xi) \sim
	\begin{cases}
	|\xi|^{-(d - s_2)} & |\xi| \to 0 \\
	|\xi|^{-s_1} & |\xi| \to \infty,
	\end{cases}
	\end{align*}
\end{frame}

\begin{frame}{Riese type kernel}
	\rdot Notice that 
	\begin{align*}
		\Upsilon_{\alpha}(0) &= \int_{\R^d} \frac{\widehat{r}_s(\xi)}{|\xi|^{2(1-\alpha)}} \ud \xi \\
		&= \int_{\R^d} \frac{\widehat{f}_{s_1}(\xi)}{|\xi|^{2(1-\alpha)}} \ud \xi 
		+ \int_{\R^d} \frac{f_{s_2}(\xi)}{|\xi|^{2(1-\alpha)}} \ud \xi \\
		&= \int_{\R^d} \frac{\widehat{f}_{s_1}(\xi)}{|\xi|^{2(1-\alpha)}} \ud \xi 
		+ (2\pi)^{-d}\int_{\R^d} \frac{\widehat{f}_{s_2}(\xi)}{|\xi|^{d - 2(1-\alpha)}} \ud \xi
	\end{align*}
	
	\rdot We can easily calculate that 
		\begin{align*}
	\Upsilon_{\alpha}(0) = 
	\frac{\Gamma\left(\frac{d}{2} -1 + \alpha\right) \Gamma\left( \frac{s_1 -d}{2} + 1 - \alpha \right)}{2^d \pi^{d/2} \: \Gamma(d/2)\:\Gamma\left(s_1/2\right)} 
	+ \frac{\Gamma\left(1 - \alpha\right) \Gamma\left(\alpha - 1 + \frac{s_2}{2}\right)}{2^{2d} \pi^{3d/2} \: \Gamma(d/2) \: \Gamma\left(s_2 / 2\right) } < \infty,
	\end{align*}
	where the finiteness requires us to assume that 
	\begin{align*} \label{E:RieszTypeCond}
	s_1 > d - 2(1-\alpha) > 0 \qquad \text{and} \qquad d > s_2 > 2(1-\alpha) > 0.
	\end{align*} 
\end{frame}

\begin{frame} 
	\small
	\textbf{Allowable Initial Conditions:}
	
		\rnum{1} For any $\zeta \in L^{\infty}(\R^d)$ then for all $t \ge 0$
		\[
		\mathbb{E}\left(\Norm{u(t,\cdot;\mu)}_{\rho}^2\right) \le \left(1+\Norm{\zeta}_{L^{\infty}(\R^d)}\right)^2 \Norm{\rho}_{L^1{\R^d}} < \infty.
		\]
		
		\rnum{2} For $\delta_0(\cdot)$ and for all $ t > 0$,
		\[
		\mathbb{E}\left(\Norm{u(t,\cdot; \delta_0)}_{\rho}^2\right) \le \left(2 + G(t,0)^2\right) \Norm{\rho}_{L^1(\R^d)} < \infty.
		\]
		
		\rnum{3} Consider $|\cdot|^{-\lambda}$ for any $\lambda \in (0,d)$. Then for all $ t> 0$,
		\[
		\mathbb{E}\left(\Norm{u(t,\cdot;|\cdot|^{-\lambda})}_{\rho}^2\right)  \le 2(1 + Ct^{-\lambda})\Norm{\rho}_{L^1(\R^d)} < \infty, \quad C = 2^{-\lambda} \left(\frac{\Gamma((d-\lambda)/2)}{\Gamma(d/2)}\right)^2,
		\] \vspace{0in}
		
	\textbf{Non-permissible $L^2_{\rho}(\R^d)$-valued initial conditions:}	\vspace{.05in}
	
	\rnum{1} $\zeta(x) = |x|^{\alpha}$. When $\rho(x) = \exp(-|x|)$, then we can show that
		\[
			C_1(1+t^{\alpha}) \le \mathcal{G}_{\rho}(t;|\zeta|) \le C_2(1+t^{\alpha}).
		\]

\end{frame}
