\section{Introduction}
In this paper we study the following fractional stochastic equation:
\begin{equation}
	\label{E:SPDE}
	\begin{cases}
		\left(\partial^b_t + \frac{\nu}{2}(-\Delta)^{a/2}\right)\: u(t,x) = I^r_t \left[\sqrt{\theta}\: u(t,x)\: \dot W(x) \right] & \text{$x\in \R^d$, $t>0$} \\
		u(0,\cdot) = 1                                                                                                             & b \in (0,1]               \\
		u(0,\cdot) = 1, \quad \partial_t u(0,\cdot) = 0                                                                            & b \in (1,2)
	\end{cases}
\end{equation}
where $a \in (0,2)$, $b \in (0,2)$, $r\ge 0$, $\nu > 0$ and $\theta>0$. Here the noise $W =
\left\{ W(\varphi) : \varphi \in \mathcal{D}(\R^d)  \right\}$ is a centered and time-independent
Gaussian process, defined on a complete probability space $(\Omega, \mathcal{F}, P)$, with mean zero
and covariance
\begin{align*}
		\mathbb{E}\left[ W(\varphi) W(\psi) \right] = \int_{\R^d} \mathcal{F}\varphi(\xi)\overline{\mathcal{F}\psi(\xi)} \mu(\ud \xi) =: \langle \varphi, \psi \rangle_{\mathcal{H}},
\end{align*}
where $\mu$ refers to the {\it spectral measure}, which is assumed to be a nonnegative and
nonnegative definite tempered measure on $\R^d$. Let $\gamma$ be the Fourier transform of $\mu$,
which is also a nonnegative and nonnegative measure on $\R^d$ thanks to Bochner's theorem.
Throughout of the paper, we use $\mathcal{F}\varphi(\xi) = \int_{\R^d} \exp(-ix \xi) \varphi(x) \ud
x$ to denote the Fourier transform of a test function $\varphi$.

In \eqref{E:SPDE}, $(-\Delta)^{a/2}$ refers to the {\it fractional Laplacian} of order $a$,
$\partial^b_t$ denotes the {\it Caputo fractional differential operator}
\begin{align*}
	\partial^b_tf(t) :=
	\begin{cases}
		\dfrac{1}{\Gamma(m-b)} \int_0^t \ud \tau \dfrac{f^{m}(\tau)}{(t-\tau)^{b+1-m}} & \text{if } m-1 < b < m, \bigskip \\
		\dfrac{\ud^m}{\ud t^m}f(t)                                                     & \text{if } b=m,
	\end{cases}
\end{align*}
where $m$ is an integer, and $I^r_t$ refers to the {\it Riemann-Liouville fractional integral} of order $r>0$
\begin{align*}
	I^r_tf(t) := \frac{1}{\Gamma(r)} \int_0^t (t-s)^{r-1}f(s)\ud s, \quad \text{for } t>0,
\end{align*}
with the convention that when $r=0$, $I^0_t = \text{Id}$ reduces to the identity operator. The
fundamental solutions to \eqref{E:SPDE} consist of a triplet:
\begin{align*}
	\left\{ Z_{a,b,r,\nu,d}(t,x),\:  Z^*_{a,b,r,\nu,d}(t,x), \: Y_{a,b,r,\nu,d}(t,x) : t >0, x \in \R^d \right\}.
\end{align*}
Each member of the triplet can be expressed explicitly in terms of the {\it Fox H-function} which is
much more complicated than the fundamental function for either the heat or wave equation; See
Theorem 4.1 of \cite{CHN19}. Since we are interested in the constant one initial condition (and zero
initial velocity when $b>1$), Theorem 4.1 ({\it ibid.}) implies that the corresponding solution to
the homogeneous equation (i.e. the solution when there is no driving source) is equal to the
constant one. Hence, by setting $G(t,x) := Y_{a,b,r,\nu,d}(t,x)$, \eqref{E:SPDE} can be written as
the following integral equation:
\begin{equation}
	\label{E:spdeint}
	u(t,x) = 1 + \sqrt{\theta}\int_0^t \left( \int_{\R^d}G(t-s,x-y)u(s,y) W(\delta y) \right) \ud s,
\end{equation}
where the stochastic integral is in the {\it Skorohod} sense; see Definition \ref{D:Sol} below.
\bigskip

Regarding the noise, we aim to cover the Riesz kernel case, the fractional noise and a mixture of
them, which is stated in the following assumption:
\begin{assumption}
	\label{A:Noise}
	Let $k \in \{1, \cdots , d\}$ and partition the $d$-coordinates of $x=(x_1, \cdots, x_d)$ into $k$
	distinct groups of size $d_i$ so that $d_1 + \cdots + d_k = d$. Denote $x_{(i)} = (x_{i_1} ,
	\cdots , x_{i_{d_i}})$ to be the coordinates in the $i^{th}$ partition. Assume that the
	correlation function of the Gaussian noise is given by
	\begin{equation}
		\label{E:corfunction}
		\gamma(x) = \prod_{i=1}^k |x_{(i)}|^{-\alpha_i}
	\end{equation}
	where $\alpha_i \in (0, d_i)$. Define $\alpha := \sum_{i=1}^k \alpha_i$.
\end{assumption}

The spectral density corresponding to \eqref{E:corfunction} is  given by $\mu (\ud \xi) =
\varphi(\xi) \ud \xi$ with
\begin{equation}
	\label{E:specfunction}
	\varphi(\xi) = \prod_{i=1}^k C_{\alpha_i,d_i} |\xi_{(i)}|^{-(d_i-\alpha_i)}.
\end{equation}
Moreover, one can find a nonnegative and nonnegative definite $K$ such that $\gamma = K*K$ where
`$*$' denotes the spatial convolution. Indeed, one can choose
\begin{align}
	\label{E:K}
	K(x) =\prod_{i=1}^k \beta_{\alpha_i,d_i} |x_i|^{-(d_i-\alpha_i)}.
\end{align}
The two constants in both \eqref{E:specfunction} and \eqref{E:K} are defined as
\begin{equation}
	\label{E:noiseconstant}
	C_{\alpha,d}     = \pi^{-d/2}2^{-\alpha} \frac{\Gamma((d-\alpha)/2)}{\Gamma(\alpha/2)} \quad \text{and}\quad
	\beta_{\alpha,d} = \pi^{-d/4}\frac{\Gamma((d+\alpha)/4)}{\Gamma((d-\alpha)/4)}\sqrt{ \frac{\Gamma((d-\alpha)/2)}{\Gamma(\alpha/2)}}.
\end{equation}

\begin{example}
	The case $k=1$ in \eqref{E:corfunction} (or \eqref{E:specfunction}) recovers the Riesz kernel case. In this case,
	\begin{align*}
		\gamma(x)  = |x|^{-\alpha}, \quad
		\varphi(x) = C_{\alpha,d}|x|^{-(d-\alpha)} \quad \text{and} \quad
		K(x)       = \beta_{\alpha,d}|x|^{-(d+\alpha)/2}.
	\end{align*}
\end{example}

\begin{example}
	The case when $k=d$ in \eqref{E:corfunction} (or \eqref{E:specfunction}) recovers the {\it
	time-independent fractional noise}.  In this case,
	\begin{align*}
		\gamma(x)    = \prod_{i=1}^d |x_i|^{-\alpha_i}, \quad
		\varphi(\xi) = \prod_{i=1}^d C_{\alpha_i,1} |\xi_i|^{-(1-\alpha_i)}\quad\text{and}\quad
		K(x)         = \prod_{i=1}^d \beta_{\alpha_i,1}|x_i|^{-(1+\alpha_i)/2}.
	\end{align*}
	The stochastic heat equation with such noise was earlier studied by Hu \cite{Hu01Heat}.
\end{example}

Next, we need to make some assumption on the parameters $a,b,r$ of \eqref{E:SPDE} as well as the
dimension $d$. Under theses assumptions, the fundamental function $G$ will be nonnegative and smooth
for $x\ne 0$; see Theorems 4.1 and 4.6 of \cite{CHHH17}.

\begin{assumption}
	\label{A:param}
	The parameters $a,b,r$ and the dimension $d$ must satisfy one of the following conditions:
	\begin{enumerate}
		\item $d \ge 1$, $b\in(0,1)$, $a\in(0,2]$, $r\ge0$;
		\item $d \ge 1$, $b=1$, $a\in (0,2]$, $r=0$ or $r>1$;
		\item $1\le d \le 3$, $1<b<a\le 2$, $r>0$;
		\item $1 \le d \le 3$, $1 < b=a < 2$, $r > \dfrac{d+3}{2}-b$.
	\end{enumerate}
\end{assumption}


We now introduce the following variational constant, which plays an important role in this paper:
\begin{align*}
	\mathcal{M}_a(\Theta f, \theta) := \sup_{g \in \mathcal{F}_a} \left\{ \langle g^2 * g^2, f
	\rangle^{1/2} - \frac{\theta}{2}\: \mathcal{E}_a(g,g) \right\} \quad \text{and} \quad
	\mathcal{M}_a := \mathcal{M}_a(|\cdot|^{-\alpha},1);
\end{align*}
see Section \ref{SS:LD} for more notation and details. Our goal is to develop exact moment
asymptotics to the solution $u(t,x)$ so that it can interpolate the known results for both
stochastic heat and wave equations; see Remark \ref{R:Inter} below. We follow the techniques
laid out in \cite{BCC21} where the stochastic wave equation, namely, $a=b=\nu=2$ and $r=0$, has been
exclusively studied. This paper acts as a generalization of that paper to the fraction stochastic
equation. Here is the main result of this paper.

\begin{theorem}
	\label{T:solvable}
	Under Assumption \ref{A:param}, equation \eqref{E:SPDE} driven by the noise with spectral
	density given in equation \eqref{E:specfunction} has a unique solution $u(t,x)$ in $L^p(\Omega)$ for
	all $p \ge 2$, $t>0$, and $x \in \R^d$ provided that
	\begin{equation}
		\alpha < \min\left( \frac{a}{b}\left[2(b+r)-1\right], 2a, d \right).
	\end{equation}
\end{theorem}

\begin{theorem}
	\label{T:asym}
	Under the setting of Theorem \ref{T:solvable}, we have the following moment asymptotics:
	\begin{equation}
		\label{E:momentasym}
		\begin{aligned}
			\lim_{t_p \to \infty} t_p^{-\beta} \log \Norm{u(t,x)}_p = & \left(\frac{1}{2}\right)\left(\frac{2a}{2a(b+r)- b\alpha} \right)^\beta\\
																																& \times \left(\theta\nu^{-\alpha/a} \mathcal{M}_a^{\frac{2a-\alpha}{a}}\right)^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right),
		\end{aligned}
	\end{equation}
	where
	\begin{align}
		\label{E:beta&tp}
		\beta     := \dfrac{2(b+r)-\dfrac{b\alpha}{a}}{ 2(b+r)-\dfrac{b\alpha}{a} -1 } \qquad \text{and} \qquad
		t_p^\beta := (p-1)^{\beta -1} t^\beta.
	\end{align}
	In particular, for any $p\ge 2$ and $t>0$ fixed, we have
	\begin{align}
		\label{E:momentasym-t}
		\begin{aligned}
			\lim_{t \to \infty} t^{-\beta} \log \E\left(|u(t,x)|^p\right) = & p (p-1)^{\frac{1}{2(b+r)-\frac{b\alpha}{a}-1}} \left(\frac{1}{2}\right)\left(\frac{2a}{2a(b+r)- b\alpha} \right)^\beta \\
                                                                      & \times \left(\theta\nu^{-\alpha/a} \mathcal{M}_a^{\frac{2a-\alpha}{a}}\right)^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right),
		\end{aligned}
	\end{align}
	and
	\begin{align}
		\label{E:momentasym-p}
		\begin{aligned}
			\lim_{p \to \infty} p^{-\beta} \log \E\left(|u(t,x)|^p\right) = & t^\beta \left(\frac{1}{2}\right)\left(\frac{2a}{2a(b+r)- b\alpha} \right)^\beta \\
                                                                      & \times \left(\theta\nu^{-\alpha/a} \mathcal{M}_a^{\frac{2a-\alpha}{a}}\right)^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right),
		\end{aligned}
	\end{align}
	respectively.
\end{theorem}
\begin{proof}
	The	asymptotics in \eqref{E:momentasym-t} and \eqref{E:momentasym-p} are immediate consequences of
	\eqref{E:momentasym}. The matching upper and lower bounds for the limit in \eqref{E:momentasym}
	are proved in Theorems \ref{T:ubound} and \ref{T:lbound} below, respectively.
\end{proof}
\begin{remark}
	\label{R:Inter}
	The asymptotics in \eqref{E:momentasym} interpolates the corresponding results for the stochastic
	wave and heat equations:

	\noindent (1) By setting
	\begin{align*}
		a = b = \nu = 2 \quad \text{and} \quad r = 0,
	\end{align*}
	we have that $t_p = (p-1)^{1/(4-\alpha)} t$, and results in \eqref{E:momentasym},
	\eqref{E:momentasym-t}, and \eqref{E:momentasym-p} recover the corresponding results for the
	stochastic wave equation, namely,  (1.9), (1.10), and (1.11) of \cite{BCC21}, respectively.

	\noindent (2) As for the stochastic heat equation case, we are not aware of the exact asymptotics
	results for SHE with time-independent noise case. However, as explained in Remark 1.9 of
	\cite{BCC21}, one may apply Chen's asymptotic results in \cite{Chen17AIHP} for SHE with
	time-dependent noise, by setting $H_0=1$, to obtain the corresponding time-independent results.
	In particular, by setting
	\begin{align*}
		a = b = 2,\quad \nu = 1 \quad \text{and} \quad r = 0,
	\end{align*}
	we have that $t_p=(p-1)^{2/(4-\alpha)}t$, and results in \eqref{E:momentasym} and
	\eqref{E:momentasym-t} recover the corresponding results for the stochastic heat equation, namely,
	(1.16) and (1.17) of \cite{BCC21}, respectively.
\end{remark}

\section{Some preliminaries}
\label{S:Pre}

\subsection{Malliavin Calculus}
\label{SS:Malliavin}

To briefly summarize \cite{Minicourse09}, we start with a nonnegative and nonnegative definite
tempered measure $\Gamma$ with density $\gamma$ in the sense that $\Gamma(\ud x) = \gamma(x) \ud x$,
and
\begin{align*}
	\int_{\R^d} \Gamma(\ud x) (\varphi * \widetilde{\varphi})(x) \ge 0 \quad \text{for all $\varphi \in \mathscr{S}(\R^d)$}
\end{align*}
where $\widetilde{\varphi}(x) = \varphi(-x)$. In addition we require the existence of an $r>0$ such that
\begin{align*}
	\int_{\R^d}\Gamma(\ud x) \frac{1}{ (1 + |x|^2 )^r} < \infty.
\end{align*}
According to the Bochner theorem, there exists a nonnegative measure $\mu$, often referred to as the
spectral measure on $\R^d$ whose Fourier transform is $\Gamma$ in the sense that for any $\varphi
\in \mathscr(\R^d)$
\begin{align*}
	\int_{\R^d}\Gamma(\ud x) \varphi(x) = \int_{\R^d} \mu(\ud t) \mathcal{F}\varphi (t)
\end{align*}

The functional defined on the space of test functions $\mathcal{D}(\R^d)$
\begin{align*}
	C(\varphi,\psi) = \int_{\R} \mathcal{F}\varphi(\xi) \overline{ \mathcal{F}\psi(\xi) } \mu(\ud \xi)
\end{align*}
is nonnegative define and thus we may associate it as the covariance functional of a zero-mean Gaussian processes, $W:= \{ W(\varphi) : \varphi \in \mathcal{D}(\R^d) \}$. In other words,
\begin{align*}
	\mathbb{E}(W(\varphi)W(\psi)) = \int_{\R} \mathcal{F}\varphi(\xi) \overline{ \mathcal{F}\psi(\xi) } \mu(\ud \xi) =: \langle \varphi, \psi \rangle_{\mathcal{H}}.
\end{align*}
Let $\mathcal{H}$ be the completion of $\mathcal{D}(\R^d)$ with respect to $\langle \cdot, \cdot \rangle_{\mathcal{H}}$ and thus we see $\varphi \mapsto W(\varphi)$ is an isometry from the normed space $(\mathcal{D}(\R^d), \Norm{\cdot}_{\mathcal{H}})$ to $L^2(\Omega)$ where $\Norm{f}_{\mathcal{H}} = \langle f, f \rangle_{\mathcal{H}}$.

We denote $\delta$ the Skorohod integral with respect to $W$ and we denote its domain $\text{Dom}(\delta)$. For any $u\in \text{Dom}(\delta)$, we write
\begin{align*}
	\delta(u) = \int_{\R^d} u(x) W(\delta x).
\end{align*}
Through construction of $\delta$, $\delta(u)$ is a random variable in $L^2(\Omega)$. For a deeper explanation, we direct the reader to Nualart \cite{Nualart18Book}.

\begin{definition}
	\label{D:Sol}
	A random field $u=\{u(t,x) : t \ge 0, x\in\R^d\}$ such that $\Norm{u(t,x)}_2 < \infty$ for all $t
	\ge 0$ and $ x \in \R^d$ is called {\it a mild solution} to the equation \eqref{E:SPDE} if the random
	field $\{G(t-s,x-y)u(s,y): s \in [0,t], y \in \R^d \}$ is Skorohod integrable for each $s\in
	[0,t]$ and $x\in \R^d$ fixed and satisfies the following integral equation:
	\begin{equation}
		u(t,x) = 1 + \sqrt{\theta}\int_0^t \left( \int_{\R^d} G(t-s,x-y)u(s,y) W(\delta y) \right)\ud s.
	\end{equation}
\end{definition}

Through the standard Picard iteration scheme, the solution can be expressed by
\begin{equation}
	\label{E:solwexp}
	u(t,x) = 1 + \sum_{k=1}^\infty \theta^{k/2}I_k(f_k(\cdot,x,t))
\end{equation}
where $I_k : \mathcal{H}^{\otimes k}((\R^d)^k) \to \mathcal{H}_k$ is the $k^{th}$ order Skorohod integral and $\mathcal{H}_k$ is the $k^{th}$ Wiener chaos space. The kernels $f_k(\cdot,x,t) \in \mathcal{H}^{\otimes k}((\R^d)^k)$ are calculated through the Piccard iteration scheme to be
\begin{align}
	\label{E:wker}
	f_n(x_1,...,x_k;x,t) & = \int_0^t \int_0^{t_n}\cdots \int_0^{t_2} G(t-t_n,x-x_n)\cdots G(t_2-t_1,x_2-x_1) \ud t_1\cdots \ud t_n \\
                       & = \int_0^t \int_0^{t_n}\cdots \int_0^{t_2} G(t_1,x-x_n)\cdots G(t_n-t_{n-1},x_2-x_1) \ud t_1\cdots \ud t_n. \nonumber
\end{align}
For ease of notation throughout this article, we write the above integrals as
\begin{align*}
	f_n(x_1,...,x_n;x,t) &= \int_{[0,t]^n_<} G(t-t_n,x-x_n)\cdots G(t_2-t_1,x_2-x_1) \ud \vec{t} \\
											 &= \int_{[0,t]^n_<}G(t_1,x-x_n)\cdots G(t_n-t_{n-1},x_2-x_1) \ud \vec{t}.
\end{align*}
where in general $\ud\vec{z} = \prod_{k=1}^d \ud z_k$ and the dimension will always be easily understood. We denote by $\widetilde{f_n}(\cdot, x, t)$ to be the symmetrization of $f_n(\cdot,x,t)$ in the variables $x_1,\cdots,x_n$:
	\begin{align}
		\widetilde{f_n}(\cdot; x, t) &= \frac{1}{n!} \sum_{\rho \in \Sigma_n} f_n\left(x_{\rho(1)},\cdots, x_{\rho(n)}\right) \\
																 &= \frac{1}{n!} \sum_{\rho \in \Sigma_n} \int_{[0,t]^n_<} G\left(t-t_n,x-x_{\rho(n)}\right)\cdots G\left(t_2-t_1,x_{\rho(2)}-x_{\rho(1)}\right) \ud \vec{t}
	\end{align}
where $\Sigma_n$ is the set of all permutations of $\{1,\cdots,n\} $. In particular, for $x=0$ we have
\begin{align}
	\label{E:fperm0}
	\widetilde{f_n}(\cdot; 0, t) = \frac{1}{n!} \sum_{\rho \in \Sigma_n} \int_{[0,t]^n_<} G(t_1,x_{\rho(1)})\cdots G(t_n-t_{n-1},x_{\rho(n)}-x_{\rho(n-1)}) \ud \vec{t}.
\end{align}
By setting $t_{n+1}=t$, the Fourier transform of the kernels, $f_n$, is given by
\begin{align}
	\label{E:fweker}
	\mathcal{F}f_n(\cdot;x,t)(\xi_1,...,\xi_n) = e^{-i \left(\sum_{j=1}^n \xi_j \right) \cdot x} \int_{[0,t]^n <} \prod_{k=1}^n \overline{\mathcal{F}G(t_{t_k+1}-t_k,\cdot)\left(\sum_{j=1}^k \xi_j\right) } 	\ud \textbf{t}.
\end{align}

Recall the notation above that $G(t,x) = Y_{a,b,r,d}(t,x)$. The following result shows an important scaling property that both $\mathcal{F}G(t,\cdot)(\xi)$ and $\Norm{\widetilde{f_n}(\cdot,t,x)}_{\mathcal{H}^{\otimes n}}$ have.

\begin{lemma}
	\label{L:Scale}

	\begin{equation}
		\label{E:greenscale}
		\mathcal{F}G(t,\cdot)(c\xi) = c^{-\frac{a}{b}(b+r -1)}\mathcal{F}G\left(c^{\frac{a}{b}}t,\cdot\right)( \xi)  \quad \text{and} \quad
		\mathcal{F}G(ct,\cdot)(\xi) = c^{b+r -1}\mathcal{F}G(t,\cdot)(c^{b/a} \xi),
	\end{equation}
	and
	\begin{equation}
		\label{E:normscale}
		\Norm{\widetilde{f_n}(\cdot,0,t)}_{\mathcal{H}^{\otimes n}}^2 = t^{[2(b + r ) -b \alpha / a]n} \Norm{\widetilde{f_n}(\cdot,0,1)}_{\mathcal{H}^{\otimes n}}^2.

Recall the following standard result about the existence and uniqueness of the solution when the
solution can be written as the Weiner chaos expansion as in \eqref{E:solwexp}.
\begin{theorem}
	Fix any $T \in (0,\infty]$. Suppose that $f_n(\cdot,x,t) \in \mathcal{H}^{\otimes n}$ for any $t
	\in (0,T)$, $x \in R^d$ and $n \ge 1$. Then equation \eqref{E:SPDE} has a unique $L^2(\Omega)$
	solution on $(0,T) \times \R^d$ if and only if the series \eqref{E:solwexp} converges in
	$L^2(\Omega)$ for any $(t,x) \in (0,T) \times \R^d$, which is equivalent to the convergence of the
	series \eqref{E:2mom}. In this case, the solution is given by \eqref{E:solwexp} with the second moment
	given by
	\begin{equation}
		\label{E:2mom}
		\mathbb{E}\left(u(t,x)^2\right) = \sum_{n \ge 0} \theta^n n!  \Norm{\widetilde{f_n}(\cdot,x,t)}_{\mathcal{H}^{\otimes n}}^2 \quad \text{for all $(t,x) \in (0,T) \times \R^d$}.
	\end{equation}
\end{theorem}

\subsection{Some large deviation results}
\label{SS:LD}

Let us first define
\begin{align*}
	\rho := \rho_{\nu,a} = \sup_{\Norm{f}_2 =1} \int_{\R^d} \left[ \int_{\R^d} \frac{f(x+y)f(y)}{\sqrt{1+\frac{\nu}{2}|x+y|^a } \sqrt{1+\frac{\nu}{2}|y|^a} } \ud y \right]^2 \mu(\ud x).
\end{align*}
Recall that when $\mu(\ud \xi) = C_{\alpha,d}|\xi|^{-(d-\alpha)} \ud \xi$ which corresponds to the
Riesz kernel $\gamma(x) = |x|^{-\alpha}$ for some $\alpha \in (0,d)$, Bass, Chen and Rosen
\cite{BCR09} established that when $a \in (0,2]$ and $\nu = 2$,
\begin{equation}
	\label{E:rhowavedev}
	\lim_{n \to \infty} \frac{1}{n} \log \left[ \frac{1}{(n!)^2} \int_{(\R^d)^n} \left( \sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+ |\sum_{j=k}^n \xi_{\sigma(j)}|^a}\right)^2 \mu (\ud \vec{\xi})\right] = \log\left( \rho_{2,a} \right),
\end{equation}
and
\begin{equation}
	\label{E:oldrhovarrep}
	\rho_{2,a} = \Lambda_{\alpha}^{2-(\alpha/a)}
\end{equation}
 where
\begin{align}
	\Lambda_{\alpha}   & := \sup_{g \in \mathcal{F}_a} \left\{ \left( \int_{\R^{2d}} \frac{g^2(x_1)g^2(x_2)}{|x_1+x_2|^\alpha} \ud \vec{x} \right)^{1/2} - \mathcal{E}_a(g,g) \right\} \\
                     & = \sup_{g \in \mathcal{F}_a} \left\{ \langle g^2 * g^2, |\cdot|^{-\alpha} \rangle_{L^2(\R^d)} - \mathcal{E}_a(g,g) \right\} \nonumber \quad \text{and}        \\
	\mathcal{E}_a(g,g) & := (2\pi)^{-d} \int_{\R^d} |\xi|^a |\mathcal{F}g(\xi)|^2 \ud \xi.
\end{align}
The following result shows the relationship between $\rho_{2,a}$ and $\rho$ and well as the generalization of \eqref{E:rhowavedev} for $\rho.$

\begin{lemma}
	\label{L:rhovar}
	For the Riesz kernel case, i.e., when $\gamma(x) = |x|^{-\alpha}$ for some $\alpha \in (0,d)$, it
	holds that
	\begin{equation}
	\label{E:rhoLam}
		\rho_{\nu, a} = \left(\frac{\nu}{2}\right)^{-\alpha/a}  \Lambda_{\alpha}^{2-(\alpha/a)}, \quad
		\text{for any $\nu >0 $ and $a \in (0,2]$}.
	\end{equation}
	In addition,
	\begin{equation}\label{E:LogRho}
			\lim_{n \to \infty} \frac{1}{n} \log \left[ \frac{1}{(n!)^2} \int_{(\R^d)^n} \left( \sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+ \frac{\nu}{2}|\sum_{j=k}^n \xi_{\sigma(j)}|^a}\right)^2 \mu (\ud \vec{\xi})\right] = \log\left( \rho_{\nu, a} \right).
	\end{equation}
\end{lemma}
This lemma can be proved in the standard scaling arguments together with \eqref{E:oldrhovarrep}. For
completeness, the proof is presented in the Appendix of the paper.  \bigskip

Next we show that $\rho$ is finite. We do this by showing $\rho_{2,a}$ is finite and then noting
that this implies that $\rho$ is finite by \ref{L:rhovar}. We stated in the introduction
that Bass, Chen and Rosen proved in \cite{BCR09} that $\rho$ is finite for the case of the Riesz
kernel. We mimic his proof but were he uses the Hardy-Littlewood-Sobolev theorem, we make use of the
weak Young's inequality instead.

\begin{lemma}
	\label{L:rhofinite}
	For any $f,g,h$ with $h\ge0$ and for $\varphi$ given as in equation \eqref{E:specfunction}, we have the following:
	\begin{align*}
		\left( \int_{\R^d} \left[ \int_{\R^d} \dfrac{|f(x+y)g(y)|}{\sqrt{h(x+y)}\sqrt{h(y)}} \ud y \right]^p \varphi(x) \ud x \right)^{1/p} \le C \Norm{f}_2 \Norm{g}_2 \Norm{h^{-1}}_{pd/\alpha}
	\end{align*}
	where $\alpha = \sum_{=1}^k \alpha_i$ with $k \in \{1, \cdots ,d\}$, $\alpha_i \in (0,d_i)$ and $\varphi(x) = \prod_{i =1}^k |x_{(i)}|^{-(d_i-\alpha_i)}$ where $x_{(i)} = (x_{i_1},\cdots , x_{i_{d_i}} )$ is the $i^{th}$ group of $x=(x_1, \cdots, x_d)$.
\end{lemma}
\begin{proof}
	By observing the proof of Lemma 1.6 \cite{BCR09}, we notice that we only need to prove equation (1.27) which says
	\begin{equation}
		\label{E:fracrhofinite}
		\int_{\R^d}\int_{\R^d} F(y)G(x)\varphi(x-y) \ud y \ud x \le C \Norm{F}_{2d/(d+\alpha)} \Norm{G}_{2d/(d+ \alpha)}.
	\end{equation}
	where
	\begin{align*}
		F(x) = \frac{|f(x)|}{(h(x))^{p/2}} \quad \text{and} \quad G(x) = \frac{|g(x)|}{(h(x))^{p/2}}
	\end{align*}
	If we can do this then the result follows by copying the rest of Chen's proof. To prove \eqref{E:fracrhofinite}, we use the weak version of Young's inequality. Recall that weak Young's inequality \cite{LiebLoss01} says
	\begin{align*}
		\left|	\int_{\R^d}\int_{\R^d}a(x)b(x-y)c(y) \ud x \ud y \right| \le K_{p,q,r,d} \Norm{a}_p\Norm{b}_{q,w}\Norm{c}_r
	\end{align*}
	where $p,q,r >1$ and $1/p + 1/q + 1/r =2$ and
	\begin{align*}
		\Norm{b}_{q,w} = \sup_{A} |A|^{-1/q'} \int_A |f(x)| \ud x
	\end{align*}
	where $A$ is an arbitrary Borel set of finite measure. So to prove \eqref{E:fracrhofinite} we wish to be able to apply this with
	\begin{align*}
		p=r=2d/(d+\alpha) \quad \text{and} \quad q= d/\left(d-\sum_{j=1}^d\alpha_j \right) = d/(d-\alpha)
	\end{align*}
	and, in addition, with
	\begin{align*}
		a=F, \quad c=G \quad \text{and} \quad b=\varphi
	\end{align*}
	Again, by following the steps of the proof from the Xia Chen paper, we note that it suffices to show
	that $\Norm{\varphi}_{q,w}$ is finite with $q = d/(d-\alpha)$. Note that $1/q' = 1 - 1/q =
	\alpha/d.$

	First we define $A_R := A_1 \times \cdots \times A_k$ where $A_i \subset \R^{d_i}$ and $A_i =
	[-R,R]$ if $d_i = 1$ and $A_i = B_R(0)$ if $d_i > 1$ where $B_R(0)$ is the ball centered at the
	origin with radius $R$. With this we have that
	\begin{equation}
		\label{E:int}
		\int_{A_i} |x_{(i)}|^{-(d_i-\alpha_i)} \ud x_{(i)} =
		\begin{cases}
			2 \dfrac{R^{\alpha_i}}{\alpha_i}                     & , d_i =1 \\
			|\mathbb{S}^{d_i - 1}|\dfrac{R^{\alpha_i}}{\alpha_i} & , d_i >1
		\end{cases}
	\end{equation}
	where we used polar coordinated to calculate the integral when $d_i > 1$ and $|\mathbb{S}^{d_i-1}|$
	is the surface area of the sphere in $\R^{d_i}$. Also,
	\begin{equation}
		\label{E:leb}
		|A_i|  =
		\begin{cases}
			2 R                                        & , d_i =1 \\
			|\mathbb{S}^{d_i - 1}|\dfrac{R^{d_i}}{d_i} & , d_i >1.
		\end{cases}
	\end{equation}
	Recall that $1/q' = \alpha/d$. Then by combining \eqref{E:int} and \eqref{E:leb} we see that
	\begin{align*}
	|A_R|^{-1/q'}	\int_{A_R} \varphi(x) \ud x &= \left(\prod_{i =1}^k |A_i|\right)^{-1/q'} \prod_{i=1}^k \int_{A_i} |x_{(i)}|^{-(d_i-\alpha_i)} \ud x_{(i)} \\
																						&= R^{-\alpha} \left( \prod_{i =1}^k\begin{cases}
				2                            & , d_i =1 \\
				|\mathbb{S}^{d_i - 1}| / d_i & , d_i >1.
	\end{cases} \right) \times R^\alpha 	\left(\prod_{i =1}^k \begin{cases}
				2/\alpha_i                        & , d_i =1 \\
				|\mathbb{S}^{d_i - 1}| / \alpha_i & , d_i >1
	\end{cases} \right) \\
																						&= K
	\end{align*}
	where the constant $K$ does not depend on $R$. Hence we that
	\begin{align*}
		\sup_{R>0} 	|A_R|^{-1/q'}	\int_{A_R} \varphi(x) \ud x = K.
	\end{align*}
	and by the symmetry of $\varphi$ we can in fact say that
	\begin{align*}
		\sup_{B} 	|B|^{-1/q'}	\int_{B} \varphi(x) \ud x = K
	\end{align*}
	where $B$ denotes the Borel sets of finite measure. Hence, $\varphi \in L_{q,w}(\R^d)$ with $q =
	d/(d-\alpha)$. This completes the proof of Lemma \ref{L:rhofinite}.
\end{proof}

We finish this section by making a slight generalization of $\Lambda_{\alpha}$. For this, we define
\begin{equation}
	\mathcal{M}_a(\Theta f , \theta) = \sup_{g \in \mathcal{F}_a} \left\{ \langle g^2 * g^2, f \rangle^{1/2} - \frac{\theta}{2}\mathcal{E}_a(g,g) \right\}.
\end{equation}
One notices that by making this generalization, we can apply Theorem \ref{T:solvable} to the SPDE
\eqref{E:SPDE} driven by an arbitrary Gaussian noise with correlation function $f$ as long as one
can show that $\rho$ is finite. By slightly modifying the proof of Lemma 2.3 \cite{BCC21}, one can
show that
\begin{equation}
		\mathcal{M}_a(\Theta f , \theta) = \Theta^{\frac{a}{2a-\alpha}}\theta^{-\frac{\alpha}{2a-\alpha}} \mathcal{M}_a( f , 1).
\end{equation}
Finally we note that
\begin{equation}
	\Lambda_{\alpha} = \mathcal{M}_a(|\cdot|^{-\alpha},2) = 2^{-\frac{\alpha}{2a-\alpha}}\mathcal{M}_a(|\cdot|^{-\alpha},1)
\end{equation}
and by defining $\mathcal{M}_a := \mathcal{M}_a(|\cdot|^{-\alpha},1) $ and by applying Theorem \ref{L:rhovar} we see that
\begin{equation}
	\label{E:rhoMrep}
	\rho = \left(\frac{\nu}{2}\right)^{-\alpha/a}
	\left[2^{-\frac{\alpha}{2a-\alpha}}\mathcal{M}_a\right]^{2-(\alpha/a)} = \nu^{-\alpha/a}
	\mathcal{M}_a^{\frac{2a-\alpha}{a}}.
\end{equation}

We first introduce some notation that will be used throughout the remainder of this paper. We start by defining $L_n(\vec{y})$ to be the Laplace transform of $\widetilde{f_n}(\vec{y},0,\cdot)$ evaluated at $1$:
\begin{align*}
	L_n(\vec{y}) & = n! \int_0^\infty e^{-t}\widetilde{f_n}(\vec{y};0,t) \ud t \\
							 & = \sum_{\sigma \in \Sigma_n} \int_0^\infty e^{-t} \int_{[0,t]^n<} \prod_{k=1}^n G(s_k - s_{k-1}, y_{\sigma(k)}-y_{\sigma(k-1)}) \ud \vec{s} \ud t
\end{align*}
with the convention that. $s_0 = y_{\sigma(0)} = 0$. Next we define,

	H_n(t, \vec{x}) & = n! \int_{(\R^d)^n} \prod_{k=1}^n K(x_k - y_k) \widetilde{f_n}(\vec{y};0,t) \ud \vec{y} \\
									& = \sum_{\sigma \in \Sigma_n} \int_{[0,t]^n <} \int_{(\R^d)^n} \prod_{k=1}^n K(x_k-y_k) 	\prod_{k=1}^n G(s_k - s_{k-1}, y_{\sigma(k)} - y_{\sigma(k-1)}) \ud \vec{y} \ud \vec{s}
\end{align*}
where in the last line we used relation \eqref{E:fperm0} for expressing  $\widetilde{f_n}(\cdot;0,t)$. Next, similarly to $L_n$, we define $L(x)$ and can calculate $\mathcal{F}L(\xi)$ to be

	L(x) = \int_0^\infty e^{-t}G(t,x) \ud t \quad \text{and} \quad \mathcal{F}L(\xi) = \int_0^\infty e^{-t}\mathcal{F}G(t,\cdot)(\xi) \ud t = \dfrac{1}{1 + \frac{\nu}{2}|\xi|^a}
\end{equation}
and through a change of variables, one sees that the Laplace transform of convolution in the time variable become a product:
\begin{equation}
		L_n(\vec{y}) = \sum_{\sigma \in \Sigma_n} L(y_{\sigma(1)}) L(y_{\sigma(2)}-y_{\sigma(1)}) \cdots L(y_{\sigma(n)}-y_{\sigma(n-1)})
\end{equation}
and using this and \eqref{E:LandFL}, we can calculate $\mathcal{F}L_n(\vec{\xi})$:
\begin{equation}
	\label{E:FLfn}
	\mathcal{F}L_n(\vec{\xi}) = \sum_{\sigma \in \Sigma_n} \prod_{k=1}^{n} \dfrac{1}{1+ \frac{\nu}{2}\left|\sum_{j=k}^n \xi_j \right|^a}.
\end{equation}

\begin{lemma}
	\label{L:Hneq1}
	Under the assumptions of Theorem \ref{T:solvable}, we have that
	\begin{equation}
		\int_{(\R^d)^n} \left[ \int_0^\infty e^{-t} H_n(t,\vec{x}) \ud t \right]^2 \ud \vec{x} = \int_{(\R^d)^n} | \mathcal{F}L_n(\vec{\xi}) |^2 \mu(\ud \vec{\xi}).
	\end{equation}
\end{lemma}
\begin{proof}
	Notice that
	\begin{align*}
      & \int_0^\infty e^{-t} H_n(t,\vec{x}) \ud t                                                                                                                                                                   \\
      & = \sum_{\sigma \in \Sigma_n} \int_0^\infty e^{-t} \int_{[0,t]^n <} \int_{(\R^d)^n} \prod_{k=1}^n K(x_k-y_k) 	\prod_{k=1}^n G(s_k - s_{k-1}, y_{\sigma(k)} - y_{\sigma(k-1)}) \ud \vec{y} \ud \vec{s} \ud t \\
      & = \int_{\R^{dn}} \prod_{k=1}^n K(x_k-y_k) L(\vec{y}) \ud \vec{y}.
	\end{align*}
Therefore,
	\begin{align*}
		\int_{\R^{dn}}\left[\int_0^\infty e^{-t} H_n(t,\vec{x}) \ud t \right]^n \ud \vec{x} &= 	\int_{\R^{dn}} \left[  \int_{\R^{dn}} \prod_{k=1}^n K(x_k-y_k) L(\vec{y}) \ud \vec{y} \right]^2 \ud \vec{x}
		\\&= \int_{\R^{dn}} |\mathcal{F}L_n(\vec{\xi})|^2 \prod_{k=1}^n\mu(\ud \xi_k)
	\end{align*}
where the last line follows from Plancherel's theorem and the fact that $K*K =\gamma$. \textcolor{red}{consider adding the plancherel details}
\end{proof}

\begin{lemma}
	\label{L:inteq1}
	Under the assumptions of Theorem \ref{T:solvable}, we have that
	\begin{equation}
		\Norm{\widetilde{f_n}(\cdot,0;t)}_{\mathcal{H}^{\otimes n}}^2 = \frac{1}{(n!)^2}\int_{(\R^d)^n} H_n^2(t,\vec{x}) \ud \vec{x}.
	\end{equation}
\end{lemma}
\begin{proof}
Writing $\gamma(y_k-y_k') = \int_{\R^d} K(y_k-x_k)K(y_k'-x_k) \ud x_k$, we see that

	\Norm{\widetilde{f_n}(\cdot,0;t)}_{\mathcal{H}^{\otimes n}}^2
  & = \int_{(\R^d)^n} \int_{(\R^d)^n} \prod_{k=1}^n \gamma(y_k-y_k') \widetilde{f}_n(\vec{y},0;t)\widetilde{f}_n(\vec{y'},0;t) \ud \vec{y} \ud \vec{y'} \\
  & = \int_{(\R^d)^n} \left[ \int_{(\R^d)^n} \prod_{k=1}^n K(y_k-x_k) \widetilde{f}_n(\vec{y},0;t) \ud \vec{y} \right]^2 \ud (\vec{x})                  \\
  & =  \int_{(\R^d)^n} \left[ \frac{1}{n!} H_n(t,\vec{x}) \right]^2 \ud \vec{x}.
\end{align*}

\end{proof}

If $f$ is a non-decreasing function on $[0,\infty)$, then the following reverse Cauchy-Schwarz inequality holds, which plays a key role in our developments.

\begin{lemma}
	\label{L:intineq1}
	If $f:[0,\infty) \to [0,\infty)$ is a non-decreasing function, then
		\begin{equation}
			2\int_0^\infty e^{-2t}f^2(t) \ud t \le \left( \int_0^\infty e^{-t} f(t) \ud t \right)^2.
		\end{equation}
\end{lemma}
\begin{proof}
	Let $\tau$ and $\tau'$ be two independent random variables with exponential distribution of mean
	one. Then $\tau \wedge \tau'$ has an exponential distribution with density $g(t)=2e^{-2t}$ and
	\begin{align*}
		\left( \int_0^\infty e^{-t}f(t) \ud t \right)^2 = \mathbb{E}[ f(\tau) ] \mathbb{E}[ f(\tau') ]
                                                    = \mathbb{E}[ f(\tau) f(\tau')]
                                                    = 2 \int_0^\infty e^{-2t} f^2(t) \ud t.
	\end{align*}
\end{proof}

Denote
\begin{equation}
	\label{E:Tn}
	T_n = T_n(\nu,a) := \int_{(\R^d)^n} \left[\sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+\frac{\nu}{2}\left|\sum_{j=k}^n \xi_{\sigma(j)} \right|^a}\right]^2 \mu(\ud \vec{\xi}).
\end{equation}
By applying the substitution $\xi_k' = \left(\frac{\nu}{2}\right)^{1/a}\xi_k$ and the scaling
property for $\varphi$, we may rewrite $T_n$ as
\begin{equation}
	T_n = \left( \frac{\nu}{2} \right)^{-n\alpha/a} T_n(2,a)
\end{equation}

\begin{lemma}
	\label{L:muassumption}
	Under the assumption of Theorem \eqref{E:momentasym}, we have that
	\begin{equation}
		\label{E:muassumption}
		C_\mu = C_\mu(\nu,a) := \int_{(\R^d)} \left(\frac{1}{1+\frac{\nu}{2}|\xi|^a}\right)^2 \mu(\ud \xi) < \infty \quad \text{and} \quad T_n \le (n!)^2(C_\mu)^2
	\end{equation}
\end{lemma}
\begin{proof}
As in the proof of Lemma 3.6 \cite{BCC21}, we apply a spherical substitution and get that
	\begin{align*}
		\int_{(\R^d)} \left(\frac{1}{1+|\xi|^a}\right)^2 \mu(\ud \xi) = \left(\int_{S^{d-1}}\varphi(\gamma) \ud \sigma(\gamma)\right) \left( \int_0^\infty \frac{\rho^{\alpha-1}}{(1+\rho^{a})^2} \ud \rho \right).
	\end{align*}
The first integral is shown to be finite in \cite{BCC21}. The second integral is finite if
	\begin{align}\label{E:alpha2}
		2a - \alpha +1 > 1
	\end{align}
which is already satisfied under the assumptions of Theorem \ref{T:solvable}. Hence the first relation in \eqref{E:muassumption} is proven.

The second relation is proven the same was as in Lemma 3.6 \cite{BCC21}. Here is the proof.
\begin{align}
	T_n & \le n! \int_{(\R^d)^n} \sum_{\sigma \in \Sigma_n}\left[ \prod_{k=1}^n \frac{1}{1+\frac{\nu}{2}\left|\sum_{j=k}^n \xi_{\sigma(j)} \right|^a}\right]^2 \mu(\ud \xi_1)\cdots \mu(\ud \xi_n) \nonumber \\
      & = (n!)^2 \int_{(\R^d)^n} \left[ \prod_{k=1}^n \frac{1}{1+\frac{\nu}{2}\left|\sum_{j=k}^n \xi_j \right|^a}\right]^2 \mu(\ud \xi_1)\cdots \mu(\ud \xi_n) \nonumber                                   \\
      & \le (n!)^2 \left[ \sup_{\eta \in \R^d} \int_{(\R^d)^n}  \prod_{k=1}^n \left( \frac{1}{1+\frac{\nu}{2}\left| \xi + \eta \right|^a}\right)^2 \mu(\ud \xi) \right]^n \nonumber                        \\
      & = (n!)^2 \left[ \int_{(\R^d)^n}  \prod_{k=1}^n \left(\frac{1}{1+\frac{\nu}{2}\left| \xi \right|^a}\right)^2 \mu(\ud \xi) \right]^n \nonumber                                                       \\
      & = (n!)^2(C_\mu)^n \label{E:lem62step}
\end{align}
where the last line follows by Lemma 4.1 of Balan and Song \cite{BS17} or by Lemma 3.1 in
\cite{Chen19AIHP}.
\end{proof}


The following lemma shows that $\widetilde{f}_n(\cdot,0;t) \in \mathcal{H}^{\otimes n}$ for any $t>0$.
\begin{lemma}
	\label{L:LNfnineq}
	Under the assumptions of Theorem \ref{T:solvable} and for any $t>0$.
	\begin{align}
		\label{E:LNfnineq}
		\int_0^\infty e^{-t} \Norm{\widetilde{f_n}(\cdot,0;t)}_{ \mathcal{H}^{ \otimes n } }^n \ud t
		\le \dfrac{2^{n\left[2(b+r)-\frac{b\alpha}{a}\right]}}{(n!)^2} T_n
		\le 2^{n\left[2(b+r)-\frac{b\alpha}{a}\right]} (C_\mu')^n.
	\end{align}
\end{lemma}

\begin{proof}
	We first derive the following scaling property for $\mathcal{F}\widetilde{f}_n$ where
	$\widetilde{f}_n$ is the kernel of the $n^{th}$ Weiner integral in the chaos expansion of the
	solution \eqref{E:solwexp}:
	\begin{equation}
		\label{E:Ffnscale}
		\mathcal{F}\widetilde{f}_n(\cdot,0,ct)(\xi_1,\cdots,\xi_n) = c^{n(b+r)} \mathcal{F}\widetilde{f}_n(\cdot,0,t)(c^{b/a}\xi_1, \cdots, c^{b/a}\xi_n).
	\end{equation}
	Let $c>0$. Notice that
	\begin{align*}
		\mathcal{F}\widetilde{f}_n(\cdot,0,ct)(\xi_1,\cdots,\xi_n) = \frac{1}{n!} \sum_{\sigma \in \Sigma_n} \mathcal{F}f_n(\cdot,0,ct)(\xi_{\sigma(1)},\cdots, \xi_{\sigma(n)}).
	\end{align*}
	So it suffices to just look at the $\mathcal{F}f(\cdot,0,ct)(\xi_{\sigma(1)},\cdots, \xi_{\sigma(n)})$ terms individually.
	\begin{align*}
		\mathcal{F}f_n(\cdot,0,ct)(\xi_1,\cdots, \xi_n) & = \int_{[0,ct]^n_<} \prod_{k=1}^n \overline{ \mathcal{F}G(t_{k+1} - t_k, \cdot)\left( \sum_{j=1}^k \xi_j \right) }  \ud \vec{t} \\
																										& = \int_{[0,t]^n_<} \prod_{k=1}^n \overline{ \mathcal{F}G\left(c(t_{k+1} - t_k), \cdot \right)\left( \sum_{j=1}^k \xi_j \right) } c^n \ud \vec{t}
	\end{align*}
	where we applied the substitution $t_i' = t_i/c$ by applying the scaling property of
	$\mathcal{F}G$ we wee that
	\begin{align*}
		\int_{[0,t]^n_<} \prod_{k=1}^n  \overline{ \mathcal{F}G\left(c(t_{k+1} - t_k), \cdot \right)\left( \sum_{j=1}^k \xi_j \right) } c^n \ud \vec{t}
		& = \int_{[0,t]^n_<} \prod_{k=1}^n \overline{ c^{b+r-1} \mathcal{F}G\left(t_{k+1} - t_k, \cdot \right)\left( c^{b/a} \sum_{j=1}^k \xi_j \right) } c^n \ud \vec{t} \\
		& = c^{n(b+r)} \mathcal{F}f_n(\cdot,0,t)(c^{b/a}\xi_1, \cdots, c^{b/a}\xi_n).
	\end{align*}
	Now by using this scaling property we see that
	\begin{align*}
		\int_0^\infty e^{-t} \Norm{ \widetilde{f}_n(\cdot,0,t) }_{\mathcal{H}^{\otimes n}}^2 \ud t
     & = \int_0^\infty e^{-t} \int_{\R^{nd}} \left| \mathcal{F}\widetilde{f_n}(\cdot,0;t)(\xi_1,\cdots,\xi_n) \right|^2 \mu(\ud \vec{\xi})  \ud t                                                             \\
     & = \int_0^\infty e^{-2t} \int_{\R^{nd}} \left| \mathcal{F}\widetilde{f_n}(\cdot,0;2t)(\xi_1,\cdots,\xi_n) \right|^2 \mu(\ud \vec{\xi})  2 \ud t                                                         \\
     & = 2^{2n(b+r)}\int_0^\infty 2e^{-2t} \int_{\R^{nd}} \left| \mathcal{F}\widetilde{f_n}(\cdot,0,t)(2^{b/a}\xi_1,\cdots,2^{b/a}\xi_n) \right|^2 \mu(\ud \vec{\xi})   \ud t                                 \\
     & = 2^{2n(b+r)}\int_0^\infty 2e^{-2t} \int_{\R^{nd}} \left| \mathcal{F}\widetilde{f_n}(\cdot,0;t)(\xi_1,\cdots,\xi_n) \right|^2 2^{\frac{-nbd}{a}} 2^{\frac{nb(d-\alpha)}{a}} \mu(\ud \vec{\xi})   \ud t \\
     & = 2^{n(2(b+r) - b\alpha/a)}  \int_0^\infty 2e^{-2t} \int_{\R^{nd}} \left| \mathcal{F}\widetilde{f_n}(\cdot,0;t)(\xi_1,\cdots,\xi_n) \right|^2 \mu(\ud \vec{\xi})   \ud t                               \\
     & =  \frac{2^{n(2(b+r) - b\alpha/a)}}{(n!)^2}  \int_0^\infty 2e^{-2t} \int_{\R^{nd}} H_n(t,\vec{x})^2 \ud \vec{x}   \ud t
	\end{align*}
	and now we apply Lemmas \ref{L:inteq1} and \ref{L:intineq1} as well as \eqref{E:FLfn} to get
	\begin{align*}
		\frac{2^{n(2(b+r) - b\alpha/a)}}{(n!)^2}  \int_0^\infty 2e^{-2t} \int_{\R^{nd}} H_n(t,\vec{x})^2 \ud \vec{x}   \ud t
    & \le \frac{2^{n(2(b+r) - b\alpha/a)}}{(n!)^2}  \int_{\R^{nd}} \left(\int_0^\infty e^{-t}  H_n(t,\vec{x}) \ud t \right)^2 \ud \vec{x} \\
    & =   \frac{2^{n(2(b+r) - b\alpha/a)}}{(n!)^2}  \int_{\R^{nd}} | \mathcal{F}L_n(\vec{\xi}) |^2\ud \vec{\xi}                           \\
    & = \frac{2^{n(2(b+r) - b\alpha/a)}}{(n!)^2}  T_n,
	\end{align*}
	which proves the first inequality in \eqref{E:LNfnineq}. An application of Lemma
	\ref{L:muassumption} gives the second inequality in \eqref{E:LNfnineq}.
\end{proof}

\begin{proof}[Proof of Theorem \ref{T:solvable}]
	First recall \eqref{E:Ffnscale}, which says
	\begin{align*}
		\mathcal{F}\widetilde{f_n}(\cdot,0,ct)(\xi_1,..., \xi_n) = c^{n(b+r)}\mathcal{F}\widetilde{f_n}(\cdot,0,t)(c^{b/a}\xi_1,...,c^{b/a}\xi_n).
	\end{align*}
	In order to prove the theorem we need to show that $\sum_{n \ge 0} \theta^n n! \Norm{
	\widetilde{f_n}(\cdot,0,t) }_{ \mathcal{H}^{\otimes n} }^2 < \infty.$ From the scaling property
	\eqref{E:normscale} we see that
	\begin{align*}
		\int_0^\infty e^{-t} \Norm{ \widetilde{f_n}(\cdot,0,t) }_{ \mathcal{H}^{\otimes n} }^2 \ud t
    & = \Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \int_0^\infty e^{-t} t^{ \left[ 2(b+r)-\frac{b \alpha}{a} \right] n } \ud t \\
    & = \Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right)
	\end{align*}
	as long as we have
	\begin{equation}
		\label{E:alpha3}
		2(b+r)-\frac{b\alpha}{a} >0.
	\end{equation}
	By using Lemma \ref{L:LNfnineq} we can deduce that
	\begin{align*}
		\Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2
    & = \dfrac{ \int_0^\infty e^{-t} \Norm{ \widetilde{f_n}(\cdot,0,t) }_{ \mathcal{H}^{\otimes n} }^2 \ud t }{ \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right) } \\
    & \le \dfrac{ 2^{n \left[ 2(b+r)- \frac{b \alpha}{a} \right]} }{ \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right) } T_n                                        \\
    & \le \dfrac{ \left(2^{ \left[ 2(b+r)- \frac{b \alpha}{a} \right]}C_\mu' \right)^n }{\Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right) }.
	\end{align*}
	Hence, there exists a constant $k_\alpha$ depending on $\alpha$ such that
	\begin{align*}
		\Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \le k_{\alpha} \dfrac{ \left(2^{ \left[ 2(b+r)- \frac{b \alpha}{a} \right]}C_\mu' \right)^n }{ (n!)^{ 2(b+r) - b\alpha/a } }.
	\end{align*}
	Therefore,
	\begin{align*}
		\sum_{n \ge 0} \theta^n n! \Norm{ \widetilde{f_n}(\cdot,0,t) }_{ \mathcal{H}^{\otimes n} }^2
		\le  \sum_{n \ge 0} \theta^n   k_{\alpha} \dfrac{ \left(2^{ \left[ 2(b+r)- \frac{b \alpha}{a} \right]}C_\mu' \right)^n }{ (n!)^{ 2(b+r) - 1 - b\alpha/a }},
	\end{align*}
	where the sum is finite provided that
	\begin{equation}\label{E:alpha4}
		2(b+r) - 1 - b\alpha/a > 0.
	\end{equation}
	This completes the proof of Theorem \ref{T:solvable}.
\end{proof}

\section{Upper bound of the asymptotics}
\label{S:upperbd}

In this section we use the result obtained in Theorem \ref{L:p=2} and the hypercontractivity
property given by Theorem B.1 in \cite{BCC21} to prove the following theorem.

\begin{theorem}
	\label{T:ubound}
	Under the assumptions of Theorem \ref{T:solvable}
	\begin{equation*}
		\limsup_{n \to \infty} \; t_p^{-\beta} \log \Norm{u(t,x)}_p  \le \left(\frac{1}{2}\right)  \left( \frac{2a}{2a(b+r)- b\alpha} \right)^\beta (\theta\rho)^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right).
	\end{equation*}
\end{theorem}

In order to prove this theorem, we need to first prove the case when $p=2$ (Lemma \ref{L:p=2} below)
and then use the hypercontractivity to establish the general case.

\begin{lemma}
	\label{L:2ineq}
	Under the assumptions of Theorem \ref{T:solvable}, we have that
	\begin{align}
		\label{E:2ineq-a}
		\lim_{n \to \infty}  \frac{1}{n} \log \left( \Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right) \right)
		&= \log\left(2^{2(b+r)-\frac{b\alpha}{a}}\rho\right)
	\end{align}
	and
	\begin{align}
		\label{E:2ineq-b}
		\lim_{n \to \infty}  \frac{1}{n} \log \left( (n!)^{2(b+r)- \frac{b \alpha}{a} }\Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2  \right)
		&=  \log\left( \left(\frac{2}{2(b+r)-\frac{b\alpha}{a}}\right)^{2(b+r)-\frac{b\alpha}{a}}\right) + \log\rho.
	\end{align}
\end{lemma}
\begin{proof}
	Recall the definition of $T_n$ from above as well as \eqref{E:LogRho} which says that
	\begin{align*}
			\lim_{n \to \infty} \frac{1}{n} \log \left[ \frac{1}{(n!)^2} \int_{(\R^d)^n} \left( \sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+ \frac{\nu}{2}|\sum_{j=k}^n \xi_{\sigma(j)}|^a}\right)^2 \mu (\ud \vec{\xi})\right] = \lim_{n \to \infty} \frac{1}{n} \log \left[ \frac{T_n}{(n!)^2} \right]  = \log\rho.
	\end{align*}
	We already showed in the proof of Theorem \ref{T:solvable} above that
	\begin{align*}
		\int_0^\infty e^{-t} \Norm{ \widetilde{f_n}(\cdot,0,t) }_{ \mathcal{H}^{\otimes n} }^2 \ud t = \Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right)
	\end{align*}
	and then Lemma \ref{L:LNfnineq} above shows that
	\begin{align*}
		\Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right) \le \dfrac{2^{n\left[2(b+r)-\frac{b\alpha}{a}\right]}}{(n!)^2} T_n.
	\end{align*}
	Combining this and \eqref{E:LogRho} we see that
	\begin{align*}
		\limsup_{n \to \infty} \frac{1}{n} & \log\left[ \Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2 \Gamma\left( \left[ 2(b+r)- \frac{b \alpha}{a} \right] n +1 \right)  \right] \\
																			 & \le \log\left( 2^{\left[2(b+r)-\frac{b\alpha}{a}\right]} \right) + \lim_{n \to \infty} \frac{1}{n} \log\left( \frac{T_n}{(n!)^2} \right)                  \\
																			 & = \log\left(  2^{\left[2(b+r)-\frac{b\alpha}{a}\right]} \right)  + \log(\rho)
	\end{align*}
	which proves the upper bound for \eqref{E:2ineq-a}.  \bigskip

	Now we prove the lower bound. Define for $t>0$ and $t'>0$
	\begin{align*}
		J_n(t,t') = \int_{\R^{nd}} H_n(t,x)H_n(t',x) \ud x
	\end{align*}
	and recall Lemma \ref{L:inteq1} which says
	\begin{align*}
		\Norm{\widetilde{f_n}(\cdot,0;t)}_{\mathcal{H}^{\otimes n}}^2 = \frac{1}{(n!)^2} \int_{(\R^{d})^n} H_n(t,\vec{x})^2\ud \vec{x} = \frac{1}{(n!)^2} J_n(t,t).
	\end{align*}
	Thus $J_n(t,t)$ satisfies the scaling property
	\begin{align*}
		J_n(t,t) =  t^{[2(b + r ) -b \alpha / a]n} J_n(1,1).
	\end{align*}
	Let $\tau$ and $\widetilde{\tau}$ be independent exponential random variables with mean 1. By lemma \ref{L:Hneq1},
	\begin{align*}
		\mathbb{E}[J_n(\tau, \tau')] & = \int_0^\infty \int_0^\infty e^{-t}e^{-\widetilde{t}}J_n(t,\widetilde{t}) \ud t \ud \widetilde{t} = \int_{\R^{dn}}\left[ \int_0^\infty e^{-t}H_n(t,x) \ud t \right]^2 \ud x \\
																 & = \int_{\R^{nd}} |\mathcal{F}L_n(\xi_1,...,\xi_n)|^2 \mu(\ud \xi_1) \cdots \mu(\ud \xi_n) =: T_n.
	\end{align*}
	In addition,
	\begin{align*}
		J_n(t,t') \le J_n(t,t)^{1/2}J_n(t',t')^{1/2} =  t^{[2(b + r ) -b \alpha / a]n/2} {t'}^{[2(b + r ) -b \alpha / a]n/2} J_n(1,1).
	\end{align*}
	Combining all of this gives us
	\begin{align*}
		T_n & = \mathbb{E}[J_n(\tau,\widetilde{\tau})] \le \mathbb{E}\left[  {\tau}^{[2(b + r ) -b \alpha / a]n/2} \right] \mathbb{E}\left[  {\widetilde{\tau}}^{[2(b + r ) -b \alpha / a]n/2} \right]J_n(1,1) \\
				& = \Gamma\left( \frac{[2(b + r ) -b \alpha / a]n}{2} \right)^2 J_n(1,1)                                                                                                                           \\
				& = \Gamma\left( \frac{[2(b + r ) -b \alpha / a]n}{2} \right)^2 (n!)^2	\Norm{\widetilde{f_n}(\cdot,0;1) }_{ \mathcal{H}^{\otimes n} }^2,
	\end{align*}
	where the last line follows from Lemma \ref{L:inteq1}. Thus so far we have shown
	\begin{align}
		\label{E:2lowbd}
		\frac{T_n}{(n!)^2} = \frac{\left( \frac{\nu}{2} \right)^{-n\alpha/a} T_n'}{(n!)^2}\le \Gamma\left( \frac{[2(b + r ) -b \alpha / a]n}{2} \right)^2 	\Norm{ \widetilde{f_n}(\cdot,0;1) }_{ \mathcal{H}^{\otimes n} }^2 .
	\end{align}
	Now, another application of Stirling's formula shows that
	\begin{align*}
		\Gamma\left( \frac{[2(b + r ) -b \alpha / a]n}{2} \right)^2  \sim \Gamma\left( [2(b + r ) -b \alpha / a]n +1 \right)2^{- [2(b + r ) -b \alpha / a]n}C_n
	\end{align*}
	where $C_n = 2^{-1}[2(b + r ) -b \alpha / a]^{1/2}(2 \pi n )^{1/2}$. We now infer from \eqref{E:2lowbd} that
	\begin{align*}
		c_{\alpha} \frac{2^{[2(b + r ) -b \alpha / a]n}}{C_n} \frac{1}{(n!)^2} T_n \le \Gamma\left( [2(b + r ) -b \alpha / a]n +1 \right) 	\Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2
	\end{align*}
	where $c_{\alpha}$ is a constant depending on $\alpha$. Since $\frac{1}{n}\log(C_n) \to 0$ as $n \to \infty$, we obtain the lower bound
	\begin{align*}
		\liminf_{n\to \infty} \frac{1}{n} & \log \left( \Gamma\left( [2(b + r ) -b \alpha / a]n +1 	\right) 	\Norm{ \widetilde{f_n}(\cdot,0,1) }_{ \mathcal{H}^{\otimes n} }^2  \right) \\
																			& \ge \log\left(  2^{[2(b + r ) -b \alpha / a]} 	\right) + \log(\rho)
	\end{align*}
	Thus \eqref{E:2ineq-a} is proven. Lastly, \eqref{E:2ineq-b} follows from \eqref{E:2ineq-a} and the fact that
	\begin{align*}
		\lim_{n \to \infty} \frac{1}{n} \log\left[ \frac{\Gamma(an+1)}{(n!)^a} \right] = a\log(a).
	\end{align*}
	This completes the proof of Lemma \ref{L:2ineq}.
\end{proof}


\begin{lemma}
	\label{L:p=2}
	Under the assumptions of Theorem \ref{T:solvable}, we have that
	\begin{equation}
	\lim_{t \to \infty}  \frac{1}{t^\beta} \log \mathbb{E}(|u(t,x)|^2) \nonumber = \left( \frac{2a}{2a(b+r)- b\alpha} \right)^\beta (\theta\rho)^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right).
	\end{equation}
\end{lemma}
\begin{proof}
	Assuming that there is an $L^2(\Omega)$ solution $u(t,x)$ and then applying the scaling property \eqref{E:normscale} gives us
	\begin{align*}
		\mathbb{E}(|u(t,x)|^2) & = \sum_{n\ge 0} 	\theta^n(n!)\Norm{\widetilde{f}_n(\cdot,0,t)}_{\mathcal{H}^{\otimes n}}^2 = \sum_{n \ge 0}\theta^n(n!)t^{(2(b+r)-b\alpha/a)n}\Norm{\widetilde{f}_n(\cdot,0,1)}_{\mathcal{H}^{\otimes n}}^2 \\
                           & = \sum_{n \ge 0}x_nR_nt^{(2(b+r)-b\alpha/a)n}
	\end{align*}
	where
	\begin{align*}
		R_n = (n!)^{2(b+r)-b\alpha/a}\Norm{\widetilde{f}_n(\cdot,0,1)}_{\mathcal{H}^{\otimes n}}^2 \quad \text{and} \quad
		x_n = \frac{\theta^n}{(n!)^{2(b+r)-(b\alpha/a)-1}}.
	\end{align*}
	Notice that \eqref{E:2ineq-b} above says that
	\begin{align*}
		\frac{1}{n}\log(R_n) \to  \log\left( \left(\frac{2}{2(b+r)-\frac{b\alpha}{a}}\right)^{2(b+r)-\frac{b\alpha}{a}}  \rho \right )  \quad \text{as } n\to \infty.
	\end{align*}
	Now define $R$ to be
	\begin{align*}
		R=\left(\frac{2}{2(b+r)-\frac{b\alpha}{a}}\right)^{2(b+r)-\frac{b\alpha}{a}}  \rho.
	\end{align*}
	We want to find a $\beta$ and $A$ so that
	\begin{align*}
		\lim_{t \to \infty}\frac{1}{t^\beta} \log \sum_{n \ge 0} x_nR^n\left(t^{(2(b+r)-b\alpha/a)}\right)^n = A
	\end{align*}
	By Lemma A.3 of \cite{BCC21},
	\begin{align*}
		\lim_{t\to \infty} \left[\frac{1}{(\theta R)t^{2(b+r)-b\alpha/a}}\right]^{\frac{1}{2(b+r)-(b\alpha/a)-1}} \log \sum_{n \ge 0} \frac{\left[ (\theta R) t^{2(b+r)-b\alpha/a}\right]^n}{(n!)^{2(b+r)-(b\alpha/a)-1}} = 2(b+r) - (b\alpha/a) -1
	\end{align*}
	and by an easy algebraic manipulation
	\begin{align*}
		\lim_{t\to \infty} & \left[\frac{1}{t^{2(b+r)-b\alpha/a}}\right]^{\frac{1}{2(b+r)-(b\alpha/a)-1}} \log \sum_{n \ge 0} \frac{\left[ (\theta R) t^{2(b+r)-b\alpha/a}\right]^n}{(n!)^{2(b+r)-(b\alpha/a)-1}} \\
                       & = [2(b+r) - (b\alpha/a) -1](\theta R)^{\frac{1}{2(b+r)-(b\alpha/a)-1}}.
	\end{align*}
	Hence,
	% \begin{align*}
	% 	\beta = \frac{2(b+r)-b\alpha/a}{2(b+r)-(b\alpha/a)-1} \quad \text{and} \quad A=[2(b+r) - (b\alpha/a) -1](\theta R)^{\frac{1}{2(b+r)-(b\alpha/a)-1}}
	% \end{align*}
	% or by removing the fractions in the denominator
	\begin{align*}
		\beta = \frac{2a(b+r)-b\alpha}{2a(b+r)-(b\alpha)-a} \quad \text{and} \quad
		  A   = [2(b+r) - (b\alpha/a) -1](\theta R)^{\frac{a}{2a(b+r)-b\alpha-a}}.
	\end{align*}
	Finally, an application of Lemma A.2 of \cite{BCC21} shows that
	\begin{align*}
		\lim_{t\to \infty} & \frac{1}{t^\beta} \log \mathbb{E}(|u(t,x)|^2) \\
                       & = \theta^{\frac{a}{2a(b+r)-b\alpha-a}}\left( \frac{2a}{2a(b+r)- b\alpha} \right)^\beta \rho^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right),
	\end{align*}
	which completes the proof of Theorem \ref{L:p=2}.
\end{proof}

Now we are ready to prove Theorem \ref{T:ubound}.

\begin{proof}[Proof of Theorem \ref{T:ubound}]
	We denote $u_{\lambda}$ for $\lambda >0$ to be the solution of the fractional equation with
	$\theta$ replaced with $\lambda$ and $u=u_\theta$. Using theorem B.1 of \cite{BCC21}, we see that
	for all $p \ge 2$,
	\begin{align*}
		\Norm{u(t,0)}_p \le \Norm{u_{(p-1)\theta}(t,0)}_2
	\end{align*}
	and by the scaling property
	\begin{align*}
		\Norm{u_{(p-1)\theta}(t,0)}_2^2 &= \sum_{n \ge 0}(n!)^n\theta^n(p-1)^n \Norm{\widetilde{f}_n(\cdot,0,t)}_{\mathcal{H}^{\otimes n}}^2 \\
																		&= \sum_{n \ge 0}(n!)^n\theta^n \Norm{\widetilde{f}_n\left(\cdot,0,t(p-1)^{\frac{1}{2(b+r)-b\alpha/a}}\right)}_{\mathcal{H}^{\otimes n}}^2 \\
																		&= \Norm{u\left(t(p-1)^{\frac{1}{2(b+r)-b\alpha/a}}, 0\right)}_2^2.
	\end{align*}
	Hence,
	\begin{align*}
		\Norm{u(t,0)}_p \le \Norm{u\left(t(p-1)^{\frac{1}{2(b+r)-b\alpha/a}}, 0\right)}_2 =: \Norm{u(t_p,0)}_2
	\end{align*}
	and an application of Lemma \ref{L:p=2} proves Theorem \ref{T:ubound}.
\end{proof}

To remove the absolute value sign in Proposition \ref{P:Wnprop2}, we want to identify all the $f \in \mathcal H$ for which $W_n(t,\mathcal{F}f)$ is nonnegative. In fact, if we consider the space
\begin{align*}
	\mathcal{H}_+ = \{ f\in \mathcal{H} : \text{$f$ is nonnegative and nonnegative definite} \}
\end{align*}
then by Plancherel's theorem
\begin{align*}
		W_n(t,\mathcal{F}f) = \int_{[0,t]^n<} \int_{\R^{nd}} \prod_{k=1}^n(f*\gamma)(x_k) \prod_{k=1} G(s_k - s_{k-1}, x_k-x_{k-1}) \ud \vec{x} \ud \vec{s} =: U_n(t,f)
\end{align*}
with $s_0=0$ and $x_0=0$. Under Assumption \ref{A:param}, the Green's function for the fractional equation is nonnegative and hence under these conditions we get that $W_n(t,\mathcal{F}(f)) \ge 0$ for all the functions $f \in \mathcal{H}_+$. Hence we have that
\begin{align*}
	0 \le W_n(t, \mathcal{F}f) = U_n(t,f) < \infty \quad \text{for all } f \in \mathcal{H}_+ \text{ and for any of the four options in Aummption \ref{A:param}.}
\end{align*}
where the finiteness comes from Proposition \ref{P:Wnprop2}.

Now we define
\begin{align*}
	W_n(t) := \sup_{f\in\mathcal{H}, \Norm{f}_{\mathcal{H}}=1} W_n(t,\mathcal{F}(f)) \quad \text{and} \quad U_n(t) :=  \sup_{f\in\mathcal{H_+}, \Norm{f}_{\mathcal{H}}=1} U_n(t,f)
\end{align*}
and with this we see that
\begin{align*}
	W_n(t) \ge U_n(t) \ge 0
\end{align*}

\begin{theorem}
	If $\tau$ is an exponential random variable with mean $1$, then
	\begin{align*}
		\liminf_{n\to \infty} \frac{1}{n} \log \mathbb{E} (U_n(\tau) ) \ge \log(\rho^{1/2})
	\end{align*}
\end{theorem}
\begin{proof}
We start by letting $\tau$ be an exponential random variable with mean $1$. With this,
	\begin{align*}
		\mathbb{E}(U_u(\tau)) = \int_0^\infty e^{-t} U_n(t) \ud t \ge \sup_{f\in \mathcal{H_+}n \Norm{f}_\mathcal{H} =1} \int_0^\infty e^{-t} U_n(t,f) \ud t
	\end{align*}
For an $f \in \mathcal{H}_+$ and $\Norm{f}_{\mathcal{H}}= 1$ we have that $\mathcal{F}f$ is nonnegative and nonnegative definite which further implies that $\mathcal{F}f$ is even. In addition, Lemma \ref{L:LWn} gives us that
	\begin{align*}
		\int_0^{\infty} e^{-t} U_n(t,f) \ud t = \int_0^\infty e^{-t} W_n(t,\mathcal{F}f) \ud t = \int_{\R^{dn}} \prod_{k=1}^n \mathcal{F}f(\xi_k) \prod_{k=1}^n \frac{1}{1+ \frac{\nu}{2}|\xi_k+\cdots + \xi_n|^a} \mu(\ud \xi_1) \cdots \mu(\ud \xi_n)
	\end{align*}
Modifying the ideas in the proof of Theorem 1.2 \cite{BCR09} gives us the following identity:
	\begin{align*}
		\lim_{n\to \infty}\frac{1}{n} \log \int_{\R^{dn}} \prod_{k=1}^n(\mathcal{F}f)(\xi_k) \prod_{k=1}^n\frac{1}{1+\frac{\nu}{2}|\xi_k+\cdots+\xi_n|^a} \mu(\ud \xi_1) \cdots \mu(\ud \xi_n) = \log[ \rho(\mathcal{F}f)]
	\end{align*}
where $\rho(\cdot)$ is defined on the space of nonnegative and nonnegative definite functions and is given by
	\begin{align*}
		\rho(g) := \sup_{\Norm{h}_{L_2}=1} \int_{\R^d}g(\xi)\left[ \int_{\R^d} \frac{h(\xi + \eta)h(\eta)}{\sqrt{1+\frac{\nu}{2}|\xi+\eta|^a}\sqrt{1+\frac{\nu}{2}|\eta|^a}}  \ud \eta\right] \mu(\ud \xi).
	\end{align*}
Note that since $g$ is nonnegative and since $\mu$ has a nonnegative density $\varphi$ then we may assume $h$ is also nonnegative for the remainder of this proof. With this being said, we see that for any $f \in \mathcal{H}_+$ and $\Norm{f}_H =1$
	\begin{align*}
		\liminf_{n \to \infty} \frac{1}{n}\log \mathbb{E}(U_n(t)) \ge \log \rho(\mathcal{F}f)
	\end{align*}
We now need to calculate
	\begin{align*}
		\sup_{f \in \mathcal{H}_+, \Norm{f}_{\mathcal{H}}=1} \rho(\mathcal{F}f)
	\end{align*}
Consider a nonnegative function $h\in L^2(\R^d)$. The function $h(\cdot)/\sqrt{1+\frac{\nu}{2}|\cdot|^a} \in L^2(\R^d)$ so that $g_h(x) := (2\pi)^{d/2} \mathcal{F}^{-1}\left( \frac{h(\cdot)}{\sqrt{1+\frac{\nu}{2}|\cdot|^a}} \right)(x)$ is well defined. \textcolor{red}{I still need to verify the details that} under these conditions, $g \in W_{1,a}(\R^d)$ with $\Norm{g}_{W_{1,a}}=1$. Here is a sketch of the details. \textcolor{red}{we no longer need this part since we do not need g to be continuous, only need to to be nonnegative and nonnegative definite.}

	\begin{align*}
		\Norm{g}_{W_{1,a}}^2 &= \Norm{g}^2_{L^2} + (2\pi)^{-d}\sum_{k=1}^d \int_{\R^d} \xi_k^a|\mathcal{F}g(\xi)|^2 \ud \xi
		\\&= (2\pi)^{-d}\int_{\R^d}(1+|\xi|^a)|\mathcal{F}g(\xi)|^2 \ud \xi
		\\&= \int_{\R^d} h(\xi)^2 \ud \xi =1
	\end{align*}
where above we used the fact that $\Norm{g}_{L^2}^2 = (2\pi)^{-d}\Norm{\mathcal{F}g}_{L^2}^2$.

\textcolor{red}{We need from this point on.}

Now, for notation let $\widetilde{h}(x) = h(-x)$. Note that
	\begin{align*}
		\int_{\R^d} \frac{h(\xi + \eta)h(\eta)}{\sqrt{1+\frac{\nu}{2}|\xi+\eta|^a}\sqrt{1+\frac{\nu}{2}|\eta|^a}}  \ud \eta&= (2\pi)^{-d}\int_{\R^d} \mathcal{F}g_h(\xi-\eta)\mathcal{F}g_h(-\eta) \ud \eta
		\\& =(2\pi)^{-d} (\mathcal{F}g_h*\widetilde{\mathcal{F}g_h})(\xi)
	\end{align*}
Since $h(\cdot)$ is real valued then $\widetilde{\mathcal{F}g_h} = \overline{\mathcal{F}\bar{g_h}}= \mathcal{F}\bar{g_h} $. Using this and the fact that  $\mathcal{F}(fg) = (2\pi)^{-d}\mathcal{F}(f)*\mathcal{F}(g)$, we see that $(2\pi)^{-d}(\mathcal{F}g_h*\widetilde{\mathcal{F}g_h})(\xi) =\mathcal{F}\left[|g_h|^2\right](\xi) $ and with this we see that
	\begin{align*}
		\rho(\mathcal{F}f) &= \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \int_{\R^d}\mathcal{F}f(\xi)(\mathcal{F}g_h*\widetilde{\mathcal{F}g_h})(\xi) \mu(\ud \xi)
		\\&= \sup_{\Norm{h}_{L^2}=1} \int_{\R^d}\mathcal{F}f(\xi)\mathcal{F}[|g_h|^2](\xi) \mu(\ud \xi)
		\\&= \sup_{\Norm{h}_{L^2}=1}\langle f, |g_h|^2 \rangle _{\mathcal{H}}.
	\end{align*}
With this we see that
	\begin{align*}
		\sup_{ f\in \mathcal{H}, \Norm{f}_{\mathcal{H}=1} } \rho(\mathcal{F}f) = 	\sup_{ f\in \mathcal{H}, \Norm{f}_{\mathcal{H}=1} }\sup_{\Norm{h}_{L^2}=1} \langle f, |g_h|^2 \rangle _{\mathcal{H}} = \sup_{\Norm{h}_{L^2}=1} \langle |g_h|^2, |g_h|^2 \rangle^{1/2} _{\mathcal{H}}
	\end{align*}
where the optimal $f$ is chosen to be $|g_h|^2/\Norm{|g_h|^2}_{\mathcal{H}}$. Now we want to show that
	\begin{equation}\label{E:GH+}
		|g_h|^2 \in \mathcal{H}_+
	\end{equation}
which would imply that the supremum can be restricted to $\mathcal{H_+}$. If this were the case then we would have that
	\begin{align*}
		\sup_{ f \in \mathcal{H}_+, \Norm{f}_{\mathcal{H}}=1 } \rho(\mathcal{F}f) = \sup_{\Norm{h}_{L^2}=1} \langle |g_h|^2, |g_h|^2 \rangle^{1/2} _{\mathcal{H}}
	\end{align*}
and notice that
	\begin{align*}
		\langle |g_h|^2, |g_h|^2 \rangle^{1/2} _{\mathcal{H}} &= \left( \int_{\R^d} \left|\mathcal{F}[|g_h|^2](\xi)\right|^2 \mu(\ud \xi) \right)^{1/2}
		\\&= (2\pi)^{-d} \left( \int_{\R^d}\left| (\mathcal{F}g_h*\widetilde{\mathcal{F}}g_h)(\xi)\right|^2 \mu(\ud \xi) \right)^{1/2}
		\\&=  \left(\int_{\R^d}\left[ \int_{\R^d} \frac{h(\xi + \eta)h(\eta)}{\sqrt{1+\frac{\nu}{2}|\xi+\eta|^a}\sqrt{1+\frac{\nu}{2}|\eta|^a}}  \ud \eta \right]^2 \mu(\ud \xi) \right)^{1/2}
		\\& = \rho^{1/2}
	\end{align*}
which finally leads us to the equality
	\begin{align*}
		\liminf_{n\to \infty} \frac{1}{n} \log \mathbb{E} (U_n(\tau) ) \ge \log(\rho^{1/2}).
	\end{align*}
Hence, we finish the proof by proving \eqref{E:GH+}. First note that from the above we see that
	\begin{align*}
		\int_{\R^d} | \mathcal{F}|g_h|^2(\xi) |^2 \mu(\ud \xi) \le \rho < \infty
	\end{align*}
and so $|g_h|^2 \in \mathcal{H}$. Lastly, since $\mathcal{F}|g_h|^2(\xi)$ is nonnegative for $\xi \in \R^d$ and for nonnegative $h\in L^2(\R^d)$ then $\nu(\ud \xi) = \mathcal{F}|g_h|^2(\xi) \ud \xi$ is a nonnegative tempered measure. The Bochner-Schwarz theorem then implies that $|g_h|^2$ is nonnegative and nonnegative definite. This shows that $|g_h|^2 \in \mathcal{H}_+$.
\end{proof}

We now give scaling properties for $W_n$ and $U_n$.

\begin{lemma}
	\label{L:WU}
	We have that
	\begin{align*}
		W_n(t) = t^{n\frac{b}{a}\left(a- \frac{\alpha}{2}+\frac{a}{b}r \right)} W_n(1) \quad \text{and} \quad
		U_n(t) = t^{n\frac{b}{a}\left(a- \frac{\alpha}{2}+\frac{a}{b}r \right)} U_n(1).
	\end{align*}
\end{lemma}
\begin{proof}
	Let $\phi = \mathcal{F}f$ and $c>0$
	\begin{align*}
		W_n & (t,\phi(c^{-1}\cdot) = \int_{[0,t]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(c^{-1}\xi_k) \prod_{k=1}^n G(s_k - s_{k-1},\cdot)(\xi_k + \cdots \xi_n) \prod_{i=1}^n \varphi(\xi_i)\ud \xi_i\ud s_i                                                                            \\
				& = c^{nd} \int_{[0,t]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(\xi_k) \prod_{k=1}^n G(s_k - s_{k-1},\cdot)(c(\xi_k + \cdots \xi_n)) \prod_{i=1}^n \varphi(c\xi_i)\ud \xi_i\ud s_i                                                                                            \\
				& = c^{nd}c^{-n(d-\alpha)}c^{-n\frac{a}{b}(b+r-1)} \int_{[0,t]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(\xi_k) \prod_{k=1}^n G\left(c^{\frac{b}{a}}(s_k - s_{k-1}),\cdot\right)(\xi_k + \cdots \xi_n) \prod_{i=1}^n \varphi(\xi_i)\ud \xi_i\ud s_i                            \\
				& = c^{nd}c^{-n(d-\alpha)}c^{-n\frac{a}{b}(b+r-1)} c^{-n\frac{a}{b}} \int_{\left[0,c^{\frac{a}{b}}t\right]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(\xi_k) \prod_{k=1}^n G\left(s_k - s_{k-1},\cdot\right)(\xi_k + \cdots \xi_n) \prod_{i=1}^n \varphi(\xi_i)\ud \xi_i\ud s_i \\
				& = c^{-n\left( a- \alpha +\frac{a}{b}r \right)} W_n\left(c^{\frac{a}{b}}t,\phi \right)
	\end{align*}
	Now we apply this with $c^{\frac{a}{b}} = \frac{1}{t}$ to get
	\begin{align*}
		W_n\left(t,\phi(t^{\frac{b}{a}}\cdot) \right)= t^{n\frac{b}{a}\left( a- \alpha +\frac{a}{b}r \right)} W_n\left(1,\phi \right)
	\end{align*}
	Now note that
	\begin{align*}
		\Norm{\phi(t^{\frac{b}{a}} \cdot )}_{L^2(\mu)}^2 = t^{-\frac{\alpha b }{a}}\Norm{\phi}_{L^2(\mu)}^2
	\end{align*}
	and from this we deduce that
	\begin{align*}
		\Norm{t^{\frac{\alpha b }{2 a}}\phi(t^{\frac{b}{a}} \cdot )}_{L^2(\mu)}^2 =  \Norm{\phi}_{L^2(\mu)}^2 = \Norm{f}_{\mathcal{H}}^2
	\end{align*}
	and in addition
	\begin{align*}
		W_n\left(t,t^{\frac{b \alpha}{2 a}}\phi(t^{\frac{a}{b}}\cdot) \right)= t^{n\frac{b}{a}\left( a- \frac{\alpha}{2} +\frac{a}{b}r \right)} W_n\left(1,\phi \right)
	\end{align*}
	and the result follows after taking the sup over all $f \in \mathcal{H}$ with $\Norm{f}_\mathcal{H} =1$.
\end{proof}

\begin{lemma}
	\begin{align*}
		\liminf_{n\to \infty} \frac{1}{n}\log \left[ (n!)^{ \frac{b}{a}(a-\frac{\alpha}{2} + \frac{a}{b}r)  } U_n(1) \right] \ge \log\left[ \frac{\rho^{1/2}}{ \left(\frac{b}{a}(a-\frac{\alpha}{2} + \frac{a}{b}r)\right)^{\frac{b}{a}(a-\frac{\alpha}{2} + \frac{a}{b}r)} } \right]
	\end{align*}
\end{lemma}
\begin{proof}
	Let $\tau$ be an exponential random variable with mean $1$. By \textcolor{magenta}{Lemma 6.7
	(Lemma \ref{L:WU})},
	\begin{align*}
		\mathbb{E}(U_n(\tau)) = \mathbb{E}\left( \tau^{n\frac{b}{a}(a-\frac{\alpha}{2}+\frac{a}{b}r)} \right) U_n(1) = \Gamma\left( n\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) +1 \right) U_n(1)
	\end{align*}
	and by \textcolor{magenta}{Theorem 6.6} we obtain
	\begin{align}\label{E:lb1}
		\liminf_{n\to \infty} \frac{1}{n} \log \left[ \Gamma\left( n\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) +1 \right) U_n(1) \right] \ge \log(\rho^{1/2}).
	\end{align}
	Recall that
	\begin{align*}
		\lim_{n \to \infty} \frac{1}{n}\log\left( \frac{\Gamma(an+1)}{(n!)^a} \right) = a \log(a)
	\end{align*}
	and with this we have that
	\begin{align*}
		\liminf_{n\to \infty} \frac{1}{n} \log \left[ \frac{ \Gamma\left( n\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) +1 \right) }{ (n!)^{\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right)} } \right] = \frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) \log\left( \frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) \right)
	\end{align*}
	and the result is proven after multiplying and dividing $(n!)^{\frac{b}{a}\left(a-\frac{\alpha}{2}
	+\frac{a}{b}r \right)} $ on the inside of the log in \eqref{E:lb1}
\end{proof}


\begin{lemma}
	\label{L:logUnlb}
	For any $k,\theta >0$, there exists a constant $c_1 = c_1(\alpha,\mathcal{M}, k, \theta) >0$ such
	that, by setting $n_t = [c_1 t]$, it holds that
	\begin{equation}\label{E:69ineq}
		\liminf_{t \to \infty} \frac{1}{t} \log( k^{n_t} \theta^{n_t/2}U_{n_t}(t)) \ge
		\left( k \sqrt{\theta} \right)^{\frac{2a}{2ab-\alpha b + 2ar}} \left( \rho^{1/2} \right)^{\frac{2a}{2ab-\alpha b + 2ar}}.
	\end{equation}
\end{lemma}
\begin{proof}
	Let $\epsilon > 0$. \textcolor{magenta}{Lemma 6.8} guarantees the existence of an $N_\epsilon >0$ so for all $n>N_\epsilon$
	\begin{align*}
		(n!)^{\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)} U_n(1) \ge \exp\left( n(\log(R)-\epsilon) \right) = R^n e^{-n\epsilon }
	\end{align*}
	where
	\begin{align*}
		R = \frac{ \rho^{1/2} } { \left( \frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \right)^{\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) }}.
	\end{align*}
	Now fix a $c>0$ and let $n_t := [ct]$. Notice that $n_t \ge N_\epsilon$ for any $t>t_\epsilon := (N_\epsilon +1)/c$.

	Let $t > t_\epsilon$.
	\begin{align*}
		k^{n_t}\theta^{n_t/2}U_{n_t}(t) = 	k^{n_t} \theta^{n_t/2} t^{\frac{b}{a} \left( a-\frac{\alpha}{2} + \frac{a}{b}r \right)n_t } U_{n_t}(1) \ge \frac{ k^{n_t} \theta^{n_t/2} t^{\frac{b}{a} \left( a-\frac{\alpha}{2} + \frac{a}{b}r \right)n_t } }{ (n!)^{\frac{b}{a} \left(a-\frac{\alpha}{2} +\frac{a}{b}r \right)} } R^{n_t}e^{-n_t \epsilon}
	\end{align*}
	Notice that $[ct]/t \to c$ as $t \to \infty$ which means that $n_t/t \to c$ as $t \to \infty$. With this we can say
	\begin{align*}
		\liminf_{t \to \infty} \frac{1}{t} \log \left( k^{n_t} \theta^{n_t/2} U_{n_t}(t) \right) = c \liminf_{t\to \infty} \frac{1}{n_t} \log \left( k^{n_t} \theta^{n_t/2} U_{n_t}(t) \right)  =: I(n_t)
	\end{align*}
	Now,
	\begin{align*}
		I(n_t) & \ge c \liminf_{t\to \infty} \frac{1}{n_t} \log \left( (kR\sqrt{\theta})^{n_t} \frac{ t^{\frac{b}{a} \left( a-\frac{\alpha}{2} + \frac{a}{b}r \right)n_t } }{ (n_t!)^{\frac{b}{a}\left( a -\frac{\alpha}{2} + \frac{a}{b}r \right) }  }\right) - c\epsilon                                                                                                  \\
           & = c \log(k \sqrt{\theta}R) + c \liminf_{t \to \infty} \frac{1}{n_t} \log \left[  \left(\frac{t}{n_t}\right)^{\frac{b}{a}\left(a-\frac{\alpha}{2} + \frac{a}{b}r\right)n_t} \frac{ n_t^{\frac{b}{a}\left( a- \frac{\alpha}{2} + \frac{a}{b}r \right)n_t } } { (n_t!)^{\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) } }\right] - c \epsilon \\
           & = c \log(k \sqrt{\theta}R) - c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \log(c) + c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \liminf_{t \to \infty} \frac{1}{n_t}\log\left(\frac{n_t^{n_t}}{(n_t)!}\right) - c\epsilon                                                                                             \\
           & = c \log(k \sqrt{\theta}R) - c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \log(c) + c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)  - c\epsilon
	\end{align*}
	and letting $\epsilon$ tend to $0$ we wee that
	\begin{align*}
		I(n_t) \ge c \left[ \log(k \sqrt{\theta}R) - \frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \log(c) + \frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \right] =: h(c).
	\end{align*}
	We can easily find the maximal $c$ by solving $h'(c)=0$ and doing so we get $c=(k \sqrt{\theta}R)^{\frac{a}{b\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)}}$. After plugging in this value for $c$ and replacing $R$ we arrive at the following inequality
	\begin{align*}
		I(n_t) \ge (k\sqrt{\theta})^{\frac{a}{b\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)} }(\rho_{new}^{1/2})^{\frac{a}{b\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)}}
	\end{align*}
	and simplifying exponents gives us \eqref{E:69ineq}.
\end{proof}


We are now ready to prove Theorem \ref{T:lbound}.

\begin{proof}[Proof of Theorem \ref{T:lbound}]
	By proposition \ref{P:Wnprop2}, for and $p,q >0$ with $p^{-1}+q^{-1} = 1$ we have that
	\begin{align*}
		\Norm{u(t,0)}_p \ge \exp\left\{ -\frac{1}{2} t_p^\beta \Norm{f}_{\mathcal{H}}^2 \right\}\left| \sum_{n \ge 0} \theta^{n/2} W_n(t_p^\beta, \mathcal{F}f) \right|.
	\end{align*}
	We now take the supremum over all $f \in \mathcal{H}_+$ with $\Norm{f}_{\mathcal{H}}=k >0 $. Recall that $W_n(t,\phi) = k^nW_n(t,\phi/k)$ for $\phi \in L^2(\mu)$ and non-negativity of $W_n(t,\cdot)$ on $\mathcal{H}_+$. Let $c>0$.
	\begin{align*}
		\Norm{u(t,0)}_p & \ge 	\exp\left\{ -\frac{1}{2} t_p^\beta k^2 \right\} \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=k}\sum_{n \ge 0} \theta^{n/2} W_n(t_p^\beta, \mathcal{F}f) \\
                    & = \exp\left\{ -\frac{1}{2} t_p^\beta k^2 \right\} \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=1}\sum_{n \ge 0} k^n \theta^{n/2} W_n(t_p^\beta, \mathcal{F}f) \\
                    & \ge \exp\left\{ -\frac{1}{2} t_p^\beta k^2 \right\} \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=1} k^{n_t} \theta^{{n_t}/2} U_{n_t}(t_p^\beta, f)
	\end{align*}
	where $n_t = [ct_p^\beta]$. Now by choosing $c$ as in Lemma \ref{L:logUnlb} we get that
	\begin{align*}
		\liminf_{n \to \infty} t_p^{-\beta} \log \Norm{u(t,x)}_p & \ge 	\liminf_{n \to \infty} \left( \frac{-\frac{1}{2}t_p^\beta k^2}{t_p^\beta} + \frac{1}{t_p^\beta}\log \left[ \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=1} k^{n_t} \theta^{{n_t}/2} U_{n_t}(t_p^\beta, f) \right]  \right) \\
                                                             & = -\frac{1}{2}k^2 + k^{ \frac{2a}{2ab-\alpha b +2ar} }(\rho \theta)^{\frac{a}{2ab -\alpha b + 2ar}} =: h(k).
	\end{align*}
	By maximizing $h$ for $k>0$, we see that $h$ is maximized at the point
	\begin{align*}
		k^* = \left( \frac{2ab -\alpha b + 2ar}{2aB} \right)^{ \frac{2ab-\alpha b +2ar}{2a-2[2ab-\alpha b + 2ar]}} \quad \text{with $B=(\theta\rho)^{ \frac{a}{2ab-\alpha b + 2ar} }$}.
	\end{align*}
	Inserting $k^*$ into $h$ gives us that
	\begin{align*}
		h(k^*) = B^{\beta}\left( \frac{2a}{2ab - \alpha b +2ar} \right)^\beta \left( \frac{2ab-\alpha b + 2ar -a }{2a} \right)
	\end{align*}
	and plugging in the value for $B$ proves Theorem \ref{T:lbound}.
\end{proof}

	\begin{align*}
		\rho & = \left( \frac{\nu}{2} \right)^{-\alpha/a} \sup_{\Norm{f}_2 =1} \int_{\R^d} \left[ \int_{\R^d} \frac{f^*\left(x+y\right)f^*\left(y\right)}{\sqrt{1+|x+y|^a } \sqrt{1+|y|^a} }  \ud y \right]^2 \varphi(x)\ud x     \\
				 & = \left( \frac{\nu}{2} \right)^{-\alpha/a}  \sup_{\Norm{f^*}_2 =1}  \int_{\R^d} \left[ \int_{\R^d} \frac{f^*\left(x+y\right)f^*\left(y\right)}{\sqrt{1+|x+y|^a } \sqrt{1+|y|^a} }  \ud y \right]^2 \varphi(x)\ud x \\
				 & = \left( \frac{\nu}{2} \right)^{-\alpha/a}  \rho_{2,a}
	\end{align*}
	and \eqref{E:rhoLam} is proven by applying \eqref{E:oldrhovarrep}.

	To prove \eqref{E:LogRho}, we first notice that by applying the substitution $\xi_{\sigma(j)}' =
	\left(\frac{\nu}{2}\right)^{1/a} \xi_{\sigma(j)}$ that
	\begin{align*}
      & \int_{(\R^d)^n} \left[\sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+\frac{\nu}{2}\left|\sum_{j=k}^n \xi_{\sigma(j)} \right|^a}\right]^2 \mu(\ud \vec{\xi}) \\
		= & \left( \frac{\nu}{2} \right)^{-n\alpha/a}  \int_{(\R^d)^n} \left[\sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+\left|\sum_{j=k}^n \xi_{\sigma(j)} \right|^a}\right]^2 \mu(\ud \vec{\xi} )
	\end{align*}
	and now we can directly calculate
	\begin{align*}
		\lim_{n \to \infty} \frac{1}{n} & \log \left[ \frac{1}{(n!)^2} \int_{(\R^d)^n} \left( \sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+ \frac{\nu}{2}|\sum_{j=k}^n \xi_{\sigma(j)}|^a}\right)^2 \mu (\ud \vec{\xi})\right]                                                                            \\
																		& = \lim_{n \to \infty} \frac{1}{n} \log \left[ \frac{1}{(n!)^2}  \left( \frac{\nu}{2} \right)^{-n\alpha/a}  \int_{(\R^d)^n} \left[\sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+\left|\sum_{j=k}^n \xi_{\sigma(j)} \right|^a}\right]^2 \mu(\ud \vec{\xi} )\right] \\
																		& = \log\left[ \left( \frac{\nu}{2} \right)^{-\alpha/a} \right] +  \lim_{n \to \infty} \frac{1}{n} \log \left[ \frac{1}{(n!)^2}   \int_{(\R^d)^n} \left[\sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+\left|\sum_{j=k}^n \xi_{\sigma(j)} \right|^a}\right]^2 \mu(\ud \vec{\xi} )\right]
	\end{align*}
	and because of both Theorem 1.5 \cite{BCR09} and \eqref{E:rhoLam} above,
	\begin{gather*}
		\lim_{n \to \infty} \frac{1}{n} \log \left( \frac{1}{(n!)^2} \int_{(\R^d)^n} \left( \sum_{\sigma \in \Sigma_n} \prod_{k=1}^n \frac{1}{1+ \frac{\nu}{2}|\sum_{j=k}^n \xi_{\sigma(j)}|^a}\right)^2 \mu (\ud \vec{\xi})\right) \\
		= \log\left( \left( \frac{\nu}{2} \right)^{-\alpha/a} \right) + \log\left( \rho_{2,a} \right)
		= \log\left( \rho_{\nu, a} \right).
	\end{gather*}
	This completes the proof of Lemma \ref{L:rhovar}.
\end{proof}
