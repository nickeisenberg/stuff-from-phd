Hi, Nick, here is the answer:

Notice that, by Theorem 4.1 of my paper,
\begin{align*}
    (Z_{\alpha,\beta,\gamma,d}(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}(t,\cdot)(0)  \\
                                              & = t^{\Ceil{\beta}-1} E_{\beta,\Ceil{\beta}}(0) \\
                                              & = \frac{ t^{\Ceil{\beta}-1} }{\Gamma(\Ceil{\beta})}.
\end{align*}
Hence, when $\beta\in(0,1]$, we see that $(Z_{\alpha,\beta,\gamma}(t,\cdot)*1)(x)=1$.
When $\beta\in (1,2]$, 
\begin{align*}
    (Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)(0) \\
                                                & = E_{\beta}(0)                     \\
                                                & = \frac{1}{\Gamma(1)} = 1.
\end{align*}
Hence, under the initial conditions given by \eqref{fspde1}, $J_0(t,x)\equiv 1$.

\begin{NickQuestion}[Scaling Property]
	Suppose that $\mu(\ud \xi) = \varphi(\xi)\ud \xi$ and $\varphi$ satisfies $\varphi(cx) = c^{-(d-\Delta)}$ for some $\Delta \in (0,d)$ and for any $c>0$. Also recall that 
	\[
	\mathcal{F}Y(ct,\cdot)(\xi) = c^{\beta+\gamma -1}\mathcal{F}Y(t,\cdot)(t^{\beta/\alpha} \xi). 
	\]
	Then 
	\[
	\Norm{\tilde{f_n}(\cdot,0,t)}_{H^{\otimes n}}^2 = t^{[2(\beta + \gamma -1)+2-\beta \Delta / \alpha]n} \Norm{\tilde{f_n}(\cdot,0,1)}_{H^{\otimes n}}^2
	\]
	\textbf{Proof:} 
	\begin{align*}
	\Norm{\tilde{f_n}(\cdot,0,t)}_{H^{\otimes n}}^2 = \frac{1}{n!} \sum_{\rho \in \Sigma_n} \int_{\R^{nd}} \mathcal{F}f_n(\cdot,0,t)^2(\vec{\xi}) \varphi(\vec{\xi})\ud \vec{\xi}
	\end{align*}
	where $\varphi(\vec{\xi}) = \varphi(\xi_1)\cdots\varphi(\xi_n)$
	\begin{align*}
	& \int_{\R^{nd}} \mathcal{F}f_n(\cdot,0,t)^2 \varphi(\vec{\xi})\ud \vec{\xi} =  \int_{\R^{nd}} \left( \int_{[0,t]^n<} \overline{\mathcal{F}Y(t-t_n,\cdot)(\xi_1+\cdots+ \xi_n)\cdots \mathcal{F}Y(t_2-t_1,\cdot)(\xi_1)} \right)^2 \varphi(\vec{\xi})\ud \vec{\xi}

	\\&= t^{2n(\beta+\gamma-1)+2n}\int_{\R^{nd}} \left( \int_{[0,1]^n<} \overline{\mathcal{F}Y\left(1-t_n,\cdot\right)(t^{\beta/\alpha}(\xi_1+\cdots+ \xi_n))\cdots \mathcal{F}Y\left(t_2-t_1,\cdot\right)(t^{\beta/\alpha}\xi_1)} \right)^2 \varphi(\vec{\xi})\ud \vec{\xi}

			\mathcal{M}_a(f,\theta)= \sup_{g \in \mathcal{F}_a}\left\{ \langle g^2*f,g^2 \rangle^{1/2} - \frac{\theta}{2}\int_{\R^d}|\xi|^a|\mathcal{F}g(\xi)|^2 \ud \xi   \right\}
		\end{equation}
		
	Now we define $\tilde{g}(x)  = B^{-d/2}g(x/B)$. Its easy to show that $\mathcal{F}\tilde{g}(\xi) = B^{d/2}\mathcal{F}g(B\xi)$. By a change of variables, we can then show that  
	\[
		\int_{\R^d}|\xi|^a|\mathcal{F}\tilde{g}(\xi)|^2 \ud \xi = B^{-a}\int_{\R^d}|\xi|^a|\mathcal{F}g(\xi)|^2 \ud \xi
	\]
	
	Next, by applying the scaling property of $f$, $f(cx) = c^{-\alpha}f(x)$, we can show by again using a change of variables that 
	\begin{align*}
		(\tilde{g}^2*f)(x) &= B^{-d} \int_{\R^d} \tilde{g}^2(y/B)f(x-y)\ud y
		\\ & = \int_{\R^d} g^2(u)f(x-uB)\ud u
		\\& = B^{-\alpha} \int_{\R^d} g^2(u)f\left(\frac{x}{B}-u\right)\ud u
		\\&= B^{-\alpha}(g^2*f)(x/B)
	\end{align*}
	
	This can be used to show that 
	\[
		\langle \tilde{g}^2*f, \tilde{g}^2  \rangle = 	B^{-\alpha}\langle g^2*f, g^2  \rangle
	\]
	
	Now we can rewrite \eqref{var} as the following 
	\[
		\mathcal{M}_a(\Theta f,\theta)=\frac{1}{\theta} \sup_{g \in \mathcal{F}_a}\left\{ \frac{\sqrt{\Theta}}{\theta} B^{\alpha/2} \langle \tilde{g}^2*f,\tilde{g}^2 \rangle^{1/2} - \frac{B^a}{2}\int_{\R^d}|\xi|^a|\mathcal{F}\tilde{g}(\xi)|^2 \ud \xi   \right\}
	\]
	
	and as before, in order to preserve a scaling property, we want to choose $B$ such that 
	\[
		\frac{\sqrt{\Theta}}{\theta} B^{\alpha/2} = B^a
	\]
	and solving for $B$ we get 
	\[
		B= \left(\frac{\sqrt{\Theta}}{\theta}\right)^{\frac{2}{2a-\alpha}}
	\]
	and after plugging this $B$ in we get that 
	\[
			\mathcal{M}_a(\Theta f,\theta) = \Theta^{\frac{a}{2a-\alpha}}\theta^{-\frac{\alpha}{2a-\alpha}}
	\]
\end{NickQuestion}
