\[
    u(t,x) = 1 + I(t,x)

Hi, Nick, here is the answer:

Notice that, by Theorem 4.1 of my paper,
\begin{align*}
    (Z_{\alpha,\beta,\gamma,d}(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}(t,\cdot)(0)  \\
                                              & = t^{\Ceil{\beta}-1} E_{\beta,\Ceil{\beta}}(0) \\
                                              & = \frac{ t^{\Ceil{\beta}-1} }{\Gamma(\Ceil{\beta})}.
\end{align*}
Hence, when $\beta\in(0,1]$, we see that $(Z_{\alpha,\beta,\gamma}(t,\cdot)*1)(x)=1$.
When $\beta\in (1,2]$, 
\begin{align*}
    (Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)(0) \\
                                                & = E_{\beta}(0)                     \\
                                                & = \frac{1}{\Gamma(1)} = 1.
\end{align*}
Hence, under the initial conditions given by \eqref{fspde1}, $J_0(t,x)\equiv 1$.

\begin{NickQuestion}[$\mathcal{H}^{\otimes n}$ Norm Scaling Property]
	Suppose that $\mu(\ud \xi) = \varphi(\xi)\ud \xi$ and $\varphi$ satisfies $\varphi(cx) = c^{-(d-\alpha)}$ for some $\alpha \in (0,d)$ and for any $c>0$. Also recall that 
	\[
	\mathcal{F}Y(ct,\cdot)(\xi) = c^{b+r -1}\mathcal{F}Y(t,\cdot)(t^{b/a} \xi). 
	\]
	Then 
	\begin{equation}\label{nscale}
	\Norm{\tilde{f_n}(\cdot,0,t)}_{H^{\otimes n}}^2 = t^{[2(b + r ) -b \alpha / a]n} \Norm{\tilde{f_n}(\cdot,0,1)}_{H^{\otimes n}}^2
	\end{equation}
	
	\textbf{Proof:} 
	\begin{align*}
	\Norm{\tilde{f_n}(\cdot,0,t)}_{H^{\otimes n}}^2 = \frac{1}{n!} \sum_{\rho \in \Sigma_n} \int_{\R^{nd}} \mathcal{F}f_n(\cdot,0,t)^2(\vec{\xi}) \varphi(\vec{\xi})\ud \vec{\xi}

	where $\varphi(\vec{\xi}) = \varphi(\xi_1)\cdots\varphi(\xi_n)$
	\begin{align*}
	& \int_{\R^{nd}} \mathcal{F}f_n(\cdot,0,t)^2 \varphi(\vec{\xi})\ud \vec{\xi} =  \int_{\R^{nd}} \left( \int_{[0,t]^n<} \overline{\mathcal{F}Y(t-t_n,\cdot)(\xi_1+\cdots+ \xi_n)\cdots \mathcal{F}Y(t_2-t_1,\cdot)(\xi_1)} \right)^2 \varphi(\vec{\xi})\ud \vec{\xi}

	\\&= t^{2n(b+r-1)+2n}\int_{\R^{nd}} \left( \int_{[0,1]^n<} \overline{\mathcal{F}Y\left(1-t_n,\cdot\right)(t^{b/a}(\xi_1+\cdots+ \xi_n))\cdots \mathcal{F}Y\left(t_2-t_1,\cdot\right)(t^{b/a}\xi_1)} \right)^2 \varphi(\vec{\xi})\ud \vec{\xi}
	\\&= t^{2n(b+r-1)+2n}\int_{\R^{nd}} \left( \int_{[0,1]^n<} \overline{\mathcal{F}Y\left(1-t_n,\cdot\right)(\xi_1+\cdots+ \xi_n)\cdots \mathcal{F}Y\left(t_2-t_1,\cdot\right)(\xi_1)} \right)^2 \varphi(t^{-b/a}\vec{\xi})t^{-nd b / a}\ud \vec{\xi}
	\\& = t^{2n(b+r-1)+2n - n b \alpha / a}\int_{\R^{nd}} \left( \int_{[0,1]^n<} \overline{\mathcal{F}Y\left(1-t_n,\cdot\right)(\xi_1+\cdots+ \xi_n)\cdots \mathcal{F}Y\left(t_2-t_1,\cdot\right)(\xi_1)} \right)^2 \varphi(\vec{\xi})\ud \vec{\xi}
	\\& = t^{2n(b+r-1)+2n - n b\alpha / a}\int_{\R^{nd}} \mathcal{F}f_n(\cdot,0,1)(\vec{\xi})^2 \varphi(\vec{\xi})\ud \vec{\xi} 
	\\&=  t^{[2(b+r-1)+2 - b \alpha / a]n}\Norm{\tilde{f_n}(\cdot,0,1)}_{H^{\otimes n}}^2
	\\&= t^{[2(b+r) - b \alpha / a]n}\Norm{\tilde{f_n}(\cdot,0,1)}_{H^{\otimes n}}^2
	\end{align*}
\end{NickQuestion}

\begin{NickQuestion}[Lemma 2.3]
	We define 
		\[
			\mathcal{F}_a = \left\{ g \in H^a(\R^d) : \Norm{g}_{L^2(\R^d)}=1, \mathcal{E}_a(g,g) < \infty \right\}
		\]
	and then we also define 
		\begin{equation} \label{var}
			\mathcal{M}_a(f,\theta)= \sup_{g \in \mathcal{F}_a}\left\{ \langle g^2*f,g^2 \rangle^{1/2} - \frac{\theta}{2}\int_{\R^d}|\xi|^a|\mathcal{F}g(\xi)|^2 \ud \xi   \right\}
		\end{equation}
		
	Now we define $\tilde{g}(x)  = B^{-d/2}g(x/B)$. Its easy to show that $\mathcal{F}\tilde{g}(\xi) = B^{d/2}\mathcal{F}g(B\xi)$. By a change of variables, we can then show that  
	\[
		\int_{\R^d}|\xi|^a|\mathcal{F}\tilde{g}(\xi)|^2 \ud \xi = B^{-a}\int_{\R^d}|\xi|^a|\mathcal{F}g(\xi)|^2 \ud \xi
	\]
	
	Next, by applying the scaling property of $f$, $f(cx) = c^{-\alpha}f(x)$, we can show by again using a change of variables that 
	\begin{align*}
		(\tilde{g}^2*f)(x) &= B^{-d} \int_{\R^d} \tilde{g}^2(y/B)f(x-y)\ud y
		\\ & = \int_{\R^d} g^2(u)f(x-uB)\ud u
