	u_t(0,\cdot) = 0
    \end{cases}
\end{equation}
and the mild solution is defined to be
\begin{equation} \label{weqs}
u(t,x) = 1 + \sqrt{\Theta}\int_0^t \left( \int_{\R^d}G(t-s,x-y)u(s,y) W(\delta y \right) \ud s
\end{equation}
but for general initial conditions for the wave equation we have the solution to be

\begin{align}
    \label{gweqs}
    u(t,x) = \dfrac{\partial}{\partial t} \int_{\R^d}G(t-s,x-y)u(0,y) \ud y &+ \int_{\R^d}G(t-s,x-y)u_t(0,y) \ud y \nonumber \\
									    & + \sqrt{\Theta}\int_0^t \left( \int_{\R^d}G(t-s,x-y)u(s,y) W(\delta y) \right) \ud s
\end{align}

\end{cases}  \label{fspde}
\end{equation}
and we define
\[
J_0(t.x) =

u(t,x) = J_0(t,x) + I(t,x) \label{feqs}

	u(0,\cdot) = 1                                                                                                         & \beta \in (0,1]            \\

    \end{cases}

So if our solution can not be written as
\[
    u(t,x) = 1 + I(t,x)
\]

Hi, Nick, here is the answer:

Notice that, by Theorem 4.1 of my paper,
\begin{align*}
    (Z_{\alpha,\beta,\gamma,d}(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}(t,\cdot)(0)  \\
                                              & = t^{\Ceil{\beta}-1} E_{\beta,\Ceil{\beta}}(0) \\
                                              & = \frac{ t^{\Ceil{\beta}-1} }{\Gamma(\Ceil{\beta})}.
\end{align*}
Hence, when $\beta\in(0,1]$, we see that $(Z_{\alpha,\beta,\gamma}(t,\cdot)*1)(x)=1$.
When $\beta\in (1,2]$,
\begin{align*}
    (Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)(0) \\
                                                & = E_{\beta}(0)                     \\
                                                & = \frac{1}{\Gamma(1)} = 1.
\end{align*}
Hence, under the initial conditions given by \eqref{fspde1}, $J_0(t,x)\equiv 1$.

The function $h(\cdot)/\sqrt{1+|\cdot|^a} \in L^2(\R^d)$ so that $g(x) := (2\pi)^{d/2} \mathcal{F}^{-1}\left( \frac{h(\cdot)}{\sqrt{1+|\cdot|^a}} \right)(x)$ is well defined. \textbf{I still need to verify the details that} under these conditions, $g \in W_{1,a}(\R^d)$ with $\Norm{g}_{W_{1,a}}=1$. Here is a sketch of the details

	\begin{align*}
		\Norm{g}_{W_{1,a}}^2 &= \Norm{g}^2_{L^2} + (2\pi)^{-d}\sum_{k=1}^d \int_{\R^d} \xi_k^a|\mathcal{F}f(\xi)|^2 \ud \xi
		\\&= (2\pi)^{-d}\int_{\R^d}(1+|\xi|^a)|\mathcal{F}f(\xi)|^2 \ud \xi
		\\&= \int_{\R^d} h(\xi)^2 \ud \xi =1
	\end{align*}
	where above we used the fact that $\Norm{g}_{L^2}^2 = (2\pi)^{-d}\Norm{\mathcal{F}g}_{L^2}^2$. 
	
	Now, for notation let $\tilde{h}(x) = h(-x)$. Note that  
	\begin{align*}
		 \int_{\R^d} \frac{h(\xi + \eta)h(\eta)}{\sqrt{1+|\xi+\eta|^a}\sqrt{1+|\eta|^a}}  \ud \eta&= (2\pi)^{-d}\int_{\R^d} \mathcal{F}g(\xi-\eta)\mathcal{F}g(-\eta) \ud \eta
		\\& =(2\pi)^{-d} (\mathcal{F}g*\widetilde{\mathcal{F}g})(\xi) 
	\end{align*}
	Since $h(\cdot)$ is real value then $\widetilde{\mathcal{F}}(g) = \overline{\mathcal{F}\bar{g}}= \mathcal{F}\bar{g} $. Using this we see that $(\mathcal{F}g*\widetilde{\mathcal{F}g})(\xi) =\mathcal{F}\left[|g|^2\right](\xi) $ and with this we see that 
	\begin{align*}
		\rho(\mathcal{F}f) &= \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \int_{\R^d}\mathcal{F}f(\xi)(\mathcal{F}g*\tilde{\mathcal{F}}g)(\xi) \mu(\ud \xi) 
		\\&= \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \int_{\R^d}\mathcal{F}f(\xi)\mathcal{F}[|g|^2](\xi) \mu(\ud \xi) 
		\\& = \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \int_{\R^d}(f*\gamma)(x)|g|^2(x) \ud x
		\\&= \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \langle f, |g|^2 \rangle _{\mathcal{H}}.
	\end{align*}
	With this we see that 
	\[
		\sup_{ f\in \mathcal{H}, \Norm{f}_{\mathcal{H}=1} } \rho(\mathcal{F}f) = 	\sup_{ f\in \mathcal{H}, \Norm{f}_{\mathcal{H}=1} }\sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \langle f, |g|^2 \rangle _{\mathcal{H}} = \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \langle |g|^2, |g|^2 \rangle^{1/2} _{\mathcal{H}}
	\]
	where the optimal $f$ is chosen to be $|g|^2/\Norm{|g|^2}_{\mathcal{H}}$. Therefore the supremum can be restricted to $\mathcal{H_+}$ which implies 
	\[
		\sup_{ f \in \mathcal{H}_+, \Norm{f}_{\mathcal{H}}=1 } \rho(\mathcal{F}f) = \sup_{\Norm{h}_{L^2}=1}(2\pi)^{-d} \langle |g|^2, |g|^2 \rangle^{1/2} _{\mathcal{H}}
	\]
	and notice that 
	\begin{align*}
		(2\pi)^{-d} \langle |g|^2, |g|^2 \rangle^{1/2} _{\mathcal{H}} &= (2\pi)^{-d} \left( \int_{\R^d} \left|\mathcal{F}[|g|^2](\xi)\right|^2 \mu(\ud \xi) \right)^{1/2}
		\\&= (2\pi)^{-d} \left( \int_{\R^d}\left| (\mathcal{F}g*\widetilde{\mathcal{F}}g)(\xi)\right|^2 \mu(\ud \xi) \right)^{1/2}
		\\&=  \left(\int_{\R^d}\left[ \int_{\R^d} \frac{h(\xi + \eta)h(\eta)}{\sqrt{1+|\xi+\eta|^a}\sqrt{1+|\eta|^a}}  \ud \eta \right]^2 \mu(\ud \xi) \right)^{1/2}
	\end{align*}
	which finally leads us to the inequality 
	\[
		\liminf_{n\to \infty} \frac{1}{n} \log \mathbb{E} (U_n(\tau) ) \ge \log(\rho^{1/2}) 
	\]
