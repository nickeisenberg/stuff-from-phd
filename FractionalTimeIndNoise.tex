	Hi, Nick, here is the answer:

	Notice that, by Theorem 4.1 of my paper,
	\begin{align*}
	(Z_{\alpha,\beta,\gamma,d}(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}(t,\cdot)(0)  \\
	& = t^{\Ceil{\beta}-1} E_{\beta,\Ceil{\beta}}(0) \\
	& = \frac{ t^{\Ceil{\beta}-1} }{\Gamma(\Ceil{\beta})}.
	\end{align*}
	Hence, when $\beta\in(0,1]$, we see that $(Z_{\alpha,\beta,\gamma}(t,\cdot)*1)(x)=1$.
	When $\beta\in (1,2]$,
	\begin{align*}
	(Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)*1)(x) & = \calF Z_{\alpha,\beta,\gamma,d}^*(t,\cdot)(0) \\
	& = E_{\beta}(0)                     \\
	& = \frac{1}{\Gamma(1)} = 1.
	\end{align*}
	Hence, under the initial conditions given by \eqref{fspde1}, $J_0(t,x)\equiv 1$.

		\noindent \textbf{Proof of Theorem 1.5:}

		\begin{enumerate}
			\item First is equation (7.21) which says by Parseval's Relation we get
			\begin{equation}\label{typo1}
			\sup_{\langle g, Qg \rangle =1} \int_{\R^d} [(Qg)*(\widetilde{Qg})]^p(-s) \varphi(s) \ud s

			where $\tilde{p}(x) = p(-x)$ and $g(x) = f(x)\sqrt{1+|x|^\beta}$ where $f$ is the same function from the definition of $\rho$. Notice the left hand side of \ref{typo1} has a $(-s)$ as the input of the function. Then when taking the Fourier transform of $[(Qg)*(\widetilde{Qg})]^p(-s) $ we should have the following

			\mathcal{F}[(Qg)*(\widetilde{Qg})]^p(-x)

			\textcolor{magenta}{Here is what is going on here: First of all, for $\tilde{p}(x) = p(-x)$,
				\begin{align*}
				\mathcal{F} \widetilde{p}(\xi) =\int_{\R^d}e^{-ix\xi}p(-x)\ud x =  \int_{\R^d}e^{ix\xi}p(x)\ud x =

				\end{align*}
				Hence,
				\begin{align*}
				\int_{\R^d} \widetilde{f}(x) g(x)\ud x
				= \frac{1}{(2\pi)^d}\int_{\R^d} \calF g (\xi) \overline{\calF \widetilde{f}(\xi)} \ud \xi
				= \frac{1}{(2\pi)^d}\int_{\R^d} \calF g (\xi) \calF f(\xi) \ud \xi.
				\end{align*}}


			\textbf{Nicks Solution:} The centered equation (7.20) in the Xia Chen papers says
			\[
			\int_{\R^d} \left[ \int_{\R^d} (Qg)(s+t)(Qg)(t) \ud t \right]^p \varphi(s) \ud s = \int_{\R^d} \left[ (Qg)*(\widetilde{Qg}) \right]^p(-s) \varphi(s) \ud s
			\]

			and this equality is true, but we can continue is a different way. Using a substitution $u=s+t$ we see that
			\begin{align*}
			\int_{\R^d} \left[ \int_{\R^d} (Qg)(s+t)(Qg)(t) \ud t \right]^p \varphi(s) \ud s &= 	\int_{\R^d} \left[ \int_{\R^d} (Qg)(u)(Qg)(u-s) \ud u \right]^p \varphi(s) \ud s
			\\&= \int_{\R^d} \left[ \int_{\R^d} (Qg)(u)(\widetilde{Qg})(s-u) \ud u \right]^p \varphi(s) \ud s

		\\&= c^{nd} \int_{[0,t]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(\xi_k) \prod_{k=1}^n Y(s_k - s_{k-1},\cdot)(c(\xi_k + \cdots \xi_n)) \varphi(c\vec{u})\ud \vec{u} \ud \vec{s}
		\\&= c^{nd}c^{-n(d-\alpha)}c^{-n\frac{a}{b}(b+r-1)} \int_{[0,t]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(\xi_k) \prod_{k=1}^n Y\left(c^{\frac{b}{a}}(s_k - s_{k-1}),\cdot\right)(\xi_k + \cdots \xi_n) \varphi(c\vec{u})\ud \vec{u} \ud \vec{s}
		\\&= c^{nd}c^{-n(d-\alpha)}c^{-n\frac{a}{b}(b+r-1)} c^{-n\frac{a}{b}} \int_{\left[0,c^{\frac{a}{b}}t\right]^n_<}\int_{\R^{nd}} \prod_{k=1}^n \phi(\xi_k) \prod_{k=1}^n Y\left(s_k - s_{k-1},\cdot\right)(\xi_k + \cdots \xi_n) \varphi(c\vec{u})\ud \vec{u} \ud \vec{s}
		\\&= c^{-n\left( a- \alpha +\frac{a}{b}r \right)} W_n\left(c^{\frac{a}{b}}t,\phi \right)
		\end{align*}
		Now we apply this with $c^{\frac{a}{b}} = \frac{1}{t}$ to get
		\[
		W_n\left(t,\phi(t^{\frac{b}{a}}\cdot) \right)= t^{n\frac{b}{a}\left( a- \alpha +\frac{a}{b}r \right)} W_n\left(1,\phi \right)
		\]
		Now note that
		\[
		\Norm{\phi(t^{\frac{b}{a}} \cdot )}_{L^2(\mu)}^2 = t^{-\frac{\alpha b }{a}}\Norm{\phi}_{L^2(\mu)}^2
		\]
		and from this we deduce that
		\[
		\Norm{t^{\frac{\alpha b }{2 a}}\phi(t^{\frac{b}{a}} \cdot )}_{L^2(\mu)}^2 =  \Norm{\phi}_{L^2(\mu)}^2 = \Norm{f}_{\mathcal{H}}^2
		\]
		and in addition
		\[
		W_n\left(t,t^{\frac{b \alpha}{2 a}}\phi(t^{\frac{a}{b}}\cdot) \right)= t^{n\frac{b}{a}\left( a- \frac{\alpha}{2} +\frac{a}{b}r \right)} W_n\left(1,\phi \right)
		\]
		and the result follows after taking the sup over all $f \in \mathcal{H}$ with $\Norm{f}_\mathcal{H} =1$.
	\end{NickQuestion}

	\begin{NickQuestion}[Lemma 6.8]
		\[
		\liminf_{n\to \infty} \frac{1}{n}\log \left[ (n!)^{ \frac{b}{a}(a-\frac{\alpha}{2} + \frac{a}{b}r)  } U_n(1) \right] \ge \log\left[ \frac{\rho_{new}^{1/2}}{ \left(\frac{b}{a}(a-\frac{\alpha}{2} + \frac{a}{b}r)\right)^{\frac{b}{a}(a-\frac{\alpha}{2} + \frac{a}{b}r)} } \right]
		\]

		\textbf{proof:} Let $\tau$ be an exponential random variable with mean $1$. By lemma 6.7,
		\[
		\mathbb{E}(U_n(\tau)) = \mathbb{E}\left( \tau^{n\frac{b}{a}(a-\frac{\alpha}{2}+\frac{a}{b}r)} \right) U_n(1) = \Gamma\left( n\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) +1 \right) U_n(1)
		\]
		and by Theorem 6.6 we obtain
		\begin{align}\label{lb1}
		\liminf_{n\to \infty} \frac{1}{n} \log \left[ \Gamma\left( n\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) +1 \right) U_n(1) \right] \ge \log(\rho_{new}^{1/2}).
		\end{align}
		Recall that
		\[
		\lim_{n \to \infty} \frac{1}{n}\log\left( \frac{\Gamma(an+1)}{(n!)^a} \right) = a \log(a)
		\]
		and with this we have that
		\[
		\liminf_{n\to \infty} \frac{1}{n} \log \left[ \frac{ \Gamma\left( n\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) +1 \right) }{ (n!)^{\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right)} } \right] = \frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) \log\left( \frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right) \right)
		\]
		and the result is proven after multiplying and dividing $(n!)^{\frac{b}{a}\left(a-\frac{\alpha}{2} +\frac{a}{b}r \right)} $ on the inside of the log in \ref{lb1}
	\end{NickQuestion}

	\begin{NickQuestion}[Lemma 6.9] For any $k,\theta >0$, there exists a constant $c_1 = c_1(\alpha,\mathcal{M}, k, \theta) >0$ such that, by setting $n_t = [c_1 t]$, it holds that
		\begin{equation}\label{69ineq}
		\liminf_{t \to \infty} \frac{1}{t} \log( k^{n_t} \theta^{n_t/2}U_{n_t}(t)) \ge \left( k \sqrt{\theta} \right)^{\frac{2a}{2ab-\alpha b + 2ar}} \left( \rho_{new}^{1/2} \right)^{\frac{2a}{2ab-\alpha b + 2ar}}
		\end{equation}

		\textbf{proof:} Let $\epsilon > 0$. Lemma 6.8 guarantees the existence of an $N_\epsilon >0$ so for all $n>N_\epsilon$
		\[
		(n!)^{\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)} U_n(1) \ge \exp\left( n(\log(R)-\epsilon) \right) = R^n e^{-n\epsilon }
		\]
		where
		\[
		R = \frac{ \rho_{new}^{1/2} } { \left( \frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \right)^{\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) }}.
		\]
		Now fix a $c>0$ and let $n_t := [ct]$. Notice that $n_t \ge N_\epsilon$ for any $t>t_\epsilon := (N_\epsilon +1)/c$.

		Let $t > t_\epsilon$.
		\begin{align*}
		k^{n_t}\theta^{n_t/2}U_{n_t}(t) = 	k^{n_t} \theta^{n_t/2} t^{\frac{b}{a} \left( a-\frac{\alpha}{2} + \frac{a}{b}r \right)n_t } U_{n_t}(1) \ge \frac{ k^{n_t} \theta^{n_t/2} t^{\frac{b}{a} \left( a-\frac{\alpha}{2} + \frac{a}{b}r \right)n_t } }{ (n!)^{\frac{b}{a} \left(a-\frac{\alpha}{2} +\frac{a}{b}r \right)} } R^{n_t}e^{-n_t \epsilon}
		\end{align*}
		Notice that $[ct]/t \to c$ as $t \to \infty$ which means that $n_t/t \to c$ as $t \to \infty$. With this we can say
		\[
		\liminf_{t \to \infty} \frac{1}{t} \log \left( k^{n_t} \theta^{n_t/2} U_{n_t}(t) \right) = c \liminf_{t\to \infty} \frac{1}{n_t} \log \left( k^{n_t} \theta^{n_t/2} U_{n_t}(t) \right)  =: I(n_t)
		\]
		Now,
		\begin{align*}
		I(n_t) & \ge  c \liminf_{t\to \infty} \frac{1}{n_t} \log \left( (kR\sqrt{\theta})^{n_t} \frac{ t^{\frac{b}{a} \left( a-\frac{\alpha}{2} + \frac{a}{b}r \right)n_t } }{ (n_t!)^{\frac{b}{a}\left( a -\frac{\alpha}{2} + \frac{a}{b}r \right) }  }\right) - c\epsilon
		\\&= c \log(k \sqrt{\theta}R) + c \liminf_{t \to \infty} \frac{1}{n_t} \log \left[  \left(\frac{t}{n_t}\right)^{\frac{b}{a}\left(a-\frac{\alpha}{2} + \frac{a}{b}r\right)n_t} \frac{ n_t^{\frac{b}{a}\left( a- \frac{\alpha}{2} + \frac{a}{b}r \right)n_t } } { (n_t!)^{\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) } }\right] - c \epsilon
		\\&=  c \log(k \sqrt{\theta}R) - c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \log(c) + c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \liminf_{t \to \infty} \frac{1}{n_t}\log\left(\frac{n_t^{n_t}}{(n_t)!}\right) - c\epsilon
		\\& =   c \log(k \sqrt{\theta}R) - c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \log(c) + c\frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)  - c\epsilon
		\end{align*}
		and letting $\epsilon$ tend to $0$ we wee that
		\[
		I(n_t) \ge c \left[ \log(k \sqrt{\theta}R) - \frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \log(c) + \frac{b}{a}\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right) \right] =: h(c).
		\]
		We can easily find the maximal $c$ by solving $h'(c)=0$ and doing so we get $c=(k \sqrt{\theta}R)^{\frac{a}{b\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)}}$. After plugging in this value for $c$ and replacing $R$ we arrive at the following inequality
		\[
		I(n_t) \ge (k\sqrt{\theta})^{\frac{a}{b\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)} }(\rho_{new}^{1/2})^{\frac{a}{b\left( a - \frac{\alpha}{2} + \frac{a}{b}r \right)}}
		\]
		and simplifying exponents gives us \ref{69ineq}.
	\end{NickQuestion}

	\begin{NickQuestion}[Theorem 6.1]
		For $\nu=2$ it holds that
		\begin{align} \label{lbound}
		\liminf_{n \to \infty} t_p^{-\beta} &\log \Norm{u(t,x)}_p
		\\ &\ge \left(\frac{1}{2}\right) \nonumber \theta^{\frac{a}{2a(b+r)-b\alpha-a}} \left( \frac{2a}{2a(b+r)- b\alpha} \right)^\beta \rho_{new}^{\frac{a}{2a(b+r)-b\alpha-a}}\left(2(b+r)-\frac{b\alpha}{a}-1\right)
		\end{align}

		\textbf{proof:} By proposition 6.5, for and $p,q >0$ with $p^{-1}+q^{-1} = 1$ we have that
		\[
		\Norm{u(t,0)}_p \ge \exp\left\{ -\frac{1}{2} t_p^\beta \Norm{f}_{\mathcal{H}}^2 \right\}\left| \sum_{n \ge 0} \theta^{n/2} W_n(t_p^\beta, \mathcal{F}f) \right|.
		\]
		We now take the supremum over all $f \in \mathcal{H}_+$ with $\Norm{f}_{\mathcal{H}}=k >0 $. Recall that $W_n(t,\phi) = k^nW_n(t,\phi/k)$ for $\phi \in L^2(\mu)$ and nonnegativity of $W_n(t,\cdot)$ on $\mathcal{H}_+$. Let $c>0$.
		\begin{align*}
		\Norm{u(t,0)}_p &\ge 	\exp\left\{ -\frac{1}{2} t_p^\beta k^2 \right\} \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=k}\sum_{n \ge 0} \theta^{n/2} W_n(t_p^\beta, \mathcal{F}f)
		\\&= \exp\left\{ -\frac{1}{2} t_p^\beta k^2 \right\} \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=1}\sum_{n \ge 0} k^n \theta^{n/2} W_n(t_p^\beta, \mathcal{F}f)
		\\& \ge \exp\left\{ -\frac{1}{2} t_p^\beta k^2 \right\} \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=1} k^{n_t} \theta^{{n_t}/2} U_{n_t}(t_p^\beta, f)
		\end{align*}
		where $n_t = [ct_p^\beta]$. Now by choosing $c$ as in Lemma 6.9 we get that
		\begin{align*}
		\liminf_{n \to \infty} t_p^{-\beta} \log \Norm{u(t,x)}_p &\ge 	\liminf_{n \to \infty} \left( \frac{-\frac{1}{2}t_p^\beta k^2}{t_p^\beta} + \frac{1}{t_p^\beta}\log \left[ \sup_{f\in \mathcal{H}_+ \; \Norm{f}_{\mathcal{H}}=1} k^{n_t} \theta^{{n_t}/2} U_{n_t}(t_p^\beta, f) \right]  \right)
		\\& -\frac{1}{2}k^2 + (k\sqrt{\theta})^{ \frac{2a}{2ab-\alpha b +2ar} }(\rho_{new}^{1/2})^{\frac{2a}{2ab -\alpha b + 2ar}} =: h(k)
		\end{align*}
		By maximizing $h$ we see that $h$ is maximized at the point $k^* = \left( \frac{2ab -\alpha b + 2ar}{2aB} \right)^{ \frac{2ab-\alpha b +2ar}{2a-2[2ab-\alpha b + 2ar]} }$ where $B=(\sqrt{\theta}\rho_{new}^{1/2})^{ \frac{2a}{2ab-\alpha b + 2ar} }$. Inserting $k*$ into $h$ gives us that
		\[
		h(k^*) = B^{\beta}\left( \frac{2a}{2ab - \alpha b +2ar} \right)^\beta \left( \frac{2ab-\alpha b + 2ar -a }{2a} \right)
		\]
		and plugging in the value for $B$ yields \ref{lbound}.
	\end{NickQuestion}
